    96→  { value: "LAPTOP", label: "Laptop", icon: Laptop, color: "text-sky-600 bg-sky-500/10" },
    97→  { value: "MONITOR", label: "Monitor", icon: Monitor, color: "text-violet-600 bg-violet-500/10" },
    98→  { value: "PHONE", label: "Phone", icon: Smartphone, color: "text-emerald-600 bg-emerald-500/10" },
    99→  { value: "TABLET", label: "Tablet", icon: Tablet, color: "text-amber-600 bg-amber-500/10" },
   100→  { value: "KEYBOARD", label: "Keyboard", icon: Keyboard, color: "text-teal-600 bg-teal-500/10" },
   101→  { value: "MOUSE", label: "Mouse", icon: Mouse, color: "text-fuchsia-600 bg-fuchsia-500/10" },
   102→  { value: "HEADSET", label: "Headset", icon: Headset, color: "text-rose-600 bg-rose-500/10" },
   103→  { value: "DESK", label: "Desk", icon: LampDesk, color: "text-orange-600 bg-orange-500/10" },
   104→  { value: "CHAIR", label: "Chair", icon: Armchair, color: "text-lime-600 bg-lime-500/10" },
   105→  { value: "PRINTER", label: "Printer", icon: Printer, color: "text-cyan-600 bg-cyan-500/10" },
   106→  { value: "CAMERA", label: "Camera", icon: Camera, color: "text-pink-600 bg-pink-500/10" },
   107→  { value: "OTHER", label: "Other", icon: Box, color: "text-muted-foreground bg-muted" },
   108→] as const;
   109→
   110→const CONDITIONS = ["NEW", "GOOD", "FAIR", "DAMAGED"] as const;
   111→const STATUSES = ["AVAILABLE", "ASSIGNED", "RETURNED", "RETIRED"] as const;
   112→
   113→type AssetType = (typeof ASSET_TYPES)[number]["value"];
   114→type AssetCondition = (typeof CONDITIONS)[number];
   115→type AssetStatus = (typeof STATUSES)[number];
   116→
   117→const CONDITION_COLOR: Record<AssetCondition, string> = {
   118→  NEW: "bg-emerald-500/15 text-emerald-700 dark:text-emerald-300 border-emerald-500/20",
   119→  GOOD: "bg-sky-500/15 text-sky-700 dark:text-sky-300 border-sky-500/20",
   120→  FAIR: "bg-amber-500/15 text-amber-700 dark:text-amber-300 border-amber-500/20",
   121→  DAMAGED: "bg-rose-500/15 text-rose-700 dark:text-rose-300 border-rose-500/20",
   122→};
   123→
   124→const STATUS_COLOR: Record<AssetStatus, string> = {
   125→  AVAILABLE: "bg-emerald-500/15 text-emerald-700 dark:text-emerald-300 border-emerald-500/20",
   126→  ASSIGNED: "bg-sky-500/15 text-sky-700 dark:text-sky-300 border-sky-500/20",
   127→  RETURNED: "bg-amber-500/15 text-amber-700 dark:text-amber-300 border-amber-500/20",
   128→  RETIRED: "bg-rose-500/15 text-rose-700 dark:text-rose-300 border-rose-500/20",
   129→};
   130→
   131→function getTypeMeta(t: string) {
   132→  return (
   133→    ASSET_TYPES.find((x) => x.value === t) ?? ASSET_TYPES[ASSET_TYPES.length - 1]
   134→  );
   135→}

   165→
   166→export function AssetsModule() {
   167→  const [search, setSearch] = useState("");
   168→  const [type, setType] = useState("");
   169→  const [status, setStatus] = useState("");
   170→  const [view, setView] = useState<"table" | "grid">("table");
   171→
   172→  const [formOpen, setFormOpen] = useState(false);
   173→  const [editAsset, setEditAsset] = useState<Asset | null>(null);
   174→  const [assignAsset, setAssignAsset] = useState<Asset | null>(null);
   175→  const [returnAsset, setReturnAsset] = useState<Asset | null>(null);
   176→
   177→  const qc = useQueryClient();
   178→  const { data, isLoading, isError } = useQuery({
   179→    queryKey: ["assets", search, type, status],
   180→    queryFn: async () => {
   181→      const params = new URLSearchParams();
   182→      if (search) params.set("search", search);
   183→      if (type) params.set("type", type);
   184→      if (status) params.set("status", status);
   185→      const r = await fetch(`/api/assets?${params.toString()}`);
   186→      if (!r.ok) throw new Error("Failed to load assets");
   187→      return r.json();
   188→    },
   189→  });
   190→
   191→  const assets: Asset[] = data?.items ?? [];
   192→  const total = assets.length;
   193→  const assigned = assets.filter((a) => a.status === "ASSIGNED").length;
   194→  const available = assets.filter((a) => a.status === "AVAILABLE").length;
   195→  const damaged = assets.filter((a) => a.condition === "DAMAGED").length;
   196→
   197→  function openCreate() {
   198→    setEditAsset(null);
   199→    setFormOpen(true);
   200→  }
   201→  function openEdit(a: Asset) {
   202→    setEditAsset(a);
   203→    setFormOpen(true);
   204→  }
   205→  async function deleteAsset(a: Asset) {
   206→    if (!confirm(`Permanently delete asset "${a.name}"?`)) return;
   207→    try {
   208→      const r = await fetch(`/api/assets/${a.id}`, { method: "DELETE" });
   209→      if (!r.ok) throw new Error("Failed to delete asset");
   210→      toast.success(`Asset "${a.name}" deleted.`);
   211→      qc.invalidateQueries({ queryKey: ["assets"] });
   212→    } catch (err: any) {
   213→      toast.error(err?.message || "Failed to delete asset.");
   214→    }
   215→  }
   216→  async function retireAsset(a: Asset) {
   217→    if (!confirm(`Retire asset "${a.name}"? It will be marked RETIRED.`))
   218→      return;
   219→    try {
   220→      const r = await fetch(`/api/assets/${a.id}`, {
   221→        method: "PATCH",
   222→        headers: { "Content-Type": "application/json" },
   223→        body: JSON.stringify({ status: "RETIRED" }),
   224→      });
   225→      if (!r.ok) throw new Error("Failed to retire asset");
   226→      toast.success(`Asset "${a.name}" retired.`);
   227→      qc.invalidateQueries({ queryKey: ["assets"] });
   228→    } catch (err: any) {
   229→      toast.error(err?.message || "Failed to retire asset.");
   230→    }
   231→  }
   232→
   233→  return (
   234→    <div className="space-y-6">
   235→      <PageHeader
   236→        title="Asset Management"
   237→        description="Track company assets and equipment assignments"
   238→        icon={<Package className="size-5" />}
   239→        actions={
   240→          <>
   241→            <ExportButton
   242→              module="assets"
   243→              filters={{ search, status, type }}
   244→            />
   245→            <div className="flex rounded-md border border-border overflow-hidden">
   246→              <Button
   247→                variant={view === "table" ? "default" : "ghost"}
   248→                size="sm"
   249→                className="rounded-none gap-1.5"
   250→                onClick={() => setView("table")}
   251→                aria-label="Table view"
   252→              >
   253→                <List className="size-4" />
   254→                <span className="hidden sm:inline">Table</span>
   255→              </Button>
   256→              <Button
   257→                variant={view === "grid" ? "default" : "ghost"}
   258→                size="sm"
   259→                className="rounded-none gap-1.5"
   260→                onClick={() => setView("grid")}
   261→                aria-label="Grid view"
   262→              >
   263→                <LayoutGrid className="size-4" />
   264→                <span className="hidden sm:inline">Grid</span>
   265→              </Button>
   266→            </div>
   267→            <Button size="sm" onClick={openCreate} className="gap-1.5">
   268→              <Plus className="size-4" />
   269→              <span className="hidden sm:inline">Add Asset</span>
   270→              <span className="sm:hidden">New</span>
   271→            </Button>
   272→          </>
   273→        }
   274→      />
   275→
   276→      {/* KPI cards */}
   277→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3 md:gap-4">
   278→        <KpiCard
   279→          label="Total Assets"
   280→          value={total}
   281→          icon={Package}
   282→          iconClass="bg-primary/10 text-primary"
   283→          footer={<span className="text-muted-foreground">All equipment</span>}
   284→        />
   285→        <KpiCard
   286→          label="Assigned"
   287→          value={assigned}
   288→          icon={PackageCheck}
   289→          iconClass="bg-sky-500/15 text-sky-600"
   290→          footer={<span className="text-muted-foreground">In use</span>}
   291→        />
   292→        <KpiCard
   293→          label="Available"
   294→          value={available}
   295→          icon={PackageOpen}
   296→          iconClass="bg-emerald-500/15 text-emerald-600"
   297→          footer={<span className="text-muted-foreground">Free to assign</span>}
   298→        />
   299→        <KpiCard
   300→          label="Damaged"
   301→          value={damaged}
   302→          icon={AlertTriangle}
   303→          iconClass="bg-rose-500/15 text-rose-600"
   304→          footer={<span className="text-muted-foreground">Needs repair</span>}
   305→        />
   306→      </div>
   307→
   308→      {/* Filter bar */}
   309→      <div className="flex flex-col md:flex-row gap-3">
   310→        <div className="relative flex-1">
   311→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   312→          <Input
   313→            placeholder="Search by name, serial, or assignee…"
   314→            value={search}
   315→            onChange={(e) => setSearch(e.target.value)}
   316→            className="pl-9"
   317→          />
   318→        </div>
   319→        <Select
   320→          value={type || "ALL"}
   321→          onValueChange={(v) => setType(v === "ALL" ? "" : v)}
   322→        >
   323→          <SelectTrigger className="md:w-44">
   324→            <SelectValue placeholder="All types" />
   325→          </SelectTrigger>
   326→          <SelectContent>
   327→            <SelectItem value="ALL">All types</SelectItem>
   328→            {ASSET_TYPES.map((t) => (
   329→              <SelectItem key={t.value} value={t.value}>
   330→                {t.label}
   331→              </SelectItem>
   332→            ))}
   333→          </SelectContent>
   334→        </Select>
   335→        <Select
   336→          value={status || "ALL"}
   337→          onValueChange={(v) => setStatus(v === "ALL" ? "" : v)}
   338→        >
   339→          <SelectTrigger className="md:w-44">
   340→            <SelectValue placeholder="All statuses" />
   341→          </SelectTrigger>
   342→          <SelectContent>
   343→            <SelectItem value="ALL">All statuses</SelectItem>
   344→            {STATUSES.map((s) => (
   345→              <SelectItem key={s} value={s}>
   346→                {s.charAt(0) + s.slice(1).toLowerCase()}
   347→              </SelectItem>
   348→            ))}
   349→          </SelectContent>
   350→        </Select>
   351→      </div>
   352→
   353→      {/* Result count */}
   354→      <div className="flex items-center justify-between text-sm">
   355→        <div className="text-muted-foreground">
   356→          {isLoading ? (
   357→            <Skeleton className="h-4 w-32" />
   358→          ) : (
   359→            <span>
   360→              Showing <span className="font-semibold text-foreground">{assets.length}</span>{" "}
   361→              asset{assets.length === 1 ? "" : "s"}
   362→              {(type || status || search) && " (filtered)"}
   363→            </span>
   364→          )}
   365→        </div>
   366→      </div>
   367→
   368→      {/* Content */}
   369→      {isLoading ? (
   370→        <AssetsSkeleton view={view} />
   371→      ) : isError ? (
   372→        <EmptyState
   373→          icon={AlertTriangle}
   374→          title="Failed to load assets"
   375→          description="Please try again. If the problem persists, check the dev server."
   376→          actionLabel="Retry"
   377→          onAction={() => qc.invalidateQueries({ queryKey: ["assets"] })}
   378→        />
   379→      ) : assets.length === 0 ? (
   380→        <EmptyState
   381→          icon={Package}
   382→          title={search || type || status ? "No matching assets" : "No assets yet"}
   383→          description={
   384→            search || type || status
   385→              ? "Try adjusting your filters."
   386→              : "Start by adding your first company asset."
   387→          }
   388→          actionLabel="Add Asset"
   389→          onAction={openCreate}
   390→        />
   391→      ) : view === "table" ? (
   392→        <AssetsTable
   393→          assets={assets}
   394→          onEdit={openEdit}
   395→          onDelete={deleteAsset}
   396→          onAssign={(a) => setAssignAsset(a)}
   397→          onReturn={(a) => setReturnAsset(a)}
   398→          onRetire={retireAsset}
   399→        />
   400→      ) : (
   401→        <AssetsGrid
   402→          assets={assets}
   403→          onEdit={openEdit}
   404→          onDelete={deleteAsset}
   405→          onAssign={(a) => setAssignAsset(a)}
   406→          onReturn={(a) => setReturnAsset(a)}
   407→          onRetire={retireAsset}
   408→        />
   409→      )}
   410→
   411→      {/* Create / Edit dialog */}
   412→      <AssetFormDialog
   413→        open={formOpen}
   414→        onOpenChange={(o) => {
   415→          setFormOpen(o);
   416→          if (!o) setEditAsset(null);
   417→        }}
   418→        asset={editAsset}
   419→        onSaved={() => qc.invalidateQueries({ queryKey: ["assets"] })}
   420→      />
   421→
   422→      {/* Assign dialog */}
   423→      <AssignDialog
   424→        asset={assignAsset}
   425→        onClose={() => setAssignAsset(null)}
   426→        onSaved={() => {
   427→          qc.invalidateQueries({ queryKey: ["assets"] });
   428→          setAssignAsset(null);
   429→        }}
   430→      />
   431→
   432→      {/* Return dialog */}
   433→      <ReturnDialog
   434→        asset={returnAsset}
   435→        onClose={() => setReturnAsset(null)}
   436→        onSaved={() => {
   437→          qc.invalidateQueries({ queryKey: ["assets"] });
   438→          setReturnAsset(null);
   439→        }}
   440→      />
   441→    </div>
   442→  );
   443→}

   445→// =========================================================

   448→
   449→function AssetsTable({
   450→  assets,
   451→  onEdit,
   452→  onDelete,
   453→  onAssign,
   454→  onReturn,
   455→  onRetire,
   456→}: {
   457→  assets: Asset[];
   458→  onEdit: (a: Asset) => void;
   459→  onDelete: (a: Asset) => void;
   460→  onAssign: (a: Asset) => void;
   461→  onReturn: (a: Asset) => void;
   462→  onRetire: (a: Asset) => void;
   463→}) {
   464→  return (
   465→    <Card className="p-0 overflow-hidden border-border/60">
   466→      <div className="overflow-x-auto max-h-[70vh] overflow-y-auto">
   467→        <Table>
   468→          <TableHeader className="sticky top-0 bg-card z-10">
   469→            <TableRow>
   470→              <TableHead>Asset</TableHead>
   471→              <TableHead>Type</TableHead>
   472→              <TableHead className="hidden md:table-cell">Serial</TableHead>

   474→              <TableHead>Assigned To</TableHead>
   475→              <TableHead className="hidden lg:table-cell">Assigned Date</TableHead>
   476→              <TableHead>Status</TableHead>
   477→              <TableHead className="text-right">Actions</TableHead>
   478→            </TableRow>
   479→          </TableHeader>
   480→          <TableBody>
   481→            {assets.map((a) => {
   482→              const tm = getTypeMeta(a.type);
   483→              const TIcon = tm.icon;
   484→              return (
   485→                <TableRow key={a.id} className="hover:bg-muted/30">
   486→                  <TableCell>
   487→                    <div className="flex items-center gap-2.5 min-w-0">
   488→                      <div
   489→                        className={cn(
   490→                          "size-9 rounded-lg flex items-center justify-center flex-shrink-0",
   491→                          tm.color
   492→                        )}
   493→                      >
   494→                        <TIcon className="size-4" />
   495→                      </div>
   496→                      <div className="min-w-0">
   497→                        <div className="font-medium text-sm truncate">{a.name}</div>
   498→                        {a.notes && (
   499→                          <div className="text-xs text-muted-foreground truncate max-w-[200px]">
   500→                            {a.notes.split("\n")[0]}
   501→                          </div>
   502→                        )}
   503→                      </div>
   504→                    </div>
   505→                  </TableCell>
   506→                  <TableCell>
   507→                    <Badge variant="outline" className="gap-1 font-medium">
   508→                      <TIcon className="size-3" />
   509→                      {tm.label}
   510→                    </Badge>
   511→                  </TableCell>
   512→                  <TableCell className="hidden md:table-cell font-mono text-xs">
   513→                    {a.serialNumber || "—"}
   514→                  </TableCell>
   515→                  <TableCell>
   516→                    <Badge
   517→                      variant="outline"
   518→                      className={cn(
   519→                        "font-medium border text-[11px]",
   520→                        CONDITION_COLOR[a.condition]
   521→                      )}
   522→                    >
   523→                      {a.condition.charAt(0) + a.condition.slice(1).toLowerCase()}
   524→                    </Badge>
   525→                  </TableCell>
   526→                  <TableCell>
   527→                    {a.assignedToName ? (
   528→                      <div className="flex items-center gap-2 min-w-0">
   529→                        <AvatarBadge name={a.assignedToName} size="sm" />
   530→                        <div className="min-w-0">
   531→                          <div className="text-sm font-medium truncate max-w-[140px]">
   532→                            {a.assignedToName}
   533→                          </div>
   534→                        </div>
   535→                      </div>
   536→                    ) : (
   537→                      <span className="text-xs text-muted-foreground italic">
   538→                        Available
   539→                      </span>
   540→                    )}
   541→                  </TableCell>
   542→                  <TableCell className="hidden lg:table-cell text-xs text-muted-foreground">
   543→                    {a.assignedDate ? formatDate(a.assignedDate) : "—"}
   544→                  </TableCell>
   545→                  <TableCell>
   546→                    <Badge
   547→                      variant="outline"
   548→                      className={cn(
   549→                        "font-medium border text-[11px]",
   550→                        STATUS_COLOR[a.status]
   551→                      )}
   552→                    >
   553→                      {a.status.charAt(0) + a.status.slice(1).toLowerCase()}
   554→                    </Badge>
   555→                  </TableCell>
   556→                  <TableCell className="text-right">
   557→                    <div className="flex items-center justify-end gap-1">
   558→                      {a.status === "AVAILABLE" && (
   559→                        <Button
   560→                          size="sm"
   561→                          variant="outline"
   562→                          className="h-8 gap-1.5"
   563→                          onClick={() => onAssign(a)}
   564→                        >
   565→                          <UserPlus className="size-3.5" />
   566→                          <span className="hidden xl:inline">Assign</span>
   567→                        </Button>
   568→                      )}
   569→                      {a.status === "ASSIGNED" && (
   570→                        <Button
   571→                          size="sm"
   572→                          variant="outline"
   573→                          className="h-8 gap-1.5"
   574→                          onClick={() => onReturn(a)}
   575→                        >
   576→                          <Undo2 className="size-3.5" />
   577→                          <span className="hidden xl:inline">Return</span>
   578→                        </Button>
   579→                      )}

  1229→function ReturnDialog({
  1230→  asset,
  1231→  onClose,
  1232→  onSaved,
  1233→}: {
  1234→  asset: Asset | null;
  1235→  onClose: () => void;
  1236→  onSaved: () => void;
  1237→}) {
  1238→  const [condition, setCondition] = useState<AssetCondition>("GOOD");
  1239→  const [notes, setNotes] = useState("");
  1240→  const [saving, setSaving] = useState(false);
  1241→
  1242→  const resetKey = asset?.id ?? "none";
  1243→  const [lastKey, setLastKey] = useState(resetKey);
  1244→  if (resetKey !== lastKey) {
  1245→    setLastKey(resetKey);
  1246→    setCondition(asset?.condition ?? "GOOD");

  1250→  async function submit() {

  1257→        method: "POST",

  1259→        body: JSON.stringify(body),
  1260→      });

  1262→        const e = await r.json().catch(() => ({}));
  1263→        throw new Error(e?.error || "Failed to return asset");

  1266→      onSaved();

  1268→      toast.error(err?.message || "Failed to return asset.");
  1269→    } finally {
  1270→      setSaving(false);
  1271→    }
  1272→  }
  1273→
  1274→  return (
  1275→    <Dialog

  1285→            Return Asset

  1300→              <SelectTrigger id="return-condition">

   732→function FunnelBars({
   733→  data,
   734→}: {
   735→  data: { stage: string; count: number; atStage: number }[];
   736→}) {
   737→  const max = Math.max(...data.map((d) => d.count), 1);
   738→  return (
   739→    <div className="space-y-2">
   740→      {data.map((d, idx) => {
   741→        const widthPct = (d.count / max) * 100;
   742→        const prevCount = idx > 0 ? data[idx - 1].count : null;
   743→        const conversion =
   744→          prevCount && prevCount > 0
   745→            ? Math.round((d.count / prevCount) * 100)
   746→            : null;
   747→        const color = CHART_COLORS[idx % CHART_COLORS.length];
   748→        return (
   749→          <div key={d.stage} className="space-y-1">
   750→            <div className="flex items-center justify-between text-xs">
   751→              <div className="flex items-center gap-2">
   752→                <span
   753→                  className="inline-block size-2.5 rounded-sm"
   754→                  style={{ backgroundColor: color }}
   755→                />
   756→                <span className="font-medium capitalize">
   757→                  {d.stage.replace(/_/g, " ").toLowerCase()}
   758→                </span>
   759→                {conversion !== null && (
   760→                  <span className="text-[10px] text-muted-foreground">
   761→                    · {conversion}% conversion
   762→                  </span>
   763→                )}
   764→              </div>
   765→              <div className="flex items-center gap-3">
   766→                <span className="font-semibold tabular-nums">{d.count}</span>
   767→                <span className="text-[10px] text-muted-foreground">
   768→                  ({d.atStage} at stage)
   769→                </span>
   770→              </div>
   771→            </div>
   772→            <div className="h-7 w-full rounded-md bg-muted/40 overflow-hidden">
   773→              <div
   774→                className="h-full rounded-md transition-all flex items-center justify-end px-2"
   775→                style={{
   776→                  width: `${widthPct}%`,
   777→                  backgroundColor: color,
   778→                  minWidth: d.count > 0 ? "8px" : "0",
   779→                }}
   780→              />
   781→            </div>
   782→          </div>
   783→        );
   784→      })}
   785→    </div>
   786→  );
   787→}

   810→// ============================================================
   811→// Generate report dialog (preserved from previous module)
   812→// ============================================================
   813→function GenerateDialog({
   814→  report,
   815→  onClose,
   816→}: {
   817→  report: ReportType;
   818→  onClose: () => void;
   819→}) {
   820→  const today = new Date();
   821→  const monthStart = new Date(today.getFullYear(), today.getMonth(), 1)
   822→    .toISOString()
   823→    .slice(0, 10);
   824→  const monthEnd = today.toISOString().slice(0, 10);
   825→  const [from, setFrom] = useState(monthStart);
   826→  const [to, setTo] = useState(monthEnd);
   827→  const [format, setFormat] = useState("csv");
   828→  const [downloading, setDownloading] = useState(false);
   829→
   830→  async function download() {
   831→    setDownloading(true);
   832→    try {
   833→      const params = new URLSearchParams({
   834→        type: report.key,
   835→        format,
   836→      });
   837→      if (from) params.set("from", from);
   838→      if (to) params.set("to", to);
   839→
   840→      const res = await fetch(`/api/reports/generate?${params.toString()}`);
   841→      if (!res.ok) {
   842→        const err = await res.json().catch(() => ({}));
   843→        throw new Error(err.error || "Failed");
   844→      }
   845→      const blob = await res.blob();
   846→      const ext = format === "pdf" ? "pdf" : format === "excel" ? "xls" : "csv";
   847→      const filename = `${report.key}-report-${new Date().toISOString().split("T")[0]}.${ext}`;
   848→      const url = URL.createObjectURL(blob);
   849→      const a = document.createElement("a");
   850→      a.href = url;
   851→      a.download = filename;
   852→      document.body.appendChild(a);
   853→      a.click();
   854→      document.body.removeChild(a);
   855→      URL.revokeObjectURL(url);
   856→      toast.success(`Report downloaded (${filename})`);
   857→      onClose();
   858→    } catch (e: any) {
   859→      toast.error(e.message || "Failed to generate report");
   860→    } finally {
   861→      setDownloading(false);
   862→    }
   863→  }
   864→
   865→  return (
   866→    <Dialog open onOpenChange={(v) => !v && onClose()}>
   867→      <DialogContent className="max-w-md">
   868→        <DialogHeader>
   869→          <DialogTitle>Generate {report.title}</DialogTitle>
   870→          <DialogDescription>
   871→            Choose a date range and format. Click download to generate.
   872→          </DialogDescription>
   873→        </DialogHeader>
   874→
   875→        <div className="space-y-4">
   876→          <div className="grid grid-cols-2 gap-3">
   877→            <div className="space-y-2">
   878→              <Label>From</Label>
   879→              <Input
   880→                type="date"
   881→                value={from}
   882→                onChange={(e) => setFrom(e.target.value)}
   883→              />
   884→            </div>
   885→            <div className="space-y-2">
   886→              <Label>To</Label>
   887→              <Input
   888→                type="date"
   889→                value={to}
   890→                onChange={(e) => setTo(e.target.value)}
   891→              />
   892→            </div>
   893→          </div>
   894→
   895→          <div className="space-y-2">
   896→            <Label>Format</Label>
   897→            <Select value={format} onValueChange={setFormat}>
   898→              <SelectTrigger>
   899→                <SelectValue />
   900→              </SelectTrigger>
   901→              <SelectContent>
   902→                {FORMATS.map((f) => (
   903→                  <SelectItem key={f.value} value={f.value}>
   904→                    {f.label}
   905→                  </SelectItem>
   906→                ))}
   907→              </SelectContent>
   908→            </Select>
   909→            <p className="text-xs text-muted-foreground">
   910→              {format === "excel" &&
   911→                "Excel exports as CSV with .xls extension for MVP."}
   912→              {format === "pdf" &&
   913→                "PDF generated using a minimal multi-page PDF writer."}
   914→              {format === "csv" &&
   915→                "Comma-separated values, opens in Excel/Sheets."}
   916→            </p>
   917→          </div>
   918→        </div>
   919→
   920→        <DialogFooter>
   921→          <Button variant="outline" onClick={onClose}>
   922→            Cancel
   923→          </Button>
   924→          <Button onClick={download} disabled={downloading}>
   925→            {downloading ? (
   926→              <>
   927→                <Loader2 className="size-4 mr-1.5 animate-spin" /> Generating…
   928→              </>
   929→            ) : (
   930→              <>
   931→                <Download className="size-4 mr-1.5" /> Download
   932→              </>
   933→            )}
   934→          </Button>
   935→        </DialogFooter>
   936→      </DialogContent>
   937→    </Dialog>
   938→  );
   939→}

   202→   202→function AllDocumentsTab({
   203→   203→  onPreview,
   204→   204→  onSendEmail,
   205→   205→}: {
   206→   206→  onPreview: (doc: any) => void;
   207→   207→  onSendEmail: (doc: any) => void;
   208→   208→}) {
   209→   209→  const { data, isLoading } = useQuery({
   210→   210→    queryKey: ["documents", "all-tab"],
   211→   211→    queryFn: async () => {
   212→   212→      const r = await fetch("/api/documents?pageSize=10");
   213→   213→      return r.json();
   214→   214→    },
   215→   215→  });
   216→   216→
   217→   217→  const { data: emailStats } = useQuery({
   218→   218→    queryKey: ["email-logs", "stats"],
   219→   219→    queryFn: async () => {
   220→   220→      const [sent, failed] = await Promise.all([
   221→   221→        fetch("/api/email-logs?status=SENT&pageSize=1").then((r) => r.json()),
   222→   222→        fetch("/api/email-logs?status=FAILED&pageSize=1").then((r) => r.json()),
   223→   223→      ]);
   224→   224→      return { sentToday: sent.total ?? 0, failed: failed.total ?? 0 };
   225→   225→    },
   226→   226→  });
   227→   227→
   228→   228→  const docs = data?.items ?? [];
   229→   229→  const total = data?.total ?? 0;
   230→   230→  const todayStr = new Date().toDateString();
   231→   231→  const generatedToday = docs.filter(
   232→   232→    (d: any) => new Date(d.createdAt).toDateString() === todayStr
   233→   233→  ).length;
   234→   234→  const sentToday = emailStats?.sentToday ?? 0;
   235→   235→  const failedEmails = emailStats?.failed ?? 0;
   236→   236→  const pendingApproval = docs.filter((d: any) =>
   237→   237→    ["PENDING_APPROVAL", "GENERATED"].includes(d.status)
   238→   238→  ).length;
   239→   239→
   240→   240→  return (
   241→   241→    <div className="space-y-4">
   242→   242→      <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-5 gap-3">
   243→   243→        <KpiCard label="Total" value={total} icon={FileStack} iconClass="bg-primary/10 text-primary" />
   244→   244→        <KpiCard label="Generated Today" value={generatedToday} icon={FilePlus} iconClass="bg-violet-500/10 text-violet-600" />
   245→   245→        <KpiCard label="Sent Today" value={sentToday} icon={MailCheck} iconClass="bg-teal-500/10 text-teal-600" />
   246→   246→        <KpiCard label="Pending Approval" value={pendingApproval} icon={Clock} iconClass="bg-amber-500/10 text-amber-600" />
   247→   247→        <KpiCard label="Failed Emails" value={failedEmails} icon={MailX} iconClass="bg-rose-500/10 text-rose-600" />
   248→   248→      </div>
   249→   249→
   250→   250→      <Card className="border-border/60 shadow-soft overflow-hidden">
   251→   251→        <div className="flex items-center justify-between px-5 py-3 border-b border-border bg-muted/30">
   252→   252→          <div className="font-medium text-sm">Recent Documents</div>
   253→   253→        </div>
   254→   254→        <DocumentsTable
   255→   255→          docs={docs}
   256→   256→          loading={isLoading}
   257→   257→          onPreview={onPreview}
   258→   258→          onSendEmail={onSendEmail}
   259→   259→        />
   260→   260→      </Card>
   261→   261→    </div>
   262→   262→  );
   263→   263→}

   286→   286→      if (type) params.set("type", type);

   291→   291→  const templates = data?.items ?? [];
   292→   292→
   293→   293→  async function archiveTemplate(id: string) {
   294→   294→    if (!confirm("Archive this template? It will be hidden from the active list.")) return;
   295→   295→    try {
   296→   296→      const res = await fetch(`/api/document-templates/${id}`, {
   297→   297→        method: "DELETE",
   298→   298→      });
   299→   299→      if (!res.ok) throw new Error("Failed to archive");
   300→   300→      toast.success("Template archived.");
   301→   301→      qc.invalidateQueries({ queryKey: ["document-templates"] });
   302→   302→    } catch (err: any) {
   303→   303→      toast.error(err.message);
   304→   304→    }
   305→   305→  }
   306→   306→
   307→   307→  async function duplicateTemplate(t: any) {

   309→   309→      const res = await fetch("/api/document-templates", {

   313→   313→          ...t,
   314→   314→          name: `${t.name} (Copy)`,
   315→   315→          code: `${t.code}-COPY`,
   316→   316→          status: "DRAFT",
   317→   317→        }),
   318→   318→      });

   320→   320→        const err = await res.json().catch(() => ({}));
   321→   321→        throw new Error(err.error || "Failed");
   322→   322→      }
   323→   323→      toast.success("Template duplicated.");

   326→   326→      toast.error(err.message);
   327→   327→    }
   328→   328→  }

   330→   330→  return (

   369→   369→      )}
   370→   370→
   371→   371→      {!isLoading && templates.length === 0 && (

  1038→  1038→      const t = setTimeout(() => {

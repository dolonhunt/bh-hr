   240→function AllDocumentsTab({
   241→  onPreview,
   242→  onSendEmail,
   243→}: {
   244→  onPreview: (doc: any) => void;
   245→  onSendEmail: (doc: any) => void;
   246→}) {
   247→  const { data, isLoading } = useQuery({
   248→    queryKey: ["documents", "all-tab"],
   249→    queryFn: async () => {
   250→      const r = await fetch("/api/documents?pageSize=10");
   251→      return r.json();
   252→    },
   253→  });
   254→
   255→  const { data: emailStats } = useQuery({
   256→    queryKey: ["email-logs", "stats"],
   257→    queryFn: async () => {
   258→      const [sent, failed] = await Promise.all([
   259→        fetch("/api/email-logs?status=SENT&pageSize=1").then((r) => r.json()),
   260→        fetch("/api/email-logs?status=FAILED&pageSize=1").then((r) => r.json()),
   261→      ]);
   262→      return { sentToday: sent.total ?? 0, failed: failed.total ?? 0 };
   263→    },
   264→  });
   265→
   266→  const docs = data?.items ?? [];
   267→  const total = data?.total ?? 0;
   268→  const todayStr = new Date().toDateString();
   269→  const generatedToday = docs.filter(
   270→    (d: any) => new Date(d.createdAt).toDateString() === todayStr
   271→  ).length;
   272→  const sentToday = emailStats?.sentToday ?? 0;
   273→  const failedEmails = emailStats?.failed ?? 0;
   274→  const pendingApproval = docs.filter((d: any) =>
   275→    ["PENDING_APPROVAL", "GENERATED"].includes(d.status)
   276→  ).length;
   277→
   278→  return (
   279→    <div className="space-y-4">
   280→      <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-5 gap-3 sm:gap-3">
   281→        <KpiCard label="Total" value={total} icon={FileStack} iconClass="bg-primary/10 text-primary" />
   282→        <KpiCard label="Generated Today" value={generatedToday} icon={FilePlus} iconClass="bg-violet-500/10 text-violet-600" />
   283→        <KpiCard label="Sent Today" value={sentToday} icon={MailCheck} iconClass="bg-teal-500/10 text-teal-600" />
   284→        <KpiCard label="Pending Approval" value={pendingApproval} icon={Clock} iconClass="bg-amber-500/10 text-amber-600" />
   285→        <KpiCard label="Failed Emails" value={failedEmails} icon={MailX} iconClass="bg-rose-500/10 text-rose-600" />
   286→      </div>
   287→
   288→      <Card className="border-border/60 shadow-soft overflow-hidden">
   289→        <div className="flex items-center justify-between px-5 py-3 border-b border-border bg-muted/30">
   290→          <div className="font-medium text-sm">Recent Documents</div>
   291→        </div>
   292→        <DocumentsTable
   293→          docs={docs}
   294→          loading={isLoading}
   295→          onPreview={onPreview}
   296→          onSendEmail={onSendEmail}
   297→        />
   298→      </Card>
   299→    </div>
   300→  );
   301→}

   307→function TemplatesTab({
   308→  onEdit,
   309→  onCreate,
   310→}: {
   311→  onEdit: (id: string) => void;
   312→  onCreate: () => void;
   313→}) {
   314→  const qc = useQueryClient();
   315→  const [search, setSearch] = useState("");
   316→  const [type, setType] = useState("");
   317→  const [previewTpl, setPreviewTpl] = useState<any | null>(null);
   318→
   319→  const { data, isLoading } = useQuery({
   320→    queryKey: ["document-templates", "templates-tab", search, type],
   321→    queryFn: async () => {
   322→      const params = new URLSearchParams();
   323→      if (search) params.set("search", search);
   324→      if (type) params.set("type", type);
   325→      const r = await fetch(`/api/document-templates?${params.toString()}`);
   326→      return r.json();
   327→    },
   328→  });
   329→  const templates = data?.items ?? [];
   330→
   331→  async function archiveTemplate(id: string) {
   332→    if (!confirm("Archive this template? It will be hidden from the active list.")) return;
   333→    try {
   334→      const res = await fetch(`/api/document-templates/${id}`, {
   335→        method: "DELETE",
   336→      });
   337→      if (!res.ok) throw new Error("Failed to archive");
   338→      toast.success("Template archived.");
   339→      qc.invalidateQueries({ queryKey: ["document-templates"] });
   340→    } catch (err: any) {
   341→      toast.error(err.message);
   342→    }
   343→  }
   344→
   345→  async function duplicateTemplate(t: any) {
   346→    try {
   347→      const res = await fetch("/api/document-templates", {
   348→        method: "POST",
   349→        headers: { "Content-Type": "application/json" },
   350→        body: JSON.stringify({
   351→          ...t,
   352→          name: `${t.name} (Copy)`,
   353→          code: `${t.code}-COPY`,
   354→          status: "DRAFT",
   355→        }),
   356→      });
   357→      if (!res.ok) {
   358→        const err = await res.json().catch(() => ({}));
   359→        throw new Error(err.error || "Failed");
   360→      }
   361→      toast.success("Template duplicated.");
   362→      qc.invalidateQueries({ queryKey: ["document-templates"] });
   363→    } catch (err: any) {
   364→      toast.error(err.message);
   365→    }
   366→  }
   367→
   368→  return (
   369→    <div className="space-y-4">
   370→      <div className="flex flex-col md:flex-row gap-3">
   371→        <div className="relative flex-1">
   372→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   373→          <Input
   374→            placeholder="Search templates by name, code, description…"
   375→            value={search}
   376→            onChange={(e) => setSearch(e.target.value)}
   377→            className="pl-9"
   378→          />
   379→        </div>
   380→        <Select
   381→          value={type || "ALL"}
   382→          onValueChange={(v) => setType(v === "ALL" ? "" : v)}
   383→        >
   384→          <SelectTrigger className="md:w-56">
   385→            <SelectValue placeholder="All types" />
   386→          </SelectTrigger>
   387→          <SelectContent>
   388→            <SelectItem value="ALL">All types</SelectItem>
   389→            {DOC_TYPES.map((t) => (
   390→              <SelectItem key={t} value={t}>
   391→                {t.replace(/_/g, " ")}
   392→              </SelectItem>
   393→            ))}
   394→          </SelectContent>
   395→        </Select>
   396→        <Button onClick={onCreate} size="sm" className="md:w-auto">
   397→          <Plus className="size-4 mr-1.5" /> Create Template
   398→        </Button>
   399→      </div>
   400→
   401→      {isLoading && (
   402→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   403→          {Array.from({ length: 6 }).map((_, i) => (
   404→            <div key={i} className="h-44 rounded-xl bg-muted/40 animate-pulse" />
   405→          ))}
   406→        </div>
   407→      )}

   410→        <EmptyState
   411→          icon={FileText}
   412→          title="No templates found"
   413→          description="Create your first template to start generating HR documents."
   414→          actionLabel="Create Template"
   415→          onAction={onCreate}
   416→        />
   417→      )}
   418→
   419→      {!isLoading && templates.length > 0 && (
   420→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   421→          {templates.map((t: any) => (
   422→            <Card
   423→              key={t.id}
   424→              className="border-border/60 shadow-soft hover:shadow-card-hover hover:-translate-y-0.5 transition-all overflow-hidden flex flex-col"
   425→            >
   426→              <CardContent className="p-5 flex-1 flex flex-col">
   427→                <div className="flex items-start justify-between gap-2 mb-3">
   428→                  <div className="min-w-0 flex-1">
   429→                    <div className="font-semibold truncate">{t.name}</div>
   430→                    <div className="text-xs text-muted-foreground truncate">
   431→                      {t.description ?? "No description"}
   432→                    </div>
   433→                  </div>
   434→                  <Badge variant="outline" className="font-mono text-[10px]">

   531→function GeneratedTab({
   532→  onPreview,
   533→  onSendEmail,
   534→}: {
   535→  onPreview: (doc: any) => void;
   536→  onSendEmail: (doc: any) => void;
   537→}) {
   538→  const qc = useQueryClient();
   539→  const [search, setSearch] = useState("");
   540→  const [type, setType] = useState("");
   541→  const [status, setStatus] = useState("");
   542→  const [page, setPage] = useState(1);
   543→
   544→  const { data, isLoading } = useQuery({
   545→    queryKey: ["documents", "generated-tab", search, type, status, page],
   546→    queryFn: async () => {
   547→      const params = new URLSearchParams();
   548→      if (search) params.set("search", search);
   549→      if (type) params.set("type", type);
   550→      if (status) params.set("status", status);
   551→      params.set("page", String(page));
   552→      params.set("pageSize", "20");
   553→      const r = await fetch(`/api/documents?${params.toString()}`);
   554→      return r.json();
   555→    },
   556→    placeholderData: (prev) => prev,
   557→  });
   558→  const docs = data?.items ?? [];
   559→  const total = data?.total ?? 0;
   560→  const totalPages = data?.totalPages ?? 1;
   561→
   562→  async function archiveDoc(id: string) {
   563→    if (!confirm("Archive this document?")) return;
   564→    try {
   565→      await fetch(`/api/documents/${id}`, { method: "DELETE" });
   566→      toast.success("Document archived.");
   567→      qc.invalidateQueries({ queryKey: ["documents"] });
   568→    } catch {
   569→      toast.error("Archive failed.");
   570→    }
   571→  }
   572→
   573→  function download(id: string, format: "docx" | "pdf", docNumber: string) {
   574→    const url = `/api/documents/${id}/download?format=${format}`;
   575→    const a = document.createElement("a");
   576→    a.href = url;
   577→    a.download = `${docNumber}.${format}`;
   578→    document.body.appendChild(a);
   579→    a.click();
   580→    document.body.removeChild(a);
   581→    toast.success(`Downloading ${format.toUpperCase()}…`);
   582→  }
   583→
   584→  return (
   585→    <div className="space-y-4">
   586→      <div className="flex flex-col md:flex-row gap-3">
   587→        <div className="relative flex-1">
   588→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   589→          <Input
   590→            placeholder="Search by document number or title…"
   591→            value={search}
   592→            onChange={(e) => {
   593→              setSearch(e.target.value);
   594→              setPage(1);
   595→            }}
   596→            className="pl-9"
   597→          />
   598→        </div>
   599→        <Select

   888→// Status pipeline shown as small horizontal pills.
   889→const STATUS_PIPELINE: { key: string; label: string }[] = [
   890→  { key: "DRAFT", label: "Draft" },
   891→  { key: "GENERATED", label: "Generated" },
   892→  { key: "PENDING_APPROVAL", label: "Pending" },
   893→  { key: "APPROVED", label: "Approved" },
   894→  { key: "ISSUED", label: "Issued" },
   895→  { key: "SENT", label: "Sent" },
   896→];
   897→
   898→function StatusFlowPills({ status }: { status: string }) {
   899→  const currentIdx = STATUS_PIPELINE.findIndex((s) => s.key === status);
   900→  if (status === "ARCHIVED") {
   901→    return <StatusBadge status="ARCHIVED" />;
   902→  }
   903→  if (currentIdx === -1) {
   904→    return <StatusBadge status={status} />;
   905→  }
   906→  return (
   907→    <div className="flex items-center gap-0.5 flex-wrap">
   908→      {STATUS_PIPELINE.map((s, idx) => {
   909→        const isCurrent = idx === currentIdx;
   910→        const isPast = idx < currentIdx;
   911→        return (
   912→          <div key={s.key} className="flex items-center">
   913→            <span
   914→              className={cn(
   915→                "inline-flex items-center px-1.5 h-5 rounded text-[9px] font-medium leading-none",
   916→                isCurrent &&
   917→                  "bg-primary text-primary-foreground shadow-soft",
   918→                isPast && "bg-primary/15 text-primary",
   919→                !isCurrent && !isPast &&
   920→                  "bg-muted text-muted-foreground"
   921→              )}
   922→              title={s.label}
   923→            >
   924→              {s.label}
   925→            </span>
   926→            {idx < STATUS_PIPELINE.length - 1 && (
   927→              <ChevronRight
   928→                className={cn(
   929→                  "size-2.5 mx-0.5 flex-shrink-0",
   930→                  idx < currentIdx ? "text-primary" : "text-muted-foreground/50"
   931→                )}
   932→              />
   933→            )}
   934→          </div>
   935→        );
   936→      })}
   937→    </div>
   938→  );

  1273→function DocumentPreviewDialog({
  1274→  doc,
  1275→  onClose,
  1276→}: {
  1277→  doc: any | null;
  1278→  onClose: () => void;
  1279→}) {
  1280→  const [preview, setPreview] = useState<any | null>(null);
  1281→  const [loading, setLoading] = useState(false);
  1282→
  1283→  useEffect(() => {
  1284→    if (!doc) {
  1285→      // Defer the reset to avoid synchronous setState in effect body.
  1286→      const t = setTimeout(() => {
  1287→        setPreview(null);
  1288→        setLoading(false);
  1289→      }, 0);
  1290→      return () => clearTimeout(t);
  1291→    }
  1292→    let cancelled = false;
  1293→    // Use queueMicrotask so the rule about synchronous setState in effects
  1294→    // doesn't fire — we genuinely want to flip loading=true before the fetch.
  1295→    queueMicrotask(() => {
  1296→      if (!cancelled) setLoading(true);
  1297→    });
  1298→    fetch(`/api/documents/${doc.id}/preview`)
  1299→      .then((r) => r.json())
  1300→      .then((d) => {
  1301→        if (!cancelled) setPreview(d);
  1302→      })
  1303→      .catch(() => toast.error("Failed to load preview"))
  1304→      .finally(() => {
  1305→        if (!cancelled) setLoading(false);
  1306→      });
  1307→    return () => {
  1308→      cancelled = true;
  1309→    };
  1310→  }, [doc]);
  1311→
  1312→  return (
  1313→    <Dialog open={!!doc} onOpenChange={(o) => !o && onClose()}>
  1314→      <DialogContent className="max-w-[95vw] sm:max-w-4xl max-h-[92vh] p-0 gap-0">

  1328→                  <span>{preview.employee?.fullName}</span>

  1330→                  <StatusBadge status={preview.status} />

  1456→function DirectSendEmailDialog({
  1457→  doc,
  1458→  onClose,
  1459→}: {
  1460→  doc: any | null;
  1461→  onClose: () => void;
  1462→}) {
  1463→  const qc = useQueryClient();
  1464→  const [sending, setSending] = useState(false);
  1465→  const [to, setTo] = useState("");
  1466→  const [cc, setCc] = useState("");
  1467→  const [bcc, setBcc] = useState("");
  1468→  const [subject, setSubject] = useState("");
  1469→  const [body, setBody] = useState("");
  1470→
  1471→  useEffect(() => {
  1472→    if (!doc) return;
  1473→    // Fetch full document + template to get email subject/body presets.
  1474→    fetch(`/api/documents/${doc.id}`)
  1475→      .then((r) => r.json())
  1476→      .then((d) => {
  1477→        setTo(d.employee?.officialEmail ?? "");
  1478→        setSubject(
  1479→          d.template?.emailSubject
  1480→            ? d.template.emailSubject.replace(
  1481→                /\{\{company\.name\}\}/g,
  1482→                "the company"
  1483→              )
  1484→            : `Your document - ${d.documentNumber}`
  1485→        );
  1486→        setBody(
  1487→          d.template?.emailBody ||
  1488→            `Dear ${d.employee?.fullName ?? ""},\n\nPlease find attached your document ${d.documentNumber}.\n\nRegards,\nHR Team`
  1489→        );
  1490→      })
  1491→      .catch(() => {
  1492→        setTo("");
  1493→        setSubject(`Your document - ${doc.documentNumber}`);
  1494→        setBody("");
  1495→      });
  1496→  }, [doc]);
  1497→
  1498→  async function handleSend() {
  1499→    if (!doc) return;
  1500→    if (!to || !subject || !body) {
  1501→      toast.error("To, subject and body are required.");
  1502→      return;
  1503→    }
  1504→    setSending(true);
  1505→    try {
  1506→      const res = await fetch(`/api/documents/${doc.id}/send-email`, {
  1507→        method: "POST",
  1508→        headers: { "Content-Type": "application/json" },
  1509→        body: JSON.stringify({ to, cc, bcc, subject, body }),
  1510→      });
  1511→      if (!res.ok) {
  1512→        const err = await res.json().catch(() => ({}));
  1513→        throw new Error(err.error || "Send failed");
  1514→      }
  1515→      toast.success(`Email sent to ${to}`);
  1516→      qc.invalidateQueries({ queryKey: ["documents"] });
  1517→      qc.invalidateQueries({ queryKey: ["email-logs"] });
  1518→      qc.invalidateQueries({ queryKey: ["dashboard"] });
  1519→      onClose();
  1520→    } catch (err: any) {
  1521→      toast.error(err.message);
  1522→    } finally {
  1523→      setSending(false);
  1524→    }
  1525→  }
  1526→
  1527→  return (
  1528→    <Dialog open={!!doc} onOpenChange={(o) => !o && onClose()}>
  1529→      <DialogContent className="max-w-[95vw] sm:max-w-2xl">
  1530→        <DialogHeader>
  1531→          <DialogTitle className="flex items-center gap-2">
  1532→            <Mail className="size-5 text-primary" />
  1533→            Send Document via Email
  1534→          </DialogTitle>
  1535→        </DialogHeader>
  1536→        <div className="space-y-3">
  1537→          <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
  1538→            <div className="md:col-span-2">
  1539→              <label className="text-xs font-medium text-muted-foreground mb-1.5 block">
  1540→                To *
  1541→              </label>
  1542→              <Input
  1543→                type="email"
  1544→                value={to}
  1545→                onChange={(e) => setTo(e.target.value)}
  1546→              />
  1547→            </div>
  1548→            <div>
  1549→              <label className="text-xs font-medium text-muted-foreground mb-1.5 block">
  1550→                CC
  1551→              </label>
  1552→              <Input value={cc} onChange={(e) => setCc(e.target.value)} />
  1553→            </div>
  1554→            <div>
  1555→              <label className="text-xs font-medium text-muted-foreground mb-1.5 block">
  1556→                BCC
  1557→              </label>
  1558→              <Input value={bcc} onChange={(e) => setBcc(e.target.value)} />
  1559→            </div>
  1560→            <div className="md:col-span-2">
  1561→              <label className="text-xs font-medium text-muted-foreground mb-1.5 block">
  1562→                Subject *
  1563→              </label>
  1564→              <Input value={subject} onChange={(e) => setSubject(e.target.value)} />
  1565→            </div>
  1566→            <div className="md:col-span-2">
  1567→              <label className="text-xs font-medium text-muted-foreground mb-1.5 block">
  1568→                Body *
  1569→              </label>
  1570→              <Textarea

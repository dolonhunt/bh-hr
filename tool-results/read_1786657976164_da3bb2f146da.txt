   611→                  <TableBody>
   612→                    {(emp.documents ?? []).map((d: any) => (
   613→                      <TableRow key={d.id}>
   614→                        <TableCell className="font-mono text-xs">
   615→                          {d.documentNumber}
   616→                        </TableCell>
   617→                        <TableCell className="text-sm">{d.type}</TableCell>
   618→                        <TableCell className="text-sm">
   619→                          {formatDate(d.createdAt)}
   620→                        </TableCell>
   621→                        <TableCell>
   622→                          <StatusBadge status={d.status} />
   623→                        </TableCell>
   624→                        <TableCell className="text-right">

   626→                            size="sm"
   627→                            variant="ghost"
   628→                            onClick={() => setModule("documents")}
   629→                          >
   630→                            View
   631→                          </Button>
   632→                        </TableCell>
   633→                      </TableRow>

   635→                    {(emp.documents ?? []).length === 0 && (
   636→                      <TableRow>

   645→                  </TableBody>
   646→                </Table>
   647→              </div>
   648→            </CardContent>
   649→          </Card>
   650→        </TabsContent>
   651→
   652→        {/* Activity */}
   653→        <TabsContent value="activity" className="mt-4">
   654→          <Card className="border-border/60 shadow-soft">
   655→            <CardHeader>
   656→              <CardTitle className="text-base">Activity Timeline</CardTitle>
   657→            </CardHeader>
   658→            <CardContent>
   659→              <div className="space-y-3">
   660→                {(emp.activities ?? []).map((a: any) => (
   661→                  <div
   662→                    key={a.id}
   663→                    className="flex items-start gap-3 pb-3 border-b border-border/40 last:border-0"
   664→                  >
   665→                    <div className="size-8 rounded-lg bg-primary/10 text-primary flex items-center justify-center flex-shrink-0">
   666→                      <Activity className="size-4" />
   667→                    </div>
   668→                    <div className="flex-1">
   669→                      <div className="text-sm font-medium">{a.title}</div>
   670→                      {a.description && (
   671→                        <div className="text-xs text-muted-foreground">
   672→                          {a.description}
   673→                        </div>
   674→                      )}
   675→                      <div className="text-[11px] text-muted-foreground mt-0.5">
   676→                        {relativeTime(a.createdAt)}
   677→                      </div>
   678→                    </div>
   679→                  </div>
   680→                ))}
   681→                {(emp.activities ?? []).length === 0 && (
   682→                  <p className="text-sm text-muted-foreground py-8 text-center">
   683→                    No activity yet.
   684→                  </p>
   685→                )}
   686→              </div>
   687→            </CardContent>
   688→          </Card>
   689→        </TabsContent>
   690→      </Tabs>
   691→
   692→      <EmployeeFormDialog
   693→        open={editOpen}
   694→        onOpenChange={setEditOpen}
   695→        employee={{ id }}
   696→        onSaved={() => qc.invalidateQueries({ queryKey: ["employee", id] })}
   697→      />
   698→    </div>
   699→  );
   700→}

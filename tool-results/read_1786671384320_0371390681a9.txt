   222→   222→function useCourses(search: string, status: string) {
   223→   223→  return useQuery({
   224→   224→    queryKey: ["training-courses", search, status],
   225→   225→    queryFn: async () => {
   226→   226→      const params = new URLSearchParams();
   227→   227→      if (search) params.set("search", search);
   228→   228→      if (status) params.set("status", status);
   229→   229→      const r = await fetch(`/api/training?${params.toString()}`);
   230→   230→      if (!r.ok) throw new Error("Failed to load training courses");
   231→   231→      return r.json();
   232→   232→    },
   233→   233→  });
   234→   234→}
   235→   235→
   236→   236→function useEnrollments(search: string, status: string) {
   237→   237→  return useQuery({
   238→   238→    queryKey: ["training-enrollments", search, status],
   239→   239→    queryFn: async () => {
   240→   240→      const params = new URLSearchParams();
   241→   241→      if (search) params.set("search", search);
   242→   242→      if (status) params.set("status", status);
   243→   243→      const r = await fetch(
   244→   244→        `/api/training/enrollments?${params.toString()}`

   246→   246→      if (!r.ok) throw new Error("Failed to load enrollments");

   249→   249→  });

   252→   252→function CoursesTab() {
   253→   253→  const qc = useQueryClient();

   255→   255→  const [status, setStatus] = useState("");

   257→   257→  const [formOpen, setFormOpen] = useState(false);
   258→   258→  const [editCourse, setEditCourse] = useState<Course | null>(null);
   259→   259→  const [enrollCourse, setEnrollCourse] = useState<Course | null>(null);
   260→   260→
   261→   261→  const { data, isLoading, isError } = useCourses(search, status);
   262→   262→
   263→   263→  const courses: Course[] = data?.items ?? [];
   264→   264→  const enrollmentsQ = useEnrollments("", "");
   265→   265→  const allEnrollments: Enrollment[] = enrollmentsQ.data?.items ?? [];
   266→   266→
   267→   267→  // KPIs
   268→   268→  const activeCourses = courses.filter(
   269→   269→    (c) => c.status === "SCHEDULED" || c.status === "IN_PROGRESS"
   270→   270→  ).length;
   271→   271→  const totalEnrollments = allEnrollments.filter(
   272→   272→    (e) => e.status !== "DROPPED"
   273→   273→  ).length;
   274→   274→  const completedEnrollments = allEnrollments.filter(
   275→   275→    (e) => e.status === "COMPLETED"
   276→   276→  ).length;
   277→   277→  const completionRate =
   278→   278→    totalEnrollments > 0
   279→   279→      ? Math.round((completedEnrollments / totalEnrollments) * 100)
   280→   280→      : 0;
   281→   281→
   282→   282→  // Upcoming = startDate within next 7 days
   283→   283→  const now = Date.now();
   284→   284→  const in7d = now + 7 * 24 * 60 * 60 * 1000;
   285→   285→  const upcoming = courses.filter((c) => {
   286→   286→    if (!c.startDate) return false;
   287→   287→    const d = new Date(c.startDate).getTime();
   288→   288→    return d >= now && d <= in7d;
   289→   289→  }).length;

   299→   299→  async function deleteCourse(c: Course) {
   300→   300→    if (
   301→   301→      !confirm(
   302→   302→        `Permanently delete course "${c.title}"? This will also remove ${c.enrolledCount} enrollment(s).`
   303→   303→      )
   304→   304→    )
   305→   305→      return;
   306→   306→    try {
   307→   307→      const r = await fetch(`/api/training/${c.id}`, { method: "DELETE" });
   308→   308→      if (!r.ok) throw new Error("Failed to delete course");
   309→   309→      toast.success(`Course "${c.title}" deleted.`);
   310→   310→      qc.invalidateQueries({ queryKey: ["training-courses"] });
   311→   311→      qc.invalidateQueries({ queryKey: ["training-enrollments"] });
   312→   312→    } catch (err: any) {
   313→   313→      toast.error(err?.message || "Failed to delete course.");
   314→   314→    }

   481→   481→      {/* Enroll dialog */}

   489→   489→        }}

   493→   493→}

  1509→  1509→
  1510→  1510→  const resetKey = enrollment?.id ?? "none";
  1511→  1511→  const [lastKey, setLastKey] = useState(resetKey);
  1512→  1512→  if (resetKey !== lastKey) {
  1513→  1513→    setLastKey(resetKey);

  1515→  1515→      enrollment && enrollment.score !== null ? String(enrollment.score) : ""

  1518→  1518→  }
  1519→  1519→
  1520→  1520→  async function submit() {

  1540→  1540→      onSaved();
  1541→  1541→    } catch (err: any) {

   222→function useCourses(search: string, status: string) {
   223→  return useQuery({
   224→    queryKey: ["training-courses", search, status],
   225→    queryFn: async () => {
   226→      const params = new URLSearchParams();
   227→      if (search) params.set("search", search);
   228→      if (status) params.set("status", status);
   229→      const r = await fetch(`/api/training?${params.toString()}`);
   230→      if (!r.ok) throw new Error("Failed to load training courses");
   231→      return r.json();
   232→    },
   233→  });
   234→}
   235→
   236→function useEnrollments(search: string, status: string) {
   237→  return useQuery({
   238→    queryKey: ["training-enrollments", search, status],
   239→    queryFn: async () => {
   240→      const params = new URLSearchParams();
   241→      if (search) params.set("search", search);
   242→      if (status) params.set("status", status);
   243→      const r = await fetch(
   244→        `/api/training/enrollments?${params.toString()}`
   245→      );
   246→      if (!r.ok) throw new Error("Failed to load enrollments");
   247→      return r.json();
   248→    },
   249→  });
   250→}

   252→function CoursesTab() {
   253→  const qc = useQueryClient();
   254→  const [search, setSearch] = useState("");
   255→  const [status, setStatus] = useState("");
   256→
   257→  const [formOpen, setFormOpen] = useState(false);
   258→  const [editCourse, setEditCourse] = useState<Course | null>(null);
   259→  const [enrollCourse, setEnrollCourse] = useState<Course | null>(null);
   260→
   261→  const { data, isLoading, isError } = useCourses(search, status);
   262→
   263→  const courses: Course[] = data?.items ?? [];
   264→  const enrollmentsQ = useEnrollments("", "");
   265→  const allEnrollments: Enrollment[] = enrollmentsQ.data?.items ?? [];
   266→
   267→  // KPIs
   268→  const activeCourses = courses.filter(
   269→    (c) => c.status === "SCHEDULED" || c.status === "IN_PROGRESS"
   270→  ).length;
   271→  const totalEnrollments = allEnrollments.filter(
   272→    (e) => e.status !== "DROPPED"
   273→  ).length;
   274→  const completedEnrollments = allEnrollments.filter(
   275→    (e) => e.status === "COMPLETED"
   276→  ).length;
   277→  const completionRate =
   278→    totalEnrollments > 0
   279→      ? Math.round((completedEnrollments / totalEnrollments) * 100)
   280→      : 0;
   281→
   282→  // Upcoming = startDate within next 7 days
   283→  const now = Date.now();
   284→  const in7d = now + 7 * 24 * 60 * 60 * 1000;
   285→  const upcoming = courses.filter((c) => {
   286→    if (!c.startDate) return false;
   287→    const d = new Date(c.startDate).getTime();
   288→    return d >= now && d <= in7d;
   289→  }).length;
   290→
   291→  function openCreate() {
   292→    setEditCourse(null);
   293→    setFormOpen(true);
   294→  }
   295→  function openEdit(c: Course) {
   296→    setEditCourse(c);
   297→    setFormOpen(true);
   298→  }
   299→  async function deleteCourse(c: Course) {
   300→    if (
   301→      !confirm(
   302→        `Permanently delete course "${c.title}"? This will also remove ${c.enrolledCount} enrollment(s).`
   303→      )
   304→    )
   305→      return;
   306→    try {
   307→      const r = await fetch(`/api/training/${c.id}`, { method: "DELETE" });
   308→      if (!r.ok) throw new Error("Failed to delete course");
   309→      toast.success(`Course "${c.title}" deleted.`);
   310→      qc.invalidateQueries({ queryKey: ["training-courses"] });
   311→      qc.invalidateQueries({ queryKey: ["training-enrollments"] });
   312→    } catch (err: any) {
   313→      toast.error(err?.message || "Failed to delete course.");
   314→    }
   315→  }
   316→  async function setCourseStatus(c: Course, s: CourseStatus) {
   317→    try {
   318→      const r = await fetch(`/api/training/${c.id}`, {
   319→        method: "PATCH",
   320→        headers: { "Content-Type": "application/json" },
   321→        body: JSON.stringify({ status: s }),
   322→      });
   323→      if (!r.ok) throw new Error("Failed to update status");
   324→      toast.success(`Course marked as ${s}.`);
   325→      qc.invalidateQueries({ queryKey: ["training-courses"] });
   326→    } catch (err: any) {
   327→      toast.error(err?.message || "Failed to update status.");
   328→    }
   329→  }
   330→
   331→  return (
   332→    <div className="space-y-6">
   333→      {/* KPI cards */}
   334→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3 md:gap-4">
   335→        <KpiCard
   336→          label="Active Courses"
   337→          value={activeCourses}
   338→          icon={BookOpen}
   339→          iconClass="bg-primary/10 text-primary"
   340→          footer={<span className="text-muted-foreground">Scheduled + ongoing</span>}
   341→        />
   342→        <KpiCard
   343→          label="Total Enrollments"
   344→          value={totalEnrollments}
   345→          icon={Users}
   346→          iconClass="bg-sky-500/15 text-sky-600"
   347→          footer={<span className="text-muted-foreground">Active learners</span>}
   348→        />
   349→        <KpiCard
   350→          label="Completion Rate"
   351→          value={`${completionRate}%`}
   352→          icon={Award}
   353→          iconClass="bg-emerald-500/15 text-emerald-600"
   354→          footer={
   355→            <span className="text-muted-foreground">
   356→              {completedEnrollments} of {totalEnrollments} completed
   357→            </span>
   358→          }
   359→        />
   360→        <KpiCard
   361→          label="Upcoming (7d)"
   362→          value={upcoming}
   363→          icon={CalendarClock}
   364→          iconClass="bg-amber-500/15 text-amber-600"
   365→          footer={<span className="text-muted-foreground">Starting soon</span>}
   366→        />
   367→      </div>
   368→
   369→      {/* Filter bar */}
   370→      <div className="flex flex-col md:flex-row gap-3">
   371→        <div className="relative flex-1">
   372→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   373→          <Input
   374→            placeholder="Search courses by title, trainer, or category…"
   375→            value={search}
   376→            onChange={(e) => setSearch(e.target.value)}
   377→            className="pl-9"
   378→          />
   379→        </div>
   380→        <Select
   381→          value={status || "ALL"}
   382→          onValueChange={(v) => setStatus(v === "ALL" ? "" : v)}
   383→        >
   384→          <SelectTrigger className="md:w-44">
   385→            <SelectValue placeholder="All statuses" />
   386→          </SelectTrigger>
   387→          <SelectContent>
   388→            <SelectItem value="ALL">All statuses</SelectItem>
   389→            {COURSE_STATUSES.map((s) => (
   390→              <SelectItem key={s} value={s}>
   391→                {s.charAt(0) + s.slice(1).toLowerCase().replace(/_/g, " ")}
   392→              </SelectItem>
   393→            ))}
   394→          </SelectContent>
   395→        </Select>
   396→        <Button size="sm" onClick={openCreate} className="gap-1.5 md:ml-1">
   397→          <Plus className="size-4" />
   398→          <span className="hidden sm:inline">Create Course</span>
   399→          <span className="sm:hidden">New</span>
   400→        </Button>
   401→      </div>
   402→
   403→      {/* Result count */}
   404→      <div className="flex items-center justify-between text-sm">
   405→        <div className="text-muted-foreground">
   406→          {isLoading ? (
   407→            <Skeleton className="h-4 w-32" />
   408→          ) : (
   409→            <span>
   410→              Showing{" "}
   411→              <span className="font-semibold text-foreground">{courses.length}</span>{" "}
   412→              course{courses.length === 1 ? "" : "s"}
   413→              {(search || status) && " (filtered)"}
   414→            </span>
   415→          )}
   416→        </div>
   417→      </div>
   418→
   419→      {/* Content */}
   420→      {isLoading ? (
   421→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   422→          {Array.from({ length: 6 }).map((_, i) => (
   423→            <Card key={i} className="p-4 h-56">
   424→              <Skeleton className="h-5 w-2/3" />
   425→              <Skeleton className="h-3 w-1/3 mt-2" />
   426→              <Skeleton className="h-3 w-full mt-3" />
   427→              <Skeleton className="h-3 w-3/4 mt-1" />
   428→              <Skeleton className="h-2 w-full mt-4" />
   429→              <Skeleton className="h-8 w-full mt-4" />
   430→            </Card>
   431→          ))}
   432→        </div>
   433→      ) : isError ? (
   434→        <EmptyState
   435→          icon={X}
   436→          title="Failed to load courses"
   437→          description="Please try again."
   438→          actionLabel="Retry"
   439→          onAction={() => qc.invalidateQueries({ queryKey: ["training-courses"] })}
   440→        />
   441→      ) : courses.length === 0 ? (
   442→        <EmptyState
   443→          icon={GraduationCap}
   444→          title={
   445→            search || status ? "No matching courses" : "No training courses yet"
   446→          }
   447→          description={
   448→            search || status
   449→              ? "Try adjusting your filters."
   450→              : "Create your first training course to start enrolling employees."
   451→          }
   452→          actionLabel="Create Course"
   453→          onAction={openCreate}
   454→        />
   455→      ) : (
   456→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   457→          {courses.map((c) => (
   458→            <CourseCard
   459→              key={c.id}
   460→              course={c}
   461→              onEdit={() => openEdit(c)}
   462→              onDelete={() => deleteCourse(c)}
   463→              onEnroll={() => setEnrollCourse(c)}
   464→              onStatusChange={(s) => setCourseStatus(c, s)}
   465→            />
   466→          ))}
   467→        </div>
   468→      )}
   469→
   470→      {/* Create / Edit dialog */}
   471→      <CourseFormDialog
   472→        open={formOpen}
   473→        onOpenChange={(o) => {
   474→          setFormOpen(o);
   475→          if (!o) setEditCourse(null);
   476→        }}
   477→        course={editCourse}
   478→        onSaved={() => qc.invalidateQueries({ queryKey: ["training-courses"] })}
   479→      />
   480→
   481→      {/* Enroll dialog */}
   482→      <EnrollDialog
   483→        course={enrollCourse}
   484→        onClose={() => setEnrollCourse(null)}
   485→        onSaved={() => {
   486→          qc.invalidateQueries({ queryKey: ["training-courses"] });
   487→          qc.invalidateQueries({ queryKey: ["training-enrollments"] });
   488→          setEnrollCourse(null);
   489→        }}
   490→      />
   491→    </div>
   492→  );
   493→}

   495→function CourseCard({
   496→  course,
   497→  onEdit,
   498→  onDelete,
   499→  onEnroll,
   500→  onStatusChange,
   501→}: {
   502→  course: Course;
   503→  onEdit: () => void;
   504→  onDelete: () => void;
   505→  onEnroll: () => void;
   506→  onStatusChange: (s: CourseStatus) => void;
   507→}) {
   508→  const pct =
   509→    course.capacity > 0
   510→      ? Math.min(100, Math.round((course.enrolledCount / course.capacity) * 100))
   511→      : 0;
   512→  const isFull = course.capacity > 0 && course.enrolledCount >= course.capacity;
   513→  return (
   514→    <Card className="p-4 flex flex-col gap-3 border-border/60 hover:border-border hover:shadow-card-hover transition-all">
   515→      <div className="flex items-start justify-between gap-2">
   516→        <div className="min-w-0 flex-1">
   517→          <h3
   518→            className="font-semibold leading-tight truncate"
   519→            title={course.title}
   520→          >
   521→            {course.title}
   522→          </h3>
   523→          <div className="flex flex-wrap items-center gap-1.5 mt-1.5">
   524→            <Badge variant="outline" className="text-[10px] font-medium bg-muted/50">
   525→              {course.category}
   526→            </Badge>
   527→            <Badge
   528→              variant="outline"
   529→              className={cn(
   530→                "font-medium border text-[10px]",
   531→                COURSE_STATUS_COLOR[course.status]
   532→              )}
   533→            >
   534→              {course.status.charAt(0) + course.status.slice(1).toLowerCase().replace(/_/g, " ")}
   535→            </Badge>
   536→          </div>
   537→        </div>
   538→        <DropdownMenu>
   539→          <DropdownMenuTrigger asChild>
   540→            <Button
   541→              size="icon"
   542→              variant="ghost"
   543→              className="size-8 -mr-1 -mt-1"
   544→              aria-label="Course actions"
   545→            >
   546→              <MoreVertical className="size-4" />
   547→            </Button>
   548→          </DropdownMenuTrigger>
   549→          <DropdownMenuContent align="end" className="w-48">
   550→            <DropdownMenuItem onClick={onEdit}>
   551→              <Pencil className="size-4 mr-2" /> Edit
   552→            </DropdownMenuItem>
   553→            <DropdownMenuItem onClick={onEnroll}>
   554→              <UserPlus className="size-4 mr-2" /> Enroll Employees
   555→            </DropdownMenuItem>
   556→            <DropdownMenuSeparator />
   557→            <DropdownMenuItem
   558→              onClick={() => onStatusChange("IN_PROGRESS")}
   559→              disabled={course.status === "IN_PROGRESS"}
   560→            >
   561→              <BookOpen className="size-4 mr-2" /> Mark In Progress
   562→            </DropdownMenuItem>
   563→            <DropdownMenuItem
   564→              onClick={() => onStatusChange("COMPLETED")}
   565→              disabled={course.status === "COMPLETED"}
   566→            >
   567→              <CheckCircle2 className="size-4 mr-2" /> Mark Completed
   568→            </DropdownMenuItem>
   569→            <DropdownMenuItem
   570→              onClick={() => onStatusChange("CANCELLED")}
   571→              disabled={course.status === "CANCELLED"}
   572→            >
   573→              <X className="size-4 mr-2" /> Cancel Course
   574→            </DropdownMenuItem>
   575→            <DropdownMenuSeparator />
   576→            <DropdownMenuItem
   577→              className="text-rose-600 focus:text-rose-700"
   578→              onClick={onDelete}
   579→            >
   580→              <Trash2 className="size-4 mr-2" /> Delete
   581→            </DropdownMenuItem>
   582→          </DropdownMenuContent>
   583→        </DropdownMenu>
   584→      </div>
   585→
   586→      {course.description && (
   587→        <p className="text-xs text-muted-foreground line-clamp-2">
   588→          {course.description}
   589→        </p>
   590→      )}
   591→
   592→      <div className="space-y-1 text-xs">
   593→        {course.trainer && (
   594→          <div className="flex items-center gap-1.5 text-muted-foreground">
   595→            <UserCheck className="size-3.5" />
   596→            <span className="truncate">{course.trainer}</span>
   597→          </div>
   598→        )}
   599→        <div className="flex items-center gap-1.5 text-muted-foreground">
   600→          <CalendarClock className="size-3.5" />
   601→          <span>
   602→            {course.startDate ? formatDate(course.startDate) : "TBD"}
   603→            {course.endDate ? ` → ${formatDate(course.endDate)}` : ""}
   604→          </span>

  1353→function EnrollDialog({
  1354→  course,
  1355→  onClose,
  1356→  onSaved,
  1357→}: {
  1358→  course: Course | null;
  1359→  onClose: () => void;
  1360→  onSaved: () => void;
  1361→}) {
  1362→  const { data: employeesData } = useEmployees(!!course);
  1363→  const employees: EmployeeOption[] = useMemo(
  1364→    () =>
  1365→      (employeesData?.items ?? []).map((e: any) => ({
  1366→        id: e.id,
  1367→        employeeId: e.employeeId,
  1368→        fullName: e.fullName,
  1369→        photo: e.photo,
  1370→        department: e.department,
  1371→        designation: e.designation,
  1372→      })) ?? [],
  1373→    [employeesData]
  1374→  );
  1375→
  1376→  const [selected, setSelected] = useState<Set<string>>(new Set());
  1377→  const [saving, setSaving] = useState(false);
  1378→
  1379→  const resetKey = course?.id ?? "none";
  1380→  const [lastKey, setLastKey] = useState(resetKey);
  1381→  if (resetKey !== lastKey) {
  1382→    setLastKey(resetKey);
  1383→    setSelected(new Set());
  1384→  }
  1385→
  1386→  function toggle(id: string) {
  1387→    setSelected((prev) => {
  1388→      const next = new Set(prev);
  1389→      if (next.has(id)) next.delete(id);
  1390→      else next.add(id);
  1391→      return next;
  1392→    });
  1393→  }
  1394→
  1395→  async function submit() {
  1396→    if (!course) return;
  1397→    if (selected.size === 0) {
  1398→      toast.error("Please select at least one employee.");
  1399→      return;
  1400→    }
  1401→    setSaving(true);
  1402→    try {
  1403→      const r = await fetch(`/api/training/${course.id}/enroll`, {
  1404→        method: "POST",
  1405→        headers: { "Content-Type": "application/json" },
  1406→        body: JSON.stringify({ employeeIds: Array.from(selected) }),
  1407→      });
  1408→      if (!r.ok) {
  1409→        const e = await r.json().catch(() => ({}));
  1410→        throw new Error(e?.error || "Failed to enroll employees");
  1411→      }
  1412→      const data = await r.json();
  1413→      toast.success(
  1414→        data.enrolled > 0
  1415→          ? `Enrolled ${data.enrolled} employee${data.enrolled === 1 ? "" : "s"} in "${course.title}".`
  1416→          : "No new enrollments (already enrolled)."
  1417→      );
  1418→      onSaved();
  1419→    } catch (err: any) {
  1420→      toast.error(err?.message || "Failed to enroll employees.");
  1421→    } finally {
  1422→      setSaving(false);
  1423→    }
  1424→  }
  1425→
  1426→  const isFull =
  1427→    course && course.capacity > 0 && course.enrolledCount >= course.capacity;
  1428→
  1429→  return (
  1430→    <Dialog
  1431→      open={!!course}
  1432→      onOpenChange={(o) => {
  1433→        if (!saving && !o) onClose();
  1434→      }}
  1435→    >
  1436→      <DialogContent className="sm:max-w-lg">
  1437→        <DialogHeader>
  1438→          <DialogTitle className="flex items-center gap-2">
  1439→            <UserPlus className="size-5 text-primary" />
  1440→            Enroll Employees
  1441→          </DialogTitle>
  1442→          <DialogDescription>
  1443→            {course
  1444→              ? `Select employees to enroll in "${course.title}".`
  1445→              : ""}
  1446→          </DialogDescription>
  1447→        </DialogHeader>
  1448→
  1449→        {isFull && (
  1450→          <div className="rounded-md bg-amber-500/10 border border-amber-500/30 text-amber-700 dark:text-amber-300 text-xs px-3 py-2">
  1451→            This course is currently at capacity. Existing enrollments may be
  1452→            reactivated, but no new active enrollments will be added.
  1453→          </div>
  1454→        )}
  1455→
  1456→        <div className="space-y-3">
  1457→          {employeesData ? (
  1458→            <MultiEmployeePicker
  1459→              employees={employees}
  1460→              selected={selected}
  1461→              onToggle={toggle}
  1462→            />
  1463→          ) : (
  1464→            <Skeleton className="h-10 w-full" />
  1465→          )}
  1466→
  1467→          {course && (
  1468→            <div className="rounded-md bg-muted/40 px-3 py-2 text-xs text-muted-foreground flex items-center justify-between">
  1469→              <span>
  1470→                Capacity: <span className="font-medium text-foreground">{course.capacity || "Unlimited"}</span>
  1471→              </span>
  1472→              <span>
  1473→                Enrolled: <span className="font-medium text-foreground">{course.enrolledCount}</span>
  1474→              </span>
  1475→            </div>
  1476→          )}
  1477→        </div>
  1478→
  1479→        <DialogFooter>
  1480→          <Button variant="outline" onClick={onClose} disabled={saving}>
  1481→            Cancel
  1482→          </Button>
  1483→          <Button onClick={submit} disabled={saving} className="gap-1.5">
  1484→            {saving && <Loader2 className="size-4 animate-spin" />}
  1485→            Enroll {selected.size > 0 ? `(${selected.size})` : ""}
  1486→          </Button>
  1487→        </DialogFooter>
  1488→      </DialogContent>
  1489→    </Dialog>
  1490→  );
  1491→}

  1497→function CompleteDialog({
  1498→  enrollment,
  1499→  onClose,
  1500→  onSaved,
  1501→}: {
  1502→  enrollment: Enrollment | null;
  1503→  onClose: () => void;
  1504→  onSaved: () => void;
  1505→}) {
  1506→  const [score, setScore] = useState<string>("");
  1507→  const [certificate, setCertificate] = useState("");
  1508→  const [saving, setSaving] = useState(false);
  1509→
  1510→  const resetKey = enrollment?.id ?? "none";
  1511→  const [lastKey, setLastKey] = useState(resetKey);
  1512→  if (resetKey !== lastKey) {
  1513→    setLastKey(resetKey);
  1514→    setScore(
  1515→      enrollment && enrollment.score !== null ? String(enrollment.score) : ""
  1516→    );
  1517→    setCertificate(enrollment?.certificate ?? "");
  1518→  }
  1519→
  1520→  async function submit() {
  1521→    if (!enrollment) return;
  1522→    setSaving(true);
  1523→    try {
  1524→      const body: any = { employeeId: enrollment.employeeId };
  1525→      if (score !== "") body.score = Number(score);
  1526→      if (certificate.trim()) body.certificate = certificate.trim();
  1527→      const r = await fetch(
  1528→        `/api/training/${enrollment.courseId}/complete`,
  1529→        {
  1530→          method: "POST",
  1531→          headers: { "Content-Type": "application/json" },
  1532→          body: JSON.stringify(body),
  1533→        }
  1534→      );
  1535→      if (!r.ok) {
  1536→        const e = await r.json().catch(() => ({}));
  1537→        throw new Error(e?.error || "Failed to mark complete");
  1538→      }
  1539→      toast.success(`Marked as completed.`);
  1540→      onSaved();
  1541→    } catch (err: any) {
  1542→      toast.error(err?.message || "Failed to mark complete.");
  1543→    } finally {
  1544→      setSaving(false);
  1545→    }
  1546→  }
  1547→
  1548→  return (
  1549→    <Dialog
  1550→      open={!!enrollment}
  1551→      onOpenChange={(o) => {
  1552→        if (!saving && !o) onClose();
  1553→      }}
  1554→    >
  1555→      <DialogContent className="sm:max-w-md">
  1556→        <DialogHeader>
  1557→          <DialogTitle className="flex items-center gap-2">
  1558→            <CheckCircle2 className="size-5 text-primary" />
  1559→            Mark Enrollment Complete
  1560→          </DialogTitle>
  1561→          <DialogDescription>
  1562→            {enrollment
  1563→              ? `Record completion for ${enrollment.employeeName ?? "employee"} in "${enrollment.courseTitle}".`
  1564→              : ""}
  1565→          </DialogDescription>
  1566→        </DialogHeader>
  1567→        <div className="space-y-4">
  1568→          <div className="space-y-1.5">
  1569→            <Label htmlFor="complete-score">Score (%) — optional</Label>
  1570→            <Input
  1571→              id="complete-score"
  1572→              type="number"
  1573→              min={0}
  1574→              max={100}
  1575→              placeholder="e.g. 85"
  1576→              value={score}
  1577→              onChange={(e) => setScore(e.target.value)}
  1578→            />
  1579→          </div>
  1580→          <div className="space-y-1.5">
  1581→            <Label htmlFor="complete-cert">Certificate / credential ID — optional</Label>
  1582→            <Input
  1583→              id="complete-cert"
  1584→              placeholder="e.g. CERT-2024-001"
  1585→              value={certificate}
  1586→              onChange={(e) => setCertificate(e.target.value)}
  1587→            />
  1588→          </div>
  1589→        </div>
  1590→        <DialogFooter>
  1591→          <Button variant="outline" onClick={onClose} disabled={saving}>

    51→    51→
    52→    52→type OnboardingStatus =
    53→    53→  | "PENDING"
    54→    54→  | "IN_PROGRESS"
    55→    55→  | "COMPLETED"
    56→    56→  | "SKIPPED";
    57→    57→
    58→    58→interface OnboardingTask {
    59→    59→  id: string;
    60→    60→  employeeId: string;
    61→    61→  title: string;
    62→    62→  description?: string | null;
    63→    63→  dueDate?: string | null;
    64→    64→  assignedTo?: string | null;
    65→    65→  status: OnboardingStatus;
    66→    66→  notes?: string | null;
    67→    67→  completedAt?: string | null;
    68→    68→  sortOrder: number;
    69→    69→  isDefault: boolean;
    70→    70→  createdAt: string;
    71→    71→  updatedAt: string;
    72→    72→}
    73→    73→
    74→    74→const STATUS_CYCLE: Record<OnboardingStatus, OnboardingStatus> = {
    75→    75→  PENDING: "IN_PROGRESS",
    76→    76→  IN_PROGRESS: "COMPLETED",
    77→    77→  COMPLETED: "PENDING",
    78→    78→  SKIPPED: "PENDING",
    79→    79→};
    80→    80→
    81→    81→const STATUS_ICON: Record<OnboardingStatus, typeof CheckCircle2> = {
    82→    82→  PENDING: CircleDashed,
    83→    83→  IN_PROGRESS: CircleDot,
    84→    84→  COMPLETED: CheckCircle2,
    85→    85→  SKIPPED: SkipForward,
    86→    86→};
    87→    87→
    88→    88→const STATUS_COLOR: Record<OnboardingStatus, string> = {
    89→    89→  PENDING: "text-muted-foreground",
    90→    90→  IN_PROGRESS: "text-amber-600",
    91→    91→  COMPLETED: "text-emerald-600",
    92→    92→  SKIPPED: "text-rose-600",
    93→    93→};
    94→    94→
    95→    95→function isOverdue(dueDate?: string | null, status?: OnboardingStatus) {
    96→    96→  if (!dueDate) return false;
    97→    97→  if (status === "COMPLETED" || status === "SKIPPED") return false;
    98→    98→  return new Date(dueDate).getTime() < Date.now();
    99→    99→}
   100→   100→
   101→   101→function ProgressRing({ percent }: { percent: number }) {
   102→   102→  const size = 96;
   103→   103→  const stroke = 8;
   104→   104→  const r = (size - stroke) / 2;
   105→   105→  const c = 2 * Math.PI * r;
   106→   106→  const offset = c - (Math.min(100, Math.max(0, percent)) / 100) * c;
   107→   107→  return (
   108→   108→    <div className="relative" style={{ width: size, height: size }}>
   109→   109→      <svg width={size} height={size} className="-rotate-90">
   110→   110→        <circle
   111→   111→          cx={size / 2}
   112→   112→          cy={size / 2}
   113→   113→          r={r}
   114→   114→          fill="none"
   115→   115→          strokeWidth={stroke}
   116→   116→          className="stroke-muted"
   117→   117→        />
   118→   118→        <circle
   119→   119→          cx={size / 2}
   120→   120→          cy={size / 2}
   121→   121→          r={r}
   122→   122→          fill="none"
   123→   123→          strokeWidth={stroke}
   124→   124→          strokeLinecap="round"
   125→   125→          className="stroke-primary transition-all duration-500"
   126→   126→          strokeDasharray={c}
   127→   127→          strokeDashoffset={offset}
   128→   128→        />
   129→   129→      </svg>
   130→   130→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   131→   131→        <span className="text-2xl font-bold tabular-nums">
   132→   132→          {Math.round(percent)}%
   133→   133→        </span>
   134→   134→        <span className="text-[10px] text-muted-foreground uppercase tracking-wide">
   135→   135→          Complete
   136→   136→        </span>
   137→   137→      </div>
   138→   138→    </div>
   139→   139→  );
   140→   140→}
   141→   141→
   142→   142→export function Onboarding({ employeeId }: { employeeId: string }) {
   143→   143→  const qc = useQueryClient();
   144→   144→  const [filter, setFilter] = useState<"all" | "pending" | "completed">("all");
   145→   145→  const [addOpen, setAddOpen] = useState(false);
   146→   146→  const [updatingId, setUpdatingId] = useState<string | null>(null);
   147→   147→  const [editingNotesId, setEditingNotesId] = useState<string | null>(null);
   148→   148→  const [notesDraft, setNotesDraft] = useState("");
   149→   149→
   150→   150→  const queryKey = ["onboarding", employeeId];
   151→   151→
   152→   152→  const { data, isLoading, isError } = useQuery({
   153→   153→    queryKey,
   154→   154→    queryFn: async () => {
   155→   155→      const r = await fetch(`/api/onboarding?employeeId=${employeeId}`);
   156→   156→      if (!r.ok) throw new Error("Failed to load onboarding tasks");
   157→   157→      return r.json();
   158→   158→    },
   159→   159→  });
   160→   160→
   161→   161→  const tasks: OnboardingTask[] = data?.items ?? [];
   162→   162→
   163→   163→  const stats = useMemo(() => {
   164→   164→    const total = tasks.length;
   165→   165→    const completed = tasks.filter((t) => t.status === "COMPLETED").length;
   166→   166→    const inProgress = tasks.filter((t) => t.status === "IN_PROGRESS").length;
   167→   167→    const skipped = tasks.filter((t) => t.status === "SKIPPED").length;
   168→   168→    const pending = tasks.filter((t) => t.status === "PENDING").length;
   169→   169→    const overdue = tasks.filter((t) => isOverdue(t.dueDate, t.status)).length;
   170→   170→    const activeTotal = total - skipped;
   171→   171→    const percent = activeTotal > 0 ? (completed / activeTotal) * 100 : 0;
   172→   172→    return { total, completed, inProgress, skipped, pending, overdue, percent };
   173→   173→  }, [tasks]);
   174→   174→
   175→   175→  const filteredTasks = useMemo(() => {
   176→   176→    if (filter === "pending") {
   177→   177→      return tasks.filter(
   178→   178→        (t) => t.status === "PENDING" || t.status === "IN_PROGRESS"
   179→   179→      );
   180→   180→    }
   181→   181→    if (filter === "completed") {
   182→   182→      return tasks.filter(
   183→   183→        (t) => t.status === "COMPLETED" || t.status === "SKIPPED"
   184→   184→      );
   185→   185→    }
   186→   186→    return tasks;
   187→   187→  }, [tasks, filter]);
   188→   188→
   189→   189→  async function patchTask(id: string, patch: Record<string, unknown>) {
   190→   190→    setUpdatingId(id);
   191→   191→    try {
   192→   192→      const r = await fetch(`/api/onboarding/${id}`, {
   193→   193→        method: "PATCH",
   194→   194→        headers: { "Content-Type": "application/json" },
   195→   195→        body: JSON.stringify(patch),
   196→   196→      });
   197→   197→      if (!r.ok) {
   198→   198→        const err = await r.json().catch(() => ({}));
   199→   199→        throw new Error(err.error || "Update failed");
   200→   200→      }
   201→   201→      await qc.invalidateQueries({ queryKey });
   202→   202→    } catch (e: any) {
   203→   203→      toast.error(e?.message || "Failed to update task");
   204→   204→    } finally {
   205→   205→      setUpdatingId(null);
   206→   206→    }
   207→   207→  }
   208→   208→
   209→   209→  async function cycleStatus(task: OnboardingTask) {
   210→   210→    const next = STATUS_CYCLE[task.status];
   211→   211→    await patchTask(task.id, { status: next });
   212→   212→    toast.success(`Marked as ${next.replace(/_/g, " ").toLowerCase()}`);
   213→   213→  }
   214→   214→
   215→   215→  async function skipTask(task: OnboardingTask) {
   216→   216→    await patchTask(task.id, { status: "SKIPPED" });
   217→   217→    toast.success(`Task skipped`);
   218→   218→  }
   219→   219→
   220→   220→  async function deleteTask(task: OnboardingTask) {
   221→   221→    setUpdatingId(task.id);
   222→   222→    try {
   223→   223→      const r = await fetch(`/api/onboarding/${task.id}`, {
   224→   224→        method: "DELETE",
   225→   225→      });
   226→   226→      if (!r.ok) throw new Error("Delete failed");
   227→   227→      await qc.invalidateQueries({ queryKey });
   228→   228→      toast.success("Task removed");
   229→   229→    } catch (e: any) {
   230→   230→      toast.error(e?.message || "Failed to delete task");
   231→   231→    } finally {
   232→   232→      setUpdatingId(null);
   233→   233→    }
   234→   234→  }
   235→   235→
   236→   236→  function startEditNotes(task: OnboardingTask) {
   237→   237→    setEditingNotesId(task.id);
   238→   238→    setNotesDraft(task.notes ?? "");
   239→   239→  }
   240→   240→
   241→   241→  async function saveNotes(task: OnboardingTask) {
   242→   242→    await patchTask(task.id, { notes: notesDraft });
   243→   243→    setEditingNotesId(null);
   244→   244→    toast.success("Notes saved");
   245→   245→  }
   246→   246→
   247→   247→  if (isLoading) {
   248→   248→    return (
   249→   249→      <div className="space-y-4">
   250→   250→        <div className="h-32 rounded-xl bg-muted/40 animate-pulse" />
   251→   251→        <div className="h-20 rounded-xl bg-muted/40 animate-pulse" />
   252→   252→        <div className="h-20 rounded-xl bg-muted/40 animate-pulse" />
   253→   253→        <div className="h-20 rounded-xl bg-muted/40 animate-pulse" />
   254→   254→      </div>
   255→   255→    );
   256→   256→  }
   257→   257→
   258→   258→  if (isError) {
   259→   259→    return (
   260→   260→      <EmptyState
   261→   261→        icon={CircleDashed}
   262→   262→        title="Couldn't load onboarding tasks"
   263→   263→        description="Please try again in a moment."
   264→   264→      />
   265→   265→    );
   266→   266→  }
   267→   267→
   268→   268→  return (
   269→   269→    <div className="space-y-4">
   270→   270→      {/* Summary card */}
   271→   271→      <Card className="border-border/60 shadow-soft">
   272→   272→        <CardContent className="p-4 sm:p-6">
   273→   273→          <div className="flex flex-col sm:flex-row items-center gap-6">
   274→   274→            <ProgressRing percent={stats.percent} />
   275→   275→            <div className="flex-1 w-full space-y-3">
   276→   276→              <div className="flex items-center justify-between flex-wrap gap-2">
   277→   277→                <div>
   278→   278→                  <div className="text-lg font-semibold">
   279→   279→                    {stats.completed} of {stats.total} tasks completed
   280→   280→                  </div>
   281→   281→                  <div className="text-xs text-muted-foreground mt-0.5">
   282→   282→                    Onboarding progress for this employee
   283→   283→                  </div>
   284→   284→                </div>
   285→   285→                <Button size="sm" onClick={() => setAddOpen(true)}>
   286→   286→                  <Plus className="size-4 mr-1.5" /> Add Task
   287→   287→                </Button>
   288→   288→              </div>
   289→   289→              <Progress value={stats.percent} className="h-2.5" />
   290→   290→              <div className="grid grid-cols-2 sm:grid-cols-4 gap-2 text-xs">
   291→   291→                <StatPill
   292→   292→                  label="Pending"
   293→   293→                  value={stats.pending}
   294→   294→                  tone="amber"
   295→   295→                />
   296→   296→                <StatPill
   297→   297→                  label="In Progress"
   298→   298→                  value={stats.inProgress}
   299→   299→                  tone="sky"
   300→   300→                />
   301→   301→                <StatPill
   302→   302→                  label="Completed"
   303→   303→                  value={stats.completed}
   304→   304→                  tone="emerald"
   305→   305→                />
   306→   306→                <StatPill label="Skipped" value={stats.skipped} tone="rose" />
   307→   307→              </div>
   308→   308→              {stats.overdue > 0 && (
   309→   309→                <div className="text-xs font-medium text-rose-600 flex items-center gap-1.5">
   310→   310→                  <CalendarDays className="size-3.5" />
   311→   311→                  {stats.overdue} task{stats.overdue > 1 ? "s" : ""} overdue
   312→   312→                </div>
   313→   313→              )}
   314→   314→            </div>
   315→   315→          </div>
   316→   316→        </CardContent>
   317→   317→      </Card>
   318→   318→
   319→   319→      {/* Filters */}
   320→   320→      <div className="flex items-center justify-between gap-3 flex-wrap">
   321→   321→        <div className="inline-flex rounded-lg border border-border overflow-hidden">
   322→   322→          {(["all", "pending", "completed"] as const).map((f) => (
   323→   323→            <button
   324→   324→              key={f}
   325→   325→              onClick={() => setFilter(f)}
   326→   326→              className={cn(
   327→   327→                "px-3 py-1.5 text-xs font-medium capitalize transition-colors",
   328→   328→                filter === f
   329→   329→                  ? "bg-primary text-primary-foreground"
   330→   330→                  : "hover:bg-muted text-muted-foreground"
   331→   331→              )}
   332→   332→            >
   333→   333→              {f === "all"
   334→   334→                ? `All (${stats.total})`
   335→   335→                : f === "pending"
   336→   336→                  ? `Pending (${stats.pending + stats.inProgress})`
   337→   337→                  : `Completed (${stats.completed + stats.skipped})`}
   338→   338→            </button>
   339→   339→          ))}
   340→   340→        </div>
   341→   341→      </div>
   342→   342→
   343→   343→      {/* Checklist */}
   344→   344→      {filteredTasks.length === 0 ? (
   345→   345→        <EmptyState
   346→   346→          icon={CheckCircle}
   347→   347→          title="No onboarding tasks here"
   348→   348→          description={
   349→   349→            filter === "all"
   350→   350→              ? "Add a custom task to start tracking onboarding progress."
   351→   351→              : "No tasks match this filter."
   352→   352→          }
   353→   353→          actionLabel="Add Task"
   354→   354→          onAction={() => setAddOpen(true)}
   355→   355→        />
   356→   356→      ) : (
   357→   357→        <div className="space-y-2.5">
   358→   358→          {filteredTasks.map((task) => {
   359→   359→            const StatusIcon = STATUS_ICON[task.status];
   360→   360→            const overdue = isOverdue(task.dueDate, task.status);
   361→   361→            const isUpdating = updatingId === task.id;
   362→   362→            const isEditingNotes = editingNotesId === task.id;
   363→   363→            return (
   364→   364→              <Card
   365→   365→                key={task.id}
   366→   366→                className={cn(
   367→   367→                  "border-border/60 shadow-soft transition-all",
   368→   368→                  task.status === "COMPLETED" && "opacity-75",
   369→   369→                  task.status === "SKIPPED" && "opacity-60"
   370→   370→                )}
   371→   371→              >
   372→   372→                <CardContent className="p-4">
   373→   373→                  <div className="flex items-start gap-3">
   374→   374→                    {/* Checkbox / status cycler */}
   375→   375→                    <button
   376→   376→                      onClick={() => cycleStatus(task)}
   377→   377→                      disabled={isUpdating}
   378→   378→                      className="flex-shrink-0 mt-0.5 disabled:opacity-50"
   379→   379→                      aria-label={`Cycle status (currently ${task.status})`}
   380→   380→                      title={`Status: ${task.status.replace(/_/g, " ")} — click to advance`}
   381→   381→                    >
   382→   382→                      <Checkbox
   383→   383→                        checked={task.status === "COMPLETED"}
   384→   384→                        className={cn(
   385→   385→                          "size-5 data-[state=checked]:bg-emerald-500 data-[state=checked]:border-emerald-500",
   386→   386→                          task.status === "IN_PROGRESS" &&
   387→   387→                            "border-amber-500 data-[state=checked]:bg-amber-500 data-[state=checked]:border-amber-500"
   388→   388→                        )}
   389→   389→                      />
   390→   390→                    </button>
   391→   391→
   392→   392→                    <div className="flex-1 min-w-0 space-y-1.5">
   393→   393→                      <div className="flex items-start justify-between gap-2 flex-wrap">
   394→   394→                        <div className="flex items-start gap-2 min-w-0">
   395→   395→                          <StatusIcon
   396→   396→                            className={cn(
   397→   397→                              "size-4 mt-0.5 flex-shrink-0",
   398→   398→                              STATUS_COLOR[task.status]
   399→   399→                            )}
   400→   400→                          />
   401→   401→                          <div className="min-w-0">
   402→   402→                            <div
   403→   403→                              className={cn(
   404→   404→                                "text-sm font-medium leading-tight",
   405→   405→                                task.status === "COMPLETED" &&
   406→   406→                                  "line-through text-muted-foreground",
   407→   407→                                task.status === "SKIPPED" &&
   408→   408→                                  "line-through text-muted-foreground"
   409→   409→                              )}
   410→   410→                            >
   411→   411→                              {task.title}
   412→   412→                              {!task.isDefault && (
   413→   413→                                <Badge
   414→   414→                                  variant="outline"
   415→   415→                                  className="ml-2 text-[10px] py-0 px-1.5 font-normal"
   416→   416→                                >
   417→   417→                                  Custom
   418→   418→                                </Badge>
   419→   419→                              )}
   420→   420→                            </div>
   421→   421→                            {task.description && (
   422→   422→                              <div className="text-xs text-muted-foreground mt-0.5">
   423→   423→                                {task.description}
   424→   424→                              </div>
   425→   425→                            )}
   426→   426→                          </div>
   427→   427→                        </div>
   428→   428→                        <StatusBadge
   429→   429→                          status={task.status}
   430→   430→                          label={task.status.replace(/_/g, " ")}
   431→   431→                        />
   432→   432→                      </div>
   433→   433→
   434→   434→                      {/* Meta row */}
   435→   435→                      <div className="flex flex-wrap items-center gap-x-4 gap-y-1 text-[11px] text-muted-foreground">
   436→   436→                        {task.assignedTo && (
   437→   437→                          <span className="inline-flex items-center gap-1">
   438→   438→                            <UserCog className="size-3" />
   439→   439→                            {task.assignedTo}
   440→   440→                          </span>
   441→   441→                        )}
   442→   442→                        {task.dueDate && (
   443→   443→                          <span
   444→   444→                            className={cn(
   445→   445→                              "inline-flex items-center gap-1",
   446→   446→                              overdue && "text-rose-600 font-medium"
   447→   447→                            )}
   448→   448→                          >
   449→   449→                            <CalendarDays className="size-3" />
   450→   450→                            Due {formatDate(task.dueDate)}
   451→   451→                            {overdue && " · overdue"}
   452→   452→                          </span>
   453→   453→                        )}
   454→   454→                        {task.completedAt && (
   455→   455→                          <span className="inline-flex items-center gap-1 text-emerald-600">
   456→   456→                            <CheckCircle className="size-3" />
   457→   457→                            Done {formatDate(task.completedAt)}
   458→   458→                          </span>
   459→   459→                        )}
   460→   460→                      </div>
   461→   461→
   462→   462→                      {/* Notes */}
   463→   463→                      {isEditingNotes ? (
   464→   464→                        <div className="space-y-2 mt-2">
   465→   465→                          <Textarea
   466→   466→                            value={notesDraft}
   467→   467→                            onChange={(e) => setNotesDraft(e.target.value)}
   468→   468→                            placeholder="Add a note about this task…"
   469→   469→                            rows={2}
   470→   470→                            className="text-xs"
   471→   471→                          />
   472→   472→                          <div className="flex gap-2">
   473→   473→                            <Button
   474→   474→                              size="sm"
   475→   475→                              onClick={() => saveNotes(task)}
   476→   476→                              disabled={isUpdating}
   477→   477→                            >
   478→   478→                              {isUpdating ? (
   479→   479→                                <Loader2 className="size-3.5 animate-spin mr-1" />
   480→   480→                              ) : (
   481→   481→                                <CheckCircle className="size-3.5 mr-1" />
   482→   482→                              )}
   483→   483→                              Save
   484→   484→                            </Button>
   485→   485→                            <Button
   486→   486→                              size="sm"
   487→   487→                              variant="ghost"
   488→   488→                              onClick={() => setEditingNotesId(null)}

   606→   606→function AddTaskDialog({
   607→   607→  open,
   608→   608→  onOpenChange,
   609→   609→  employeeId,
   610→   610→  onSaved,
   611→   611→}: {
   612→   612→  open: boolean;
   613→   613→  onOpenChange: (v: boolean) => void;
   614→   614→  employeeId: string;
   615→   615→  onSaved: () => void;
   616→   616→}) {
   617→   617→  const [title, setTitle] = useState("");
   618→   618→  const [description, setDescription] = useState("");
   619→   619→  const [dueDate, setDueDate] = useState("");
   620→   620→  const [assignedTo, setAssignedTo] = useState("");
   621→   621→  const [saving, setSaving] = useState(false);
   622→   622→
   623→   623→  function reset() {
   624→   624→    setTitle("");
   625→   625→    setDescription("");
   626→   626→    setDueDate("");
   627→   627→    setAssignedTo("");
   628→   628→  }
   629→   629→
   630→   630→  async function handleSave() {
   631→   631→    if (!title.trim()) {
   632→   632→      toast.error("Task title is required");
   633→   633→      return;
   634→   634→    }
   635→   635→    setSaving(true);
   636→   636→    try {
   637→   637→      const body: Record<string, unknown> = {
   638→   638→        employeeId,
   639→   639→        title: title.trim(),
   640→   640→      };

   646→   646→        method: "POST",

   649→   649→      });

   652→   652→        throw new Error(err.error || "Failed to add task");
   653→   653→      }

   656→   656→      onOpenChange(false);

   665→   665→  return (

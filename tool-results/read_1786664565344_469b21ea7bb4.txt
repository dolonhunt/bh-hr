   460→type HeatmapItem = {
   461→  date: string;
   462→  status: string;
   463→  workingHours: number | null;
   464→  count?: number;
   465→};
   466→
   467→function AttendanceHeatmap() {
   468→  const [employeeId, setEmployeeId] = useState("");
   469→  const [months, setMonths] = useState(3);
   470→
   471→  const { data: employeesData } = useQuery({
   472→    queryKey: ["employees-for-heatmap"],
   473→    queryFn: () =>
   474→      fetch("/api/employees?pageSize=1000").then((r) => r.json()),
   475→  });
   476→
   477→  const { data, isLoading } = useQuery({
   478→    queryKey: ["attendance-heatmap", employeeId, months],
   479→    queryFn: async () => {
   480→      const params = new URLSearchParams();
   481→      if (employeeId) params.set("employeeId", employeeId);
   482→      params.set("months", String(months));
   483→      const r = await fetch(`/api/attendance/heatmap?${params.toString()}`);
   484→      return r.json();
   485→    },
   486→  });
   487→
   488→  const items: HeatmapItem[] = data?.items ?? [];
   489→  const startDateStr = data?.startDate;
   490→  const endDateStr = data?.endDate;
   491→
   492→  const { weeks, monthLabels } = useMemo(() => {
   493→    if (!startDateStr || !endDateStr) {
   494→      return { weeks: [] as Date[][], monthLabels: [] as { col: number; label: string }[] };
   495→    }
   496→    const start = new Date(startDateStr);
   497→    start.setDate(start.getDate() - start.getDay()); // shift to Sunday
   498→    const end = new Date(endDateStr);
   499→    end.setDate(end.getDate() + (6 - end.getDay())); // shift to Saturday
   500→
   501→    const w: Date[][] = [];
   502→    const cur = new Date(start);
   503→    while (cur.getTime() <= end.getTime()) {
   504→      const wk: Date[] = [];
   505→      for (let i = 0; i < 7; i++) {
   506→        wk.push(new Date(cur));
   507→        cur.setDate(cur.getDate() + 1);
   508→      }
   509→      w.push(wk);
   510→    }
   511→
   512→    const labels: { col: number; label: string }[] = [];
   513→    w.forEach((wk, i) => {
   514→      const first = wk[0];
   515→      const isFirst = i === 0 || first.getMonth() !== w[i - 1][0].getMonth();
   516→      if (isFirst) {
   517→        labels.push({
   518→          col: i,
   519→          label: first.toLocaleDateString("en-US", { month: "short" }),
   520→        });
   521→      }
   522→    });
   523→    return { weeks: w, monthLabels: labels };
   524→  }, [startDateStr, endDateStr]);
   525→
   526→  // Map items by date for O(1) lookup
   527→  const byDate = useMemo(() => {
   528→    const m = new Map<string, HeatmapItem>();
   529→    items.forEach((i) => m.set(i.date, i));
   530→    return m;
   531→  }, [items]);
   532→
   533→  const todayKey = new Date().toISOString().slice(0, 10);
   534→
   535→  return (
   536→    <Card className="border-border/60 shadow-soft p-4 sm:p-5">
   537→      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3 mb-4">
   538→        <div>
   539→          <h3 className="text-sm font-semibold flex items-center gap-1.5">
   540→            <CalendarDays className="size-4 text-primary" />
   541→            Attendance Heatmap
   542→          </h3>
   543→          <p className="text-xs text-muted-foreground mt-0.5">
   544→            Last {months} month{months === 1 ? "" : "s"} — color intensity reflects attendance status
   545→          </p>
   546→        </div>
   547→        <div className="flex items-center gap-2 flex-wrap">
   548→          <Select
   549→            value={employeeId || "ALL"}
   550→            onValueChange={(v) => setEmployeeId(v === "ALL" ? "" : v)}
   551→          >
   552→            <SelectTrigger className="w-[180px] h-8 text-xs">
   553→              <SelectValue placeholder="All employees" />
   554→            </SelectTrigger>
   555→            <SelectContent>
   556→              <SelectItem value="ALL">All employees (aggregated)</SelectItem>
   557→              {(employeesData?.items ?? []).map((e: any) => (
   558→                <SelectItem key={e.id} value={e.id}>
   559→                  {e.fullName} · {e.employeeId}
   560→                </SelectItem>
   561→              ))}
   562→            </SelectContent>
   563→          </Select>
   564→          <Select
   565→            value={String(months)}
   566→            onValueChange={(v) => setMonths(Number(v))}
   567→          >
   568→            <SelectTrigger className="w-[110px] h-8 text-xs">
   569→              <SelectValue />
   570→            </SelectTrigger>
   571→            <SelectContent>
   572→              <SelectItem value="1">1 month</SelectItem>
   573→              <SelectItem value="3">3 months</SelectItem>
   574→              <SelectItem value="6">6 months</SelectItem>
   575→              <SelectItem value="12">12 months</SelectItem>
   576→            </SelectContent>
   577→          </Select>
   578→        </div>
   579→      </div>
   580→
   581→      {isLoading ? (
   582→        <div className="space-y-2">
   583→          <Skeleton className="h-4 w-full" />
   584→          <Skeleton className="h-32 w-full" />
   585→        </div>
   586→      ) : weeks.length === 0 ? (
   587→        <div className="text-sm text-muted-foreground py-8 text-center">
   588→          No attendance data for this range.
   589→        </div>
   590→      ) : (
   591→        <div className="overflow-x-auto pb-1 -mx-1 px-1">
   592→          <div className="inline-block min-w-full">
   593→            {/* Month labels */}
   594→            <div
   595→              className="relative mb-1 ml-7"
   596→              style={{ height: "14px" }}
   597→            >
   598→              {monthLabels.map((m, i) => (
   599→                <div
   600→                  key={i}
   601→                  className="absolute top-0 text-[10px] font-medium text-muted-foreground whitespace-nowrap"
   602→                  style={{ left: `${m.col * (CELL_PX + GAP_PX)}px` }}
   603→                >
   604→                  {m.label}
   605→                </div>
   606→              ))}
   607→            </div>
   608→
   609→            {/* Heatmap body: day labels + grid */}
   610→            <div className="flex gap-1">
   611→              {/* Day labels */}
   612→              <div
   613→                className="flex flex-col text-[10px] text-muted-foreground pr-1 flex-shrink-0"
   614→                style={{ width: "1.5rem" }}
   615→              >
   616→                {["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"].map(
   617→                  (d, i) => (
   618→                    <div
   619→                      key={d}
   620→                      className="leading-none flex items-end"
   621→                      style={{
   622→                        height: `${CELL_PX}px`,
   623→                        marginBottom: i < 6 ? `${GAP_PX}px` : 0,
   624→                      }}
   625→                    >
   626→                      {d === "Mon" || d === "Wed" || d === "Fri" ? d : ""}
   627→                    </div>
   628→                  )
   629→                )}
   630→              </div>
   631→
   632→              {/* Grid */}
   633→              <TooltipProvider delayDuration={150}>
   634→                <div
   635→                  className="grid grid-flow-col"
   636→                  style={{
   637→                    gridTemplateRows: `repeat(7, ${CELL_PX}px)`,
   638→                    gap: `${GAP_PX}px`,
   639→                  }}
   640→                >
   641→                  {weeks.flat().map((day, idx) => {
   642→                    const key = localDateKey(day);
   643→                    const info = byDate.get(key);
   644→                    const isFuture = key > todayKey;
   645→                    const status = info?.status;
   646→                    const colorClass = !status
   647→                      ? isFuture
   648→                        ? "bg-transparent"
   649→                        : "bg-muted/30"
   650→                      : INTENSITY_COLORS[status] ?? "bg-muted/30";
   651→                    const tooltipText = buildTooltipText(day, info);
   652→
   653→                    return (
   654→                      <Tooltip key={idx}>
   655→                        <TooltipTrigger asChild>
   656→                          <div
   657→                            className={cn(
   658→                              "rounded-[3px] ring-0 transition-transform hover:ring-1 hover:ring-primary/50 hover:scale-110",
   659→                              colorClass
   660→                            )}
   661→                            style={{ width: `${CELL_PX}px`, height: `${CELL_PX}px` }}
   662→                          />
   663→                        </TooltipTrigger>
   664→                        <TooltipContent
   665→                          side="top"
   666→                          className="text-xs whitespace-nowrap"
   667→                        >
   668→                          {tooltipText}
   669→                        </TooltipContent>
   670→                      </Tooltip>
   671→                    );
   672→                  })}
   673→                </div>
   674→              </TooltipProvider>
   675→            </div>
   676→
   677→            {/* Legend */}
   678→            <div className="mt-4 flex items-center gap-1.5 flex-wrap text-[10px] text-muted-foreground ml-7">
   679→              <span className="font-medium">Less</span>
   680→              {LEGEND_ITEMS.slice(1, -1).map((it, i) => (
   681→                <span
   682→                  key={i}
   683→                  className={cn(
   684→                    "size-3 rounded-[3px] inline-block",
   685→                    it.color
   686→                  )}
   687→                  title={it.label}
   688→                />
   689→              ))}
   690→              <span className="font-medium">More</span>
   691→            </div>
   692→          </div>
   693→        </div>
   694→      )}
   695→    </Card>
   696→  );
   697→}
   698→
   699→function localDateKey(d: Date): string {
   700→  const y = d.getFullYear();
   701→  const m = String(d.getMonth() + 1).padStart(2, "0");
   702→  const day = String(d.getDate()).padStart(2, "0");
   703→  return `${y}-${m}-${day}`;
   704→}
   705→
   706→function buildTooltipText(day: Date, info: HeatmapItem | undefined): string {
   707→  const dateStr = day.toLocaleDateString("en-US", {
   708→    weekday: "short",
   709→    day: "numeric",
   710→    month: "short",
   711→    year: "numeric",
   712→  });
   713→  if (!info) {
   714→    return `${dateStr} — No data`;
   715→  }
   716→  const statusLabel = STATUS_LABEL[info.status] ?? info.status;
   717→  const countSuffix =
   718→    info.count !== undefined && info.count > 0
   719→      ? ` · ${info.count} record${info.count === 1 ? "" : "s"}`
   720→      : "";
   721→  const hoursSuffix =
   722→    info.workingHours !== null && info.workingHours !== undefined
   723→      ? ` · ${info.workingHours}h avg`
   724→      : "";
   725→  return `${dateStr} — ${statusLabel}${hoursSuffix}${countSuffix}`;
   726→}

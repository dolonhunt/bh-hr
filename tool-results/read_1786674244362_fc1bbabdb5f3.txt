  1631→function CompleteDialog({
  1632→  enrollment,
  1633→  onClose,
  1634→  onSaved,
  1635→}: {
  1636→  enrollment: Enrollment | null;
  1637→  onClose: () => void;
  1638→  onSaved: () => void;
  1639→}) {
  1640→  const [score, setScore] = useState<string>("");
  1641→  const [certificate, setCertificate] = useState("");
  1642→  const [saving, setSaving] = useState(false);
  1643→
  1644→  const resetKey = enrollment?.id ?? "none";
  1645→  const [lastKey, setLastKey] = useState(resetKey);
  1646→  if (resetKey !== lastKey) {
  1647→    setLastKey(resetKey);
  1648→    setScore(
  1649→      enrollment && enrollment.score !== null ? String(enrollment.score) : ""
  1650→    );
  1651→    setCertificate(enrollment?.certificate ?? "");
  1652→  }
  1653→
  1654→  async function submit() {
  1655→    if (!enrollment) return;
  1656→    setSaving(true);
  1657→    try {
  1658→      const body: any = { employeeId: enrollment.employeeId };
  1659→      if (score !== "") body.score = Number(score);
  1660→      if (certificate.trim()) body.certificate = certificate.trim();
  1661→      const r = await fetch(
  1662→        `/api/training/${enrollment.courseId}/complete`,
  1663→        {
  1664→          method: "POST",
  1665→          headers: { "Content-Type": "application/json" },
  1666→          body: JSON.stringify(body),
  1667→        }
  1668→      );
  1669→      if (!r.ok) {
  1670→        const e = await r.json().catch(() => ({}));
  1671→        throw new Error(e?.error || "Failed to mark complete");
  1672→      }
  1673→      toast.success(`Marked as completed.`);
  1674→      onSaved();
  1675→    } catch (err: any) {
  1676→      toast.error(err?.message || "Failed to mark complete.");
  1677→    } finally {
  1678→      setSaving(false);
  1679→    }
  1680→  }
  1681→
  1682→  return (
  1683→    <Dialog
  1684→      open={!!enrollment}
  1685→      onOpenChange={(o) => {
  1686→        if (!saving && !o) onClose();
  1687→      }}
  1688→    >
  1689→      <DialogContent className="sm:max-w-md">
  1690→        <DialogHeader>
  1691→          <DialogTitle className="flex items-center gap-2">
  1692→            <CheckCircle2 className="size-5 text-primary" />
  1693→            Mark Enrollment Complete
  1694→          </DialogTitle>
  1695→          <DialogDescription>
  1696→            {enrollment
  1697→              ? `Record completion for ${enrollment.employeeName ?? "employee"} in "${enrollment.courseTitle}".`
  1698→              : ""}
  1699→          </DialogDescription>
  1700→        </DialogHeader>
  1701→        <div className="space-y-4">
  1702→          <div className="space-y-1.5">
  1703→            <Label htmlFor="complete-score">Score (%) — optional</Label>
  1704→            <Input
  1705→              id="complete-score"
  1706→              type="number"
  1707→              min={0}
  1708→              max={100}
  1709→              placeholder="e.g. 85"
  1710→              value={score}
  1711→              onChange={(e) => setScore(e.target.value)}
  1712→            />
  1713→          </div>
  1714→          <div className="space-y-1.5">
  1715→            <Label htmlFor="complete-cert">Certificate / credential ID — optional</Label>
  1716→            <Input
  1717→              id="complete-cert"
  1718→              placeholder="e.g. CERT-2024-001"
  1719→              value={certificate}
  1720→              onChange={(e) => setCertificate(e.target.value)}
  1721→            />
  1722→          </div>
  1723→        </div>
  1724→        <DialogFooter>
  1725→          <Button variant="outline" onClick={onClose} disabled={saving}>
  1726→            Cancel
  1727→          </Button>
  1728→          <Button onClick={submit} disabled={saving} className="gap-1.5">
  1729→            {saving && <Loader2 className="size-4 animate-spin" />}
  1730→            Mark Complete
  1731→          </Button>
  1732→        </DialogFooter>
  1733→      </DialogContent>
  1734→    </Dialog>
  1735→  );

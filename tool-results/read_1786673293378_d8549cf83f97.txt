   235→export function AssetsModule() {
   236→  const [search, setSearch] = useState("");
   237→  const [type, setType] = useState("");
   238→  const [status, setStatus] = useState("");
   239→  const [view, setView] = useState<"table" | "grid" | "depreciation">("table");
   240→
   241→  const [formOpen, setFormOpen] = useState(false);
   242→  const [editAsset, setEditAsset] = useState<Asset | null>(null);
   243→  const [assignAsset, setAssignAsset] = useState<Asset | null>(null);
   244→  const [returnAsset, setReturnAsset] = useState<Asset | null>(null);
   245→  const [depreciationAsset, setDepreciationAsset] = useState<DepreciationRow | null>(null);
   246→
   247→  const qc = useQueryClient();
   248→  const { data, isLoading, isError } = useQuery({
   249→    queryKey: ["assets", search, type, status],
   250→    queryFn: async () => {
   251→      const params = new URLSearchParams();
   252→      if (search) params.set("search", search);
   253→      if (type) params.set("type", type);
   254→      if (status) params.set("status", status);
   255→      const r = await fetch(`/api/assets?${params.toString()}`);
   256→      if (!r.ok) throw new Error("Failed to load assets");
   257→      return r.json();
   258→    },
   259→  });
   260→
   261→  const assets: Asset[] = data?.items ?? [];
   262→  const total = assets.length;
   263→  const assigned = assets.filter((a) => a.status === "ASSIGNED").length;
   264→  const available = assets.filter((a) => a.status === "AVAILABLE").length;
   265→  const damaged = assets.filter((a) => a.condition === "DAMAGED").length;
   266→
   267→  function openCreate() {
   268→    setEditAsset(null);
   269→    setFormOpen(true);
   270→  }
   271→  function openEdit(a: Asset) {
   272→    setEditAsset(a);
   273→    setFormOpen(true);
   274→  }
   275→  async function deleteAsset(a: Asset) {
   276→    if (!confirm(`Permanently delete asset "${a.name}"?`)) return;
   277→    try {
   278→      const r = await fetch(`/api/assets/${a.id}`, { method: "DELETE" });
   279→      if (!r.ok) throw new Error("Failed to delete asset");
   280→      toast.success(`Asset "${a.name}" deleted.`);
   281→      qc.invalidateQueries({ queryKey: ["assets"] });
   282→    } catch (err: any) {
   283→      toast.error(err?.message || "Failed to delete asset.");
   284→    }
   285→  }
   286→  async function retireAsset(a: Asset) {
   287→    if (!confirm(`Retire asset "${a.name}"? It will be marked RETIRED.`))
   288→      return;
   289→    try {
   290→      const r = await fetch(`/api/assets/${a.id}`, {
   291→        method: "PATCH",
   292→        headers: { "Content-Type": "application/json" },
   293→        body: JSON.stringify({ status: "RETIRED" }),
   294→      });
   295→      if (!r.ok) throw new Error("Failed to retire asset");
   296→      toast.success(`Asset "${a.name}" retired.`);
   297→      qc.invalidateQueries({ queryKey: ["assets"] });
   298→    } catch (err: any) {
   299→      toast.error(err?.message || "Failed to retire asset.");
   300→    }
   301→  }
   302→
   303→  return (
   304→    <div className="space-y-6">
   305→      <PageHeader
   306→        title="Asset Management"
   307→        description="Track company assets and equipment assignments"
   308→        icon={<Package className="size-5" />}
   309→        actions={
   310→          <>
   311→            <ExportButton
   312→              module="assets"
   313→              filters={{ search, status, type }}
   314→            />
   315→            <div className="flex rounded-md border border-border overflow-hidden">
   316→              <Button
   317→                variant={view === "table" ? "default" : "ghost"}
   318→                size="sm"
   319→                className="rounded-none gap-1.5"
   320→                onClick={() => setView("table")}
   321→                aria-label="Table view"
   322→              >
   323→                <List className="size-4" />
   324→                <span className="hidden sm:inline">Table</span>
   325→              </Button>
   326→              <Button
   327→                variant={view === "grid" ? "default" : "ghost"}
   328→                size="sm"
   329→                className="rounded-none gap-1.5"
   330→                onClick={() => setView("grid")}
   331→                aria-label="Grid view"
   332→              >
   333→                <LayoutGrid className="size-4" />
   334→                <span className="hidden sm:inline">Grid</span>
   335→              </Button>
   336→              <Button
   337→                variant={view === "depreciation" ? "default" : "ghost"}
   338→                size="sm"
   339→                className="rounded-none gap-1.5"
   340→                onClick={() => setView("depreciation")}
   341→                aria-label="Depreciation view"
   342→              >
   343→                <TrendingDown className="size-4" />
   344→                <span className="hidden sm:inline">Depreciation</span>
   345→              </Button>
   346→            </div>
   347→            <Button size="sm" onClick={openCreate} className="gap-1.5">
   348→              <Plus className="size-4" />
   349→              <span className="hidden sm:inline">Add Asset</span>
   350→              <span className="sm:hidden">New</span>
   351→            </Button>
   352→          </>
   353→        }
   354→      />
   355→
   356→      {/* KPI cards */}
   357→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3 md:gap-4">
   358→        <KpiCard
   359→          label="Total Assets"
   360→          value={total}
   361→          icon={Package}
   362→          iconClass="bg-primary/10 text-primary"
   363→          footer={<span className="text-muted-foreground">All equipment</span>}
   364→        />
   365→        <KpiCard
   366→          label="Assigned"
   367→          value={assigned}
   368→          icon={PackageCheck}
   369→          iconClass="bg-sky-500/15 text-sky-600"
   370→          footer={<span className="text-muted-foreground">In use</span>}
   371→        />
   372→        <KpiCard
   373→          label="Available"
   374→          value={available}
   375→          icon={PackageOpen}
   376→          iconClass="bg-emerald-500/15 text-emerald-600"
   377→          footer={<span className="text-muted-foreground">Free to assign</span>}
   378→        />
   379→        <KpiCard
   380→          label="Damaged"
   381→          value={damaged}
   382→          icon={AlertTriangle}
   383→          iconClass="bg-rose-500/15 text-rose-600"
   384→          footer={<span className="text-muted-foreground">Needs repair</span>}
   385→        />
   386→      </div>
   387→
   388→      {/* Filter bar */}
   389→      <div className="flex flex-col md:flex-row gap-3">
   390→        <div className="relative flex-1">
   391→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   392→          <Input
   393→            placeholder="Search by name, serial, or assignee…"
   394→            value={search}
   395→            onChange={(e) => setSearch(e.target.value)}
   396→            className="pl-9"
   397→          />
   398→        </div>
   399→        <Select
   400→          value={type || "ALL"}
   401→          onValueChange={(v) => setType(v === "ALL" ? "" : v)}
   402→        >
   403→          <SelectTrigger className="md:w-44">
   404→            <SelectValue placeholder="All types" />
   405→          </SelectTrigger>
   406→          <SelectContent>
   407→            <SelectItem value="ALL">All types</SelectItem>
   408→            {ASSET_TYPES.map((t) => (
   409→              <SelectItem key={t.value} value={t.value}>
   410→                {t.label}
   411→              </SelectItem>
   412→            ))}
   413→          </SelectContent>
   414→        </Select>
   415→        <Select
   416→          value={status || "ALL"}
   417→          onValueChange={(v) => setStatus(v === "ALL" ? "" : v)}
   418→        >
   419→          <SelectTrigger className="md:w-44">
   420→            <SelectValue placeholder="All statuses" />
   421→          </SelectTrigger>
   422→          <SelectContent>
   423→            <SelectItem value="ALL">All statuses</SelectItem>
   424→            {STATUSES.map((s) => (
   425→              <SelectItem key={s} value={s}>
   426→                {s.charAt(0) + s.slice(1).toLowerCase()}
   427→              </SelectItem>
   428→            ))}
   429→          </SelectContent>
   430→        </Select>
   431→      </div>
   432→
   433→      {/* Result count */}
   434→      <div className="flex items-center justify-between text-sm">
   435→        <div className="text-muted-foreground">
   436→          {isLoading ? (
   437→            <Skeleton className="h-4 w-32" />
   438→          ) : (
   439→            <span>
   440→              Showing <span className="font-semibold text-foreground">{assets.length}</span>{" "}
   441→              asset{assets.length === 1 ? "" : "s"}
   442→              {(type || status || search) && " (filtered)"}
   443→            </span>
   444→          )}
   445→        </div>
   446→      </div>
   447→
   448→      {/* Content */}
   449→      {view === "depreciation" ? (
   450→        <DepreciationView
   451→          search={search}
   452→          type={type}
   453→          status={status}
   454→          onInspect={(row) => setDepreciationAsset(row)}
   455→        />
   456→      ) : isLoading ? (
   457→        <AssetsSkeleton view={view} />
   458→      ) : isError ? (
   459→        <EmptyState
   460→          icon={AlertTriangle}
   461→          title="Failed to load assets"
   462→          description="Please try again. If the problem persists, check the dev server."
   463→          actionLabel="Retry"
   464→          onAction={() => qc.invalidateQueries({ queryKey: ["assets"] })}
   465→        />
   466→      ) : assets.length === 0 ? (
   467→        <EmptyState
   468→          icon={Package}
   469→          title={search || type || status ? "No matching assets" : "No assets yet"}
   470→          description={
   471→            search || type || status
   472→              ? "Try adjusting your filters."
   473→              : "Start by adding your first company asset."
   474→          }
   475→          actionLabel="Add Asset"
   476→          onAction={openCreate}
   477→        />
   478→      ) : view === "table" ? (
   479→        <AssetsTable
   480→          assets={assets}
   481→          onEdit={openEdit}
   482→          onDelete={deleteAsset}
   483→          onAssign={(a) => setAssignAsset(a)}
   484→          onReturn={(a) => setReturnAsset(a)}
   485→          onRetire={retireAsset}
   486→        />
   487→      ) : (
   488→        <AssetsGrid
   489→          assets={assets}
   490→          onEdit={openEdit}
   491→          onDelete={deleteAsset}
   492→          onAssign={(a) => setAssignAsset(a)}
   493→          onReturn={(a) => setReturnAsset(a)}
   494→          onRetire={retireAsset}
   495→        />
   496→      )}
   497→
   498→      {/* Create / Edit dialog */}
   499→      <AssetFormDialog
   500→        open={formOpen}
   501→        onOpenChange={(o) => {
   502→          setFormOpen(o);
   503→          if (!o) setEditAsset(null);
   504→        }}
   505→        asset={editAsset}
   506→        onSaved={() => qc.invalidateQueries({ queryKey: ["assets"] })}
   507→      />
   508→
   509→      {/* Assign dialog */}
   510→      <AssignDialog
   511→        asset={assignAsset}
   512→        onClose={() => setAssignAsset(null)}
   513→        onSaved={() => {
   514→          qc.invalidateQueries({ queryKey: ["assets"] });
   515→          setAssignAsset(null);
   516→        }}
   517→      />
   518→
   519→      {/* Return dialog */}
   520→      <ReturnDialog
   521→        asset={returnAsset}
   522→        onClose={() => setReturnAsset(null)}
   523→        onSaved={() => {
   524→          qc.invalidateQueries({ queryKey: ["assets"] });
   525→          setReturnAsset(null);
   526→        }}
   527→      />
   528→
   529→      {/* Depreciation detail dialog */}
   530→      <DepreciationDetailDialog
   531→        asset={depreciationAsset}
   532→        onClose={() => setDepreciationAsset(null)}
   533→      />
   534→    </div>
   535→  );
   536→}

   542→function AssetsTable({
   543→  assets,
   544→  onEdit,
   545→  onDelete,
   546→  onAssign,
   547→  onReturn,
   548→  onRetire,
   549→}: {
   550→  assets: Asset[];
   551→  onEdit: (a: Asset) => void;
   552→  onDelete: (a: Asset) => void;
   553→  onAssign: (a: Asset) => void;
   554→  onReturn: (a: Asset) => void;
   555→  onRetire: (a: Asset) => void;
   556→}) {
   557→  return (
   558→    <Card className="p-0 overflow-hidden border-border/60">
   559→      <div className="overflow-x-auto max-h-[70vh] overflow-y-auto">
   560→        <Table>
   561→          <TableHeader className="sticky top-0 bg-card z-10">
   562→            <TableRow>
   563→              <TableHead>Asset</TableHead>
   564→              <TableHead>Type</TableHead>
   565→              <TableHead className="hidden md:table-cell">Serial</TableHead>
   566→              <TableHead>Condition</TableHead>
   567→              <TableHead>Assigned To</TableHead>
   568→              <TableHead className="hidden lg:table-cell">Assigned Date</TableHead>
   569→              <TableHead>Status</TableHead>
   570→              <TableHead className="text-right">Actions</TableHead>
   571→            </TableRow>
   572→          </TableHeader>
   573→          <TableBody>
   574→            {assets.map((a) => {
   575→              const tm = getTypeMeta(a.type);
   576→              const TIcon = tm.icon;
   577→              return (
   578→                <TableRow key={a.id} className="hover:bg-muted/30">
   579→                  <TableCell>
   580→                    <div className="flex items-center gap-2.5 min-w-0">
   581→                      <div
   582→                        className={cn(
   583→                          "size-9 rounded-lg flex items-center justify-center flex-shrink-0",
   584→                          tm.color
   585→                        )}
   586→                      >
   587→                        <TIcon className="size-4" />
   588→                      </div>
   589→                      <div className="min-w-0">
   590→                        <div className="font-medium text-sm truncate">{a.name}</div>
   591→                        {a.notes && (

   951→function AssetFormBody({
   952→  asset,
   953→  savingState,
   954→  onSaved,
   955→  onCancel,
   956→}: {
   957→  asset: Asset | null;
   958→  savingState: [boolean, (b: boolean) => void];
   959→  onSaved: () => void;
   960→  onCancel: () => void;
   961→}) {
   962→  const [name, setName] = useState(asset?.name ?? "");
   963→  const [type, setType] = useState<AssetType>(asset?.type ?? "LAPTOP");
   964→  const [serialNumber, setSerialNumber] = useState(asset?.serialNumber ?? "");
   965→  const [condition, setCondition] = useState<AssetCondition>(
   966→    asset?.condition ?? "GOOD"
   967→  );
   968→  const [purchaseValue, setPurchaseValue] = useState<string>(
   969→    asset?.purchaseValue !== undefined && asset.purchaseValue !== null
   970→      ? String(asset.purchaseValue)
   971→      : "1000"
   972→  );
   973→  const [notes, setNotes] = useState(asset?.notes ?? "");
   974→  const [saving, setSaving] = savingState;
   975→
   976→  async function submit() {
   977→    if (!name.trim()) {
   978→      toast.error("Asset name is required.");
   979→      return;
   980→    }
   981→    const pvNum = Number(purchaseValue);
   982→    const pv =
   983→      purchaseValue.trim() !== "" && !isNaN(pvNum) && pvNum >= 0
   984→        ? pvNum
   985→        : 1000;
   986→    setSaving(true);
   987→    try {
   988→      const body = { name, type, serialNumber, condition, purchaseValue: pv, notes };
   989→      const r = asset
   990→        ? await fetch(`/api/assets/${asset.id}`, {
   991→            method: "PATCH",
   992→            headers: { "Content-Type": "application/json" },
   993→            body: JSON.stringify(body),
   994→          })
   995→        : await fetch("/api/assets", {
   996→            method: "POST",
   997→            headers: { "Content-Type": "application/json" },
   998→            body: JSON.stringify(body),
   999→          });
  1000→      if (!r.ok) {
  1001→        const e = await r.json().catch(() => ({}));
  1002→        throw new Error(e?.error || "Failed to save asset");
  1003→      }
  1004→      onSaved();
  1005→    } catch (err: any) {
  1006→      toast.error(err?.message || "Failed to save asset.");
  1007→    } finally {
  1008→      setSaving(false);
  1009→    }
  1010→  }
  1011→
  1012→  return (
  1013→    <div className="space-y-4">
  1014→      <div className="space-y-1.5">
  1015→        <Label htmlFor="asset-name">Asset name *</Label>
  1016→        <Input
  1017→          id="asset-name"
  1018→          placeholder="e.g. MacBook Pro 16&quot; M3"
  1019→          value={name}
  1020→          onChange={(e) => setName(e.target.value)}
  1021→        />
  1022→      </div>
  1023→      <div className="grid grid-cols-2 gap-3">
  1024→        <div className="space-y-1.5">
  1025→          <Label htmlFor="asset-type">Type</Label>
  1026→          <Select value={type} onValueChange={(v) => setType(v as AssetType)}>
  1027→            <SelectTrigger id="asset-type">
  1028→              <SelectValue />
  1029→            </SelectTrigger>
  1030→            <SelectContent>
  1031→              {ASSET_TYPES.map((t) => (
  1032→                <SelectItem key={t.value} value={t.value}>
  1033→                  <span className="flex items-center gap-2">
  1034→                    <t.icon className="size-3.5" />
  1035→                    {t.label}
  1036→                  </span>
  1037→                </SelectItem>
  1038→              ))}
  1039→            </SelectContent>
  1040→          </Select>
  1041→        </div>
  1042→        <div className="space-y-1.5">
  1043→          <Label htmlFor="asset-condition">Condition</Label>
  1044→          <Select
  1045→            value={condition}
  1046→            onValueChange={(v) => setCondition(v as AssetCondition)}
  1047→          >
  1048→            <SelectTrigger id="asset-condition">
  1049→              <SelectValue />
  1050→            </SelectTrigger>
  1051→            <SelectContent>
  1052→              {CONDITIONS.map((c) => (
  1053→                <SelectItem key={c} value={c}>
  1054→                  {c.charAt(0) + c.slice(1).toLowerCase()}
  1055→                </SelectItem>
  1056→              ))}
  1057→            </SelectContent>
  1058→          </Select>
  1059→        </div>
  1060→      </div>
  1061→      <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
  1062→        <div className="space-y-1.5">
  1063→          <Label htmlFor="asset-serial">Serial number</Label>
  1064→          <Input
  1065→            id="asset-serial"

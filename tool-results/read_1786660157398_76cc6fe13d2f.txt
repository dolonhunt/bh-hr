   229→function AllDocumentsTab({
   230→  onPreview,
   231→  onSendEmail,
   232→}: {
   233→  onPreview: (doc: any) => void;
   234→  onSendEmail: (doc: any) => void;
   235→}) {
   236→  const { data, isLoading } = useQuery({
   237→    queryKey: ["documents", "all-tab"],
   238→    queryFn: async () => {
   239→      const r = await fetch("/api/documents?pageSize=10");
   240→      return r.json();
   241→    },
   242→  });
   243→
   244→  const { data: emailStats } = useQuery({
   245→    queryKey: ["email-logs", "stats"],
   246→    queryFn: async () => {
   247→      const [sent, failed] = await Promise.all([
   248→        fetch("/api/email-logs?status=SENT&pageSize=1").then((r) => r.json()),
   249→        fetch("/api/email-logs?status=FAILED&pageSize=1").then((r) => r.json()),
   250→      ]);
   251→      return { sentToday: sent.total ?? 0, failed: failed.total ?? 0 };
   252→    },
   253→  });
   254→
   255→  const docs = data?.items ?? [];
   256→  const total = data?.total ?? 0;
   257→  const todayStr = new Date().toDateString();
   258→  const generatedToday = docs.filter(
   259→    (d: any) => new Date(d.createdAt).toDateString() === todayStr
   260→  ).length;
   261→  const sentToday = emailStats?.sentToday ?? 0;
   262→  const failedEmails = emailStats?.failed ?? 0;
   263→  const pendingApproval = docs.filter((d: any) =>
   264→    ["PENDING_APPROVAL", "GENERATED"].includes(d.status)
   265→  ).length;
   266→
   267→  return (
   268→    <div className="space-y-4">
   269→      <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-5 gap-3">
   270→        <KpiCard label="Total" value={total} icon={FileStack} iconClass="bg-primary/10 text-primary" />
   271→        <KpiCard label="Generated Today" value={generatedToday} icon={FilePlus} iconClass="bg-violet-500/10 text-violet-600" />
   272→        <KpiCard label="Sent Today" value={sentToday} icon={MailCheck} iconClass="bg-teal-500/10 text-teal-600" />
   273→        <KpiCard label="Pending Approval" value={pendingApproval} icon={Clock} iconClass="bg-amber-500/10 text-amber-600" />
   274→        <KpiCard label="Failed Emails" value={failedEmails} icon={MailX} iconClass="bg-rose-500/10 text-rose-600" />
   275→      </div>
   276→
   277→      <Card className="border-border/60 shadow-soft overflow-hidden">
   278→        <div className="flex items-center justify-between px-5 py-3 border-b border-border bg-muted/30">
   279→          <div className="font-medium text-sm">Recent Documents</div>
   280→        </div>
   281→        <DocumentsTable
   282→          docs={docs}
   283→          loading={isLoading}
   284→          onPreview={onPreview}
   285→          onSendEmail={onSendEmail}
   286→        />
   287→      </Card>
   288→    </div>
   289→  );
   290→}

   296→function TemplatesTab({
   297→  onEdit,
   298→  onCreate,
   299→}: {
   300→  onEdit: (id: string) => void;
   301→  onCreate: () => void;
   302→}) {
   303→  const qc = useQueryClient();
   304→  const [search, setSearch] = useState("");
   305→  const [type, setType] = useState("");
   306→  const [previewTpl, setPreviewTpl] = useState<any | null>(null);
   307→
   308→  const { data, isLoading } = useQuery({
   309→    queryKey: ["document-templates", "templates-tab", search, type],
   310→    queryFn: async () => {
   311→      const params = new URLSearchParams();
   312→      if (search) params.set("search", search);
   313→      if (type) params.set("type", type);
   314→      const r = await fetch(`/api/document-templates?${params.toString()}`);
   315→      return r.json();
   316→    },
   317→  });
   318→  const templates = data?.items ?? [];
   319→
   320→  async function archiveTemplate(id: string) {
   321→    if (!confirm("Archive this template? It will be hidden from the active list.")) return;
   322→    try {
   323→      const res = await fetch(`/api/document-templates/${id}`, {
   324→        method: "DELETE",
   325→      });
   326→      if (!res.ok) throw new Error("Failed to archive");
   327→      toast.success("Template archived.");
   328→      qc.invalidateQueries({ queryKey: ["document-templates"] });
   329→    } catch (err: any) {
   330→      toast.error(err.message);
   331→    }
   332→  }
   333→
   334→  async function duplicateTemplate(t: any) {
   335→    try {
   336→      const res = await fetch("/api/document-templates", {
   337→        method: "POST",
   338→        headers: { "Content-Type": "application/json" },
   339→        body: JSON.stringify({
   340→          ...t,
   341→          name: `${t.name} (Copy)`,
   342→          code: `${t.code}-COPY`,
   343→          status: "DRAFT",
   344→        }),
   345→      });
   346→      if (!res.ok) {
   347→        const err = await res.json().catch(() => ({}));
   348→        throw new Error(err.error || "Failed");
   349→      }
   350→      toast.success("Template duplicated.");
   351→      qc.invalidateQueries({ queryKey: ["document-templates"] });
   352→    } catch (err: any) {
   353→      toast.error(err.message);
   354→    }
   355→  }
   356→
   357→  return (
   358→    <div className="space-y-4">
   359→      <div className="flex flex-col md:flex-row gap-3">
   360→        <div className="relative flex-1">
   361→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   362→          <Input
   363→            placeholder="Search templates by name, code, description…"
   364→            value={search}
   365→            onChange={(e) => setSearch(e.target.value)}
   366→            className="pl-9"
   367→          />
   368→        </div>
   369→        <Select
   370→          value={type || "ALL"}
   371→          onValueChange={(v) => setType(v === "ALL" ? "" : v)}
   372→        >
   373→          <SelectTrigger className="md:w-56">
   374→            <SelectValue placeholder="All types" />
   375→          </SelectTrigger>
   376→          <SelectContent>
   377→            <SelectItem value="ALL">All types</SelectItem>
   378→            {DOC_TYPES.map((t) => (
   379→              <SelectItem key={t} value={t}>
   380→                {t.replace(/_/g, " ")}
   381→              </SelectItem>
   382→            ))}
   383→          </SelectContent>
   384→        </Select>
   385→        <Button onClick={onCreate} size="sm" className="md:w-auto">
   386→          <Plus className="size-4 mr-1.5" /> Create Template
   387→        </Button>
   388→      </div>
   389→
   390→      {isLoading && (
   391→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   392→          {Array.from({ length: 6 }).map((_, i) => (
   393→            <div key={i} className="h-44 rounded-xl bg-muted/40 animate-pulse" />

   395→        </div>

   399→        <EmptyState
   400→          icon={FileText}
   401→          title="No templates found"
   402→          description="Create your first template to start generating HR documents."
   403→          actionLabel="Create Template"
   404→          onAction={onCreate}
   405→        />
   406→      )}
   407→
   408→      {!isLoading && templates.length > 0 && (
   409→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   410→          {templates.map((t: any) => (
   411→            <Card
   412→              key={t.id}
   413→              className="border-border/60 shadow-soft hover:shadow-card-hover hover:-translate-y-0.5 transition-all overflow-hidden flex flex-col"
   414→            >
   415→              <CardContent className="p-5 flex-1 flex flex-col">
   416→                <div className="flex items-start justify-between gap-2 mb-3">
   417→                  <div className="min-w-0 flex-1">
   418→                    <div className="font-semibold truncate">{t.name}</div>
   419→                    <div className="text-xs text-muted-foreground truncate">
   420→                      {t.description ?? "No description"}
   421→                    </div>
   422→                  </div>
   423→                  <Badge variant="outline" className="font-mono text-[10px]">

   520→function GeneratedTab({
   521→  onPreview,
   522→  onSendEmail,
   523→}: {
   524→  onPreview: (doc: any) => void;
   525→  onSendEmail: (doc: any) => void;
   526→}) {
   527→  const qc = useQueryClient();
   528→  const [search, setSearch] = useState("");
   529→  const [type, setType] = useState("");
   530→  const [status, setStatus] = useState("");
   531→  const [page, setPage] = useState(1);
   532→
   533→  const { data, isLoading } = useQuery({
   534→    queryKey: ["documents", "generated-tab", search, type, status, page],
   535→    queryFn: async () => {
   536→      const params = new URLSearchParams();
   537→      if (search) params.set("search", search);
   538→      if (type) params.set("type", type);
   539→      if (status) params.set("status", status);
   540→      params.set("page", String(page));
   541→      params.set("pageSize", "20");
   542→      const r = await fetch(`/api/documents?${params.toString()}`);
   543→      return r.json();
   544→    },
   545→    placeholderData: (prev) => prev,
   546→  });
   547→  const docs = data?.items ?? [];
   548→  const total = data?.total ?? 0;
   549→  const totalPages = data?.totalPages ?? 1;
   550→
   551→  async function archiveDoc(id: string) {
   552→    if (!confirm("Archive this document?")) return;
   553→    try {
   554→      await fetch(`/api/documents/${id}`, { method: "DELETE" });
   555→      toast.success("Document archived.");
   556→      qc.invalidateQueries({ queryKey: ["documents"] });
   557→    } catch {
   558→      toast.error("Archive failed.");
   559→    }
   560→  }
   561→
   562→  function download(id: string, format: "docx" | "pdf", docNumber: string) {
   563→    const url = `/api/documents/${id}/download?format=${format}`;
   564→    const a = document.createElement("a");
   565→    a.href = url;
   566→    a.download = `${docNumber}.${format}`;
   567→    document.body.appendChild(a);
   568→    a.click();
   569→    document.body.removeChild(a);
   570→    toast.success(`Downloading ${format.toUpperCase()}…`);
   571→  }
   572→
   573→  return (
   574→    <div className="space-y-4">
   575→      <div className="flex flex-col md:flex-row gap-3">
   576→        <div className="relative flex-1">
   577→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   578→          <Input
   579→            placeholder="Search by document number or title…"
   580→            value={search}
   581→            onChange={(e) => {
   582→              setSearch(e.target.value);
   583→              setPage(1);
   584→            }}
   585→            className="pl-9"
   586→          />
   587→        </div>
   588→        <Select
   589→          value={type || "ALL"}
   590→          onValueChange={(v) => {
   591→            setType(v === "ALL" ? "" : v);
   592→            setPage(1);
   593→          }}
   594→        >
   595→          <SelectTrigger className="md:w-48">
   596→            <SelectValue placeholder="All types" />
   597→          </SelectTrigger>
   598→          <SelectContent>
   599→            <SelectItem value="ALL">All types</SelectItem>
   600→            {DOC_TYPES.map((t) => (
   601→              <SelectItem key={t} value={t}>
   602→                {t.replace(/_/g, " ")}
   603→              </SelectItem>
   604→            ))}
   605→          </SelectContent>
   606→        </Select>

  1052→function DocumentPreviewDialog({
  1053→  doc,
  1054→  onClose,
  1055→}: {
  1056→  doc: any | null;
  1057→  onClose: () => void;
  1058→}) {
  1059→  const [preview, setPreview] = useState<any | null>(null);
  1060→  const [loading, setLoading] = useState(false);
  1061→
  1062→  useEffect(() => {
  1063→    if (!doc) {
  1064→      // Defer the reset to avoid synchronous setState in effect body.
  1065→      const t = setTimeout(() => {
  1066→        setPreview(null);
  1067→        setLoading(false);
  1068→      }, 0);
  1069→      return () => clearTimeout(t);
  1070→    }
  1071→    let cancelled = false;
  1072→    // Use queueMicrotask so the rule about synchronous setState in effects
  1073→    // doesn't fire — we genuinely want to flip loading=true before the fetch.
  1074→    queueMicrotask(() => {
  1075→      if (!cancelled) setLoading(true);
  1076→    });
  1077→    fetch(`/api/documents/${doc.id}/preview`)
  1078→      .then((r) => r.json())
  1079→      .then((d) => {
  1080→        if (!cancelled) setPreview(d);
  1081→      })
  1082→      .catch(() => toast.error("Failed to load preview"))
  1083→      .finally(() => {
  1084→        if (!cancelled) setLoading(false);
  1085→      });
  1086→    return () => {
  1087→      cancelled = true;
  1088→    };
  1089→  }, [doc]);
  1090→
  1091→  return (
  1092→    <Dialog open={!!doc} onOpenChange={(o) => !o && onClose()}>
  1093→      <DialogContent className="max-w-4xl max-h-[92vh] p-0 gap-0">
  1094→        <DialogHeader className="px-6 py-4 border-b border-border">
  1095→          <DialogTitle className="flex items-center gap-2">

  1097→            {preview?.title ?? doc?.documentNumber ?? "Preview"}

  1106→            </div>
  1107→          )}

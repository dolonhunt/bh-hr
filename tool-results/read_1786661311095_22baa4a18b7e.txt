   237→function AllDocumentsTab({
   238→  onPreview,
   239→  onSendEmail,
   240→}: {
   241→  onPreview: (doc: any) => void;
   242→  onSendEmail: (doc: any) => void;
   243→}) {
   244→  const { data, isLoading } = useQuery({
   245→    queryKey: ["documents", "all-tab"],
   246→    queryFn: async () => {
   247→      const r = await fetch("/api/documents?pageSize=10");
   248→      return r.json();
   249→    },
   250→  });
   251→
   252→  const { data: emailStats } = useQuery({
   253→    queryKey: ["email-logs", "stats"],
   254→    queryFn: async () => {
   255→      const [sent, failed] = await Promise.all([
   256→        fetch("/api/email-logs?status=SENT&pageSize=1").then((r) => r.json()),
   257→        fetch("/api/email-logs?status=FAILED&pageSize=1").then((r) => r.json()),
   258→      ]);
   259→      return { sentToday: sent.total ?? 0, failed: failed.total ?? 0 };
   260→    },
   261→  });
   262→
   263→  const docs = data?.items ?? [];
   264→  const total = data?.total ?? 0;
   265→  const todayStr = new Date().toDateString();
   266→  const generatedToday = docs.filter(
   267→    (d: any) => new Date(d.createdAt).toDateString() === todayStr
   268→  ).length;
   269→  const sentToday = emailStats?.sentToday ?? 0;
   270→  const failedEmails = emailStats?.failed ?? 0;
   271→  const pendingApproval = docs.filter((d: any) =>
   272→    ["PENDING_APPROVAL", "GENERATED"].includes(d.status)
   273→  ).length;
   274→
   275→  return (
   276→    <div className="space-y-4">
   277→      <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-5 gap-3">
   278→        <KpiCard label="Total" value={total} icon={FileStack} iconClass="bg-primary/10 text-primary" />
   279→        <KpiCard label="Generated Today" value={generatedToday} icon={FilePlus} iconClass="bg-violet-500/10 text-violet-600" />
   280→        <KpiCard label="Sent Today" value={sentToday} icon={MailCheck} iconClass="bg-teal-500/10 text-teal-600" />
   281→        <KpiCard label="Pending Approval" value={pendingApproval} icon={Clock} iconClass="bg-amber-500/10 text-amber-600" />
   282→        <KpiCard label="Failed Emails" value={failedEmails} icon={MailX} iconClass="bg-rose-500/10 text-rose-600" />
   283→      </div>
   284→
   285→      <Card className="border-border/60 shadow-soft overflow-hidden">
   286→        <div className="flex items-center justify-between px-5 py-3 border-b border-border bg-muted/30">
   287→          <div className="font-medium text-sm">Recent Documents</div>
   288→        </div>
   289→        <DocumentsTable
   290→          docs={docs}
   291→          loading={isLoading}
   292→          onPreview={onPreview}
   293→          onSendEmail={onSendEmail}
   294→        />
   295→      </Card>
   296→    </div>
   297→  );
   298→}

   304→function TemplatesTab({
   305→  onEdit,
   306→  onCreate,
   307→}: {
   308→  onEdit: (id: string) => void;
   309→  onCreate: () => void;
   310→}) {
   311→  const qc = useQueryClient();
   312→  const [search, setSearch] = useState("");
   313→  const [type, setType] = useState("");
   314→  const [previewTpl, setPreviewTpl] = useState<any | null>(null);
   315→
   316→  const { data, isLoading } = useQuery({
   317→    queryKey: ["document-templates", "templates-tab", search, type],
   318→    queryFn: async () => {
   319→      const params = new URLSearchParams();
   320→      if (search) params.set("search", search);
   321→      if (type) params.set("type", type);
   322→      const r = await fetch(`/api/document-templates?${params.toString()}`);
   323→      return r.json();
   324→    },
   325→  });
   326→  const templates = data?.items ?? [];
   327→
   328→  async function archiveTemplate(id: string) {
   329→    if (!confirm("Archive this template? It will be hidden from the active list.")) return;
   330→    try {
   331→      const res = await fetch(`/api/document-templates/${id}`, {
   332→        method: "DELETE",
   333→      });
   334→      if (!res.ok) throw new Error("Failed to archive");
   335→      toast.success("Template archived.");
   336→      qc.invalidateQueries({ queryKey: ["document-templates"] });
   337→    } catch (err: any) {
   338→      toast.error(err.message);
   339→    }
   340→  }
   341→
   342→  async function duplicateTemplate(t: any) {
   343→    try {
   344→      const res = await fetch("/api/document-templates", {
   345→        method: "POST",
   346→        headers: { "Content-Type": "application/json" },
   347→        body: JSON.stringify({
   348→          ...t,
   349→          name: `${t.name} (Copy)`,
   350→          code: `${t.code}-COPY`,
   351→          status: "DRAFT",
   352→        }),
   353→      });
   354→      if (!res.ok) {
   355→        const err = await res.json().catch(() => ({}));
   356→        throw new Error(err.error || "Failed");
   357→      }
   358→      toast.success("Template duplicated.");
   359→      qc.invalidateQueries({ queryKey: ["document-templates"] });
   360→    } catch (err: any) {
   361→      toast.error(err.message);
   362→    }
   363→  }
   364→
   365→  return (
   366→    <div className="space-y-4">
   367→      <div className="flex flex-col md:flex-row gap-3">
   368→        <div className="relative flex-1">
   369→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   370→          <Input
   371→            placeholder="Search templates by name, code, description…"
   372→            value={search}
   373→            onChange={(e) => setSearch(e.target.value)}
   374→            className="pl-9"
   375→          />
   376→        </div>
   377→        <Select
   378→          value={type || "ALL"}
   379→          onValueChange={(v) => setType(v === "ALL" ? "" : v)}
   380→        >
   381→          <SelectTrigger className="md:w-56">
   382→            <SelectValue placeholder="All types" />
   383→          </SelectTrigger>
   384→          <SelectContent>
   385→            <SelectItem value="ALL">All types</SelectItem>
   386→            {DOC_TYPES.map((t) => (
   387→              <SelectItem key={t} value={t}>
   388→                {t.replace(/_/g, " ")}
   389→              </SelectItem>
   390→            ))}
   391→          </SelectContent>
   392→        </Select>
   393→        <Button onClick={onCreate} size="sm" className="md:w-auto">
   394→          <Plus className="size-4 mr-1.5" /> Create Template
   395→        </Button>
   396→      </div>
   397→
   398→      {isLoading && (
   399→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   400→          {Array.from({ length: 6 }).map((_, i) => (
   401→            <div key={i} className="h-44 rounded-xl bg-muted/40 animate-pulse" />
   402→          ))}
   403→        </div>
   404→      )}

   407→        <EmptyState
   408→          icon={FileText}
   409→          title="No templates found"
   410→          description="Create your first template to start generating HR documents."
   411→          actionLabel="Create Template"
   412→          onAction={onCreate}
   413→        />
   414→      )}
   415→
   416→      {!isLoading && templates.length > 0 && (
   417→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   418→          {templates.map((t: any) => (
   419→            <Card
   420→              key={t.id}
   421→              className="border-border/60 shadow-soft hover:shadow-card-hover hover:-translate-y-0.5 transition-all overflow-hidden flex flex-col"
   422→            >
   423→              <CardContent className="p-5 flex-1 flex flex-col">
   424→                <div className="flex items-start justify-between gap-2 mb-3">
   425→                  <div className="min-w-0 flex-1">

   428→                      {t.description ?? "No description"}
   429→                    </div>
   430→                  </div>
   431→                  <Badge variant="outline" className="font-mono text-[10px]">

   528→function GeneratedTab({
   529→  onPreview,
   530→  onSendEmail,
   531→}: {
   532→  onPreview: (doc: any) => void;
   533→  onSendEmail: (doc: any) => void;
   534→}) {
   535→  const qc = useQueryClient();
   536→  const [search, setSearch] = useState("");
   537→  const [type, setType] = useState("");
   538→  const [status, setStatus] = useState("");
   539→  const [page, setPage] = useState(1);
   540→
   541→  const { data, isLoading } = useQuery({
   542→    queryKey: ["documents", "generated-tab", search, type, status, page],
   543→    queryFn: async () => {
   544→      const params = new URLSearchParams();
   545→      if (search) params.set("search", search);
   546→      if (type) params.set("type", type);
   547→      if (status) params.set("status", status);
   548→      params.set("page", String(page));
   549→      params.set("pageSize", "20");
   550→      const r = await fetch(`/api/documents?${params.toString()}`);
   551→      return r.json();
   552→    },
   553→    placeholderData: (prev) => prev,
   554→  });
   555→  const docs = data?.items ?? [];
   556→  const total = data?.total ?? 0;
   557→  const totalPages = data?.totalPages ?? 1;
   558→
   559→  async function archiveDoc(id: string) {
   560→    if (!confirm("Archive this document?")) return;
   561→    try {
   562→      await fetch(`/api/documents/${id}`, { method: "DELETE" });
   563→      toast.success("Document archived.");
   564→      qc.invalidateQueries({ queryKey: ["documents"] });
   565→    } catch {
   566→      toast.error("Archive failed.");
   567→    }
   568→  }
   569→
   570→  function download(id: string, format: "docx" | "pdf", docNumber: string) {
   571→    const url = `/api/documents/${id}/download?format=${format}`;
   572→    const a = document.createElement("a");
   573→    a.href = url;
   574→    a.download = `${docNumber}.${format}`;
   575→    document.body.appendChild(a);
   576→    a.click();
   577→    document.body.removeChild(a);
   578→    toast.success(`Downloading ${format.toUpperCase()}…`);
   579→  }
   580→
   581→  return (
   582→    <div className="space-y-4">
   583→      <div className="flex flex-col md:flex-row gap-3">
   584→        <div className="relative flex-1">
   585→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   586→          <Input
   587→            placeholder="Search by document number or title…"
   588→            value={search}
   589→            onChange={(e) => {
   590→              setSearch(e.target.value);
   591→              setPage(1);
   592→            }}
   593→            className="pl-9"
   594→          />
   595→        </div>
   596→        <Select
   597→          value={type || "ALL"}
   598→          onValueChange={(v) => {
   599→            setType(v === "ALL" ? "" : v);
   600→            setPage(1);
   601→          }}
   602→        >
   603→          <SelectTrigger className="md:w-48">
   604→            <SelectValue placeholder="All types" />

   877→// Status pipeline shown as small horizontal pills.
   878→const STATUS_PIPELINE: { key: string; label: string }[] = [
   879→  { key: "DRAFT", label: "Draft" },
   880→  { key: "GENERATED", label: "Generated" },
   881→  { key: "PENDING_APPROVAL", label: "Pending" },
   882→  { key: "APPROVED", label: "Approved" },
   883→  { key: "ISSUED", label: "Issued" },
   884→  { key: "SENT", label: "Sent" },
   885→];
   886→
   887→function StatusFlowPills({ status }: { status: string }) {
   888→  const currentIdx = STATUS_PIPELINE.findIndex((s) => s.key === status);
   889→  if (status === "ARCHIVED") {
   890→    return <StatusBadge status="ARCHIVED" />;
   891→  }
   892→  if (currentIdx === -1) {
   893→    return <StatusBadge status={status} />;
   894→  }
   895→  return (
   896→    <div className="flex items-center gap-0.5 flex-wrap">
   897→      {STATUS_PIPELINE.map((s, idx) => {
   898→        const isCurrent = idx === currentIdx;
   899→        const isPast = idx < currentIdx;
   900→        return (
   901→          <div key={s.key} className="flex items-center">
   902→            <span
   903→              className={cn(
   904→                "inline-flex items-center px-1.5 h-5 rounded text-[9px] font-medium leading-none",
   905→                isCurrent &&
   906→                  "bg-primary text-primary-foreground shadow-soft",
   907→                isPast && "bg-primary/15 text-primary",
   908→                !isCurrent && !isPast &&
   909→                  "bg-muted text-muted-foreground"
   910→              )}
   911→              title={s.label}
   912→            >
   913→              {s.label}
   914→            </span>
   915→            {idx < STATUS_PIPELINE.length - 1 && (
   916→              <ChevronRight
   917→                className={cn(
   918→                  "size-2.5 mx-0.5 flex-shrink-0",
   919→                  idx < currentIdx ? "text-primary" : "text-muted-foreground/50"
   920→                )}
   921→              />
   922→            )}
   923→          </div>
   924→        );
   925→      })}
   926→    </div>
   927→  );

  1262→function DocumentPreviewDialog({
  1263→  doc,
  1264→  onClose,
  1265→}: {
  1266→  doc: any | null;
  1267→  onClose: () => void;
  1268→}) {
  1269→  const [preview, setPreview] = useState<any | null>(null);
  1270→  const [loading, setLoading] = useState(false);
  1271→
  1272→  useEffect(() => {
  1273→    if (!doc) {
  1274→      // Defer the reset to avoid synchronous setState in effect body.
  1275→      const t = setTimeout(() => {
  1276→        setPreview(null);
  1277→        setLoading(false);
  1278→      }, 0);
  1279→      return () => clearTimeout(t);
  1280→    }
  1281→    let cancelled = false;
  1282→    // Use queueMicrotask so the rule about synchronous setState in effects
  1283→    // doesn't fire — we genuinely want to flip loading=true before the fetch.
  1284→    queueMicrotask(() => {
  1285→      if (!cancelled) setLoading(true);
  1286→    });
  1287→    fetch(`/api/documents/${doc.id}/preview`)
  1288→      .then((r) => r.json())
  1289→      .then((d) => {
  1290→        if (!cancelled) setPreview(d);
  1291→      })
  1292→      .catch(() => toast.error("Failed to load preview"))
  1293→      .finally(() => {
  1294→        if (!cancelled) setLoading(false);
  1295→      });
  1296→    return () => {
  1297→      cancelled = true;
  1298→    };
  1299→  }, [doc]);
  1300→
  1301→  return (
  1302→    <Dialog open={!!doc} onOpenChange={(o) => !o && onClose()}>
  1303→      <DialogContent className="max-w-4xl max-h-[92vh] p-0 gap-0">
  1304→        <DialogHeader className="px-6 py-4 border-b border-border">
  1305→          <DialogTitle className="flex items-center gap-2">
  1306→            <Eye className="size-5 text-primary" />
  1307→            {preview?.title ?? doc?.documentNumber ?? "Preview"}
  1308→          </DialogTitle>
  1309→          {preview && (

  1311→              <span className="font-mono">{preview.documentNumber}</span>

  1313→              <span>{preview.employee?.fullName}</span>
  1314→              <span>·</span>
  1315→              <StatusBadge status={preview.status} />
  1316→            </div>
  1317→          )}
  1318→        </DialogHeader>
  1319→        <ScrollArea className="max-h-[70vh]">
  1320→          <div className="px-6 py-5">
  1321→            {loading && (
  1322→              <div className="p-12 flex flex-col items-center text-muted-foreground">

  1401→function DirectSendEmailDialog({
  1402→  doc,
  1403→  onClose,
  1404→}: {
  1405→  doc: any | null;
  1406→  onClose: () => void;
  1407→}) {
  1408→  const qc = useQueryClient();
  1409→  const [sending, setSending] = useState(false);
  1410→  const [to, setTo] = useState("");
  1411→  const [cc, setCc] = useState("");
  1412→  const [bcc, setBcc] = useState("");
  1413→  const [subject, setSubject] = useState("");
  1414→  const [body, setBody] = useState("");
  1415→
  1416→  useEffect(() => {
  1417→    if (!doc) return;
  1418→    // Fetch full document + template to get email subject/body presets.
  1419→    fetch(`/api/documents/${doc.id}`)
  1420→      .then((r) => r.json())
  1421→      .then((d) => {
  1422→        setTo(d.employee?.officialEmail ?? "");
  1423→        setSubject(
  1424→          d.template?.emailSubject
  1425→            ? d.template.emailSubject.replace(
  1426→                /\{\{company\.name\}\}/g,
  1427→                "the company"
  1428→              )
  1429→            : `Your document - ${d.documentNumber}`
  1430→        );
  1431→        setBody(
  1432→          d.template?.emailBody ||
  1433→            `Dear ${d.employee?.fullName ?? ""},\n\nPlease find attached your document ${d.documentNumber}.\n\nRegards,\nHR Team`
  1434→        );
  1435→      })
  1436→      .catch(() => {
  1437→        setTo("");
  1438→        setSubject(`Your document - ${doc.documentNumber}`);
  1439→        setBody("");
  1440→      });
  1441→  }, [doc]);
  1442→
  1443→  async function handleSend() {
  1444→    if (!doc) return;
  1445→    if (!to || !subject || !body) {
  1446→      toast.error("To, subject and body are required.");
  1447→      return;
  1448→    }
  1449→    setSending(true);
  1450→    try {
  1451→      const res = await fetch(`/api/documents/${doc.id}/send-email`, {
  1452→        method: "POST",
  1453→        headers: { "Content-Type": "application/json" },
  1454→        body: JSON.stringify({ to, cc, bcc, subject, body }),
  1455→      });
  1456→      if (!res.ok) {
  1457→        const err = await res.json().catch(() => ({}));
  1458→        throw new Error(err.error || "Send failed");
  1459→      }
  1460→      toast.success(`Email sent to ${to}`);
  1461→      qc.invalidateQueries({ queryKey: ["documents"] });
  1462→      qc.invalidateQueries({ queryKey: ["email-logs"] });
  1463→      qc.invalidateQueries({ queryKey: ["dashboard"] });
  1464→      onClose();
  1465→    } catch (err: any) {
  1466→      toast.error(err.message);
  1467→    } finally {
  1468→      setSending(false);
  1469→    }
  1470→  }
  1471→
  1472→  return (
  1473→    <Dialog open={!!doc} onOpenChange={(o) => !o && onClose()}>
  1474→      <DialogContent className="max-w-2xl">
  1475→        <DialogHeader>
  1476→          <DialogTitle className="flex items-center gap-2">
  1477→            <Mail className="size-5 text-primary" />
  1478→            Send Document via Email
  1479→          </DialogTitle>
  1480→        </DialogHeader>
  1481→        <div className="space-y-3">
  1482→          <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
  1483→            <div className="md:col-span-2">
  1484→              <label className="text-xs font-medium text-muted-foreground mb-1.5 block">
  1485→                To *
  1486→              </label>
  1487→              <Input
  1488→                type="email"
  1489→                value={to}
  1490→                onChange={(e) => setTo(e.target.value)}
  1491→              />
  1492→            </div>
  1493→            <div>
  1494→              <label className="text-xs font-medium text-muted-foreground mb-1.5 block">
  1495→                CC

  1500→              <label className="text-xs font-medium text-muted-foreground mb-1.5 block">

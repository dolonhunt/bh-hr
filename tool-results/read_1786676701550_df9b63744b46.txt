   263→
   264→function AllDocumentsTab({
   265→  onPreview,
   266→  onSendEmail,
   267→}: {
   268→  onPreview: (doc: any) => void;
   269→  onSendEmail: (doc: any) => void;
   270→}) {
   271→  const { data, isLoading } = useQuery({
   272→    queryKey: ["documents", "all-tab"],
   273→    queryFn: async () => {
   274→      const r = await fetch("/api/documents?pageSize=10");
   275→      return r.json();
   276→    },
   277→  });
   278→
   279→  const { data: emailStats } = useQuery({
   280→    queryKey: ["email-logs", "stats"],
   281→    queryFn: async () => {
   282→      const [sent, failed] = await Promise.all([
   283→        fetch("/api/email-logs?status=SENT&pageSize=1").then((r) => r.json()),
   284→        fetch("/api/email-logs?status=FAILED&pageSize=1").then((r) => r.json()),
   285→      ]);
   286→      return { sentToday: sent.total ?? 0, failed: failed.total ?? 0 };
   287→    },
   288→  });
   289→
   290→  const docs = data?.items ?? [];
   291→  const total = data?.total ?? 0;
   292→  const todayStr = new Date().toDateString();
   293→  const generatedToday = docs.filter(
   294→    (d: any) => new Date(d.createdAt).toDateString() === todayStr
   295→  ).length;
   296→  const sentToday = emailStats?.sentToday ?? 0;
   297→  const failedEmails = emailStats?.failed ?? 0;
   298→  const pendingApproval = docs.filter((d: any) =>
   299→    ["PENDING_APPROVAL", "GENERATED"].includes(d.status)
   300→  ).length;
   301→
   302→  return (
   303→    <div className="space-y-4">
   304→      <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-5 gap-3 sm:gap-3">
   305→        <KpiCard label="Total" value={total} icon={FileStack} iconClass="bg-primary/10 text-primary" />
   306→        <KpiCard label="Generated Today" value={generatedToday} icon={FilePlus} iconClass="bg-violet-500/10 text-violet-600" />
   307→        <KpiCard label="Sent Today" value={sentToday} icon={MailCheck} iconClass="bg-teal-500/10 text-teal-600" />
   308→        <KpiCard label="Pending Approval" value={pendingApproval} icon={Clock} iconClass="bg-amber-500/10 text-amber-600" />
   309→        <KpiCard label="Failed Emails" value={failedEmails} icon={MailX} iconClass="bg-rose-500/10 text-rose-600" />
   310→      </div>
   311→
   312→      <Card className="border-border/60 shadow-soft overflow-hidden">
   313→        <div className="flex items-center justify-between px-5 py-3 border-b border-border bg-muted/30">
   314→          <div className="font-medium text-sm">Recent Documents</div>
   315→        </div>
   316→        <DocumentsTable
   317→          docs={docs}
   318→          loading={isLoading}
   319→          onPreview={onPreview}
   320→          onSendEmail={onSendEmail}
   321→        />
   322→      </Card>
   323→    </div>
   324→  );
   325→}

   331→function TemplatesTab({
   332→  onEdit,
   333→  onCreate,
   334→  onCompare,
   335→}: {
   336→  onEdit: (id: string) => void;

   338→  onCompare: (id1?: string, id2?: string) => void;
   339→}) {
   340→  const qc = useQueryClient();
   341→  const [search, setSearch] = useState("");
   342→  const [type, setType] = useState("");
   343→  const [previewTpl, setPreviewTpl] = useState<any | null>(null);
   344→
   345→  const { data, isLoading } = useQuery({
   346→    queryKey: ["document-templates", "templates-tab", search, type],
   347→    queryFn: async () => {
   348→      const params = new URLSearchParams();
   349→      if (search) params.set("search", search);
   350→      if (type) params.set("type", type);
   351→      const r = await fetch(`/api/document-templates?${params.toString()}`);
   352→      return r.json();
   353→    },
   354→  });
   355→  const templates = data?.items ?? [];
   356→
   357→  async function archiveTemplate(id: string) {
   358→    if (!confirm("Archive this template? It will be hidden from the active list.")) return;
   359→    try {
   360→      const res = await fetch(`/api/document-templates/${id}`, {
   361→        method: "DELETE",
   362→      });
   363→      if (!res.ok) throw new Error("Failed to archive");
   364→      toast.success("Template archived.");
   365→      qc.invalidateQueries({ queryKey: ["document-templates"] });
   366→    } catch (err: any) {
   367→      toast.error(err.message);
   368→    }
   369→  }
   370→
   371→  async function duplicateTemplate(t: any) {
   372→    try {
   373→      const res = await fetch("/api/document-templates", {
   374→        method: "POST",
   375→        headers: { "Content-Type": "application/json" },
   376→        body: JSON.stringify({
   377→          ...t,
   378→          name: `${t.name} (Copy)`,
   379→          code: `${t.code}-COPY`,
   380→          status: "DRAFT",
   381→        }),
   382→      });
   383→      if (!res.ok) {
   384→        const err = await res.json().catch(() => ({}));
   385→        throw new Error(err.error || "Failed");
   386→      }
   387→      toast.success("Template duplicated.");
   388→      qc.invalidateQueries({ queryKey: ["document-templates"] });
   389→    } catch (err: any) {
   390→      toast.error(err.message);
   391→    }
   392→  }
   393→
   394→  return (
   395→    <div className="space-y-4">
   396→      <div className="flex flex-col md:flex-row gap-3">
   397→        <div className="relative flex-1">
   398→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   399→          <Input
   400→            placeholder="Search templates by name, code, description…"
   401→            value={search}
   402→            onChange={(e) => setSearch(e.target.value)}
   403→            className="pl-9"
   404→          />
   405→        </div>
   406→        <Select
   407→          value={type || "ALL"}
   408→          onValueChange={(v) => setType(v === "ALL" ? "" : v)}
   409→        >
   410→          <SelectTrigger className="md:w-56">
   411→            <SelectValue placeholder="All types" />
   412→          </SelectTrigger>
   413→          <SelectContent>
   414→            <SelectItem value="ALL">All types</SelectItem>
   415→            {DOC_TYPES.map((t) => (
   416→              <SelectItem key={t} value={t}>
   417→                {t.replace(/_/g, " ")}
   418→              </SelectItem>
   419→            ))}
   420→          </SelectContent>
   421→        </Select>
   422→        <Button
   423→          variant="outline"
   424→          onClick={() => onCompare(undefined, undefined)}
   425→          size="sm"
   426→          className="md:w-auto"
   427→        >
   428→          <GitCompareArrows className="size-4 mr-1.5" /> Compare
   429→        </Button>
   430→        <Button onClick={onCreate} size="sm" className="md:w-auto">
   431→          <Plus className="size-4 mr-1.5" /> Create Template
   432→        </Button>
   433→      </div>
   434→
   435→      {isLoading && (
   436→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   437→          {Array.from({ length: 6 }).map((_, i) => (
   438→            <div key={i} className="h-44 rounded-xl bg-muted/40 animate-pulse" />
   439→          ))}
   440→        </div>
   441→      )}
   442→
   443→      {!isLoading && templates.length === 0 && (
   444→        <EmptyState
   445→          icon={FileText}
   446→          title="No templates found"
   447→          description="Create your first template to start generating HR documents."
   448→          actionLabel="Create Template"
   449→          onAction={onCreate}
   450→        />
   451→      )}
   452→
   453→      {!isLoading && templates.length > 0 && (
   454→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   455→          {templates.map((t: any) => (
   456→            <Card
   457→              key={t.id}
   458→              className="border-border/60 shadow-soft hover:shadow-card-hover hover:-translate-y-0.5 transition-all overflow-hidden flex flex-col"
   459→            >
   460→              <CardContent className="p-5 flex-1 flex flex-col">
   461→                <div className="flex items-start justify-between gap-2 mb-3">
   462→                  <div className="min-w-0 flex-1">
   463→                    <div className="font-semibold truncate">{t.name}</div>

   568→function GeneratedTab({
   569→  onPreview,
   570→  onSendEmail,
   571→}: {
   572→  onPreview: (doc: any) => void;
   573→  onSendEmail: (doc: any) => void;
   574→}) {
   575→  const qc = useQueryClient();
   576→  const [search, setSearch] = useState("");
   577→  const [type, setType] = useState("");
   578→  const [status, setStatus] = useState("");
   579→  const [page, setPage] = useState(1);
   580→
   581→  const { data, isLoading } = useQuery({
   582→    queryKey: ["documents", "generated-tab", search, type, status, page],
   583→    queryFn: async () => {
   584→      const params = new URLSearchParams();
   585→      if (search) params.set("search", search);
   586→      if (type) params.set("type", type);
   587→      if (status) params.set("status", status);
   588→      params.set("page", String(page));
   589→      params.set("pageSize", "20");
   590→      const r = await fetch(`/api/documents?${params.toString()}`);
   591→      return r.json();
   592→    },
   593→    placeholderData: (prev) => prev,
   594→  });
   595→  const docs = data?.items ?? [];
   596→  const total = data?.total ?? 0;
   597→  const totalPages = data?.totalPages ?? 1;
   598→
   599→  async function archiveDoc(id: string) {

   925→// Status pipeline shown as small horizontal pills.
   926→const STATUS_PIPELINE: { key: string; label: string }[] = [
   927→  { key: "DRAFT", label: "Draft" },
   928→  { key: "GENERATED", label: "Generated" },
   929→  { key: "PENDING_APPROVAL", label: "Pending" },
   930→  { key: "APPROVED", label: "Approved" },
   931→  { key: "ISSUED", label: "Issued" },
   932→  { key: "SENT", label: "Sent" },
   933→];
   934→
   935→function StatusFlowPills({ status }: { status: string }) {
   936→  const currentIdx = STATUS_PIPELINE.findIndex((s) => s.key === status);
   937→  if (status === "ARCHIVED") {
   938→    return <StatusBadge status="ARCHIVED" />;
   939→  }
   940→  if (currentIdx === -1) {
   941→    return <StatusBadge status={status} />;
   942→  }
   943→  return (
   944→    <div className="flex items-center gap-0.5 flex-wrap">
   945→      {STATUS_PIPELINE.map((s, idx) => {
   946→        const isCurrent = idx === currentIdx;
   947→        const isPast = idx < currentIdx;
   948→        return (
   949→          <div key={s.key} className="flex items-center">
   950→            <span
   951→              className={cn(
   952→                "inline-flex items-center px-1.5 h-5 rounded text-[9px] font-medium leading-none",
   953→                isCurrent &&
   954→                  "bg-primary text-primary-foreground shadow-soft",
   955→                isPast && "bg-primary/15 text-primary",
   956→                !isCurrent && !isPast &&
   957→                  "bg-muted text-muted-foreground"
   958→              )}
   959→              title={s.label}
   960→            >
   961→              {s.label}
   962→            </span>
   963→            {idx < STATUS_PIPELINE.length - 1 && (
   964→              <ChevronRight
   965→                className={cn(
   966→                  "size-2.5 mx-0.5 flex-shrink-0",
   967→                  idx < currentIdx ? "text-primary" : "text-muted-foreground/50"
   968→                )}
   969→              />
   970→            )}
   971→          </div>
   972→        );
   973→      })}

  1318→  const [loading, setLoading] = useState(false);
  1319→

  1321→    if (!doc) {
  1322→      // Defer the reset to avoid synchronous setState in effect body.
  1323→      const t = setTimeout(() => {
  1324→        setPreview(null);

  1345→      cancelled = true;

  1493→function DirectSendEmailDialog({
  1494→  doc,
  1495→  onClose,
  1496→}: {
  1497→  doc: any | null;
  1498→  onClose: () => void;
  1499→}) {
  1500→  const qc = useQueryClient();
  1501→  const [sending, setSending] = useState(false);
  1502→  const [to, setTo] = useState("");
  1503→  const [cc, setCc] = useState("");
  1504→  const [bcc, setBcc] = useState("");
  1505→  const [subject, setSubject] = useState("");
  1506→  const [body, setBody] = useState("");
  1507→
  1508→  useEffect(() => {
  1509→    if (!doc) return;
  1510→    // Fetch full document + template to get email subject/body presets.
  1511→    fetch(`/api/documents/${doc.id}`)
  1512→      .then((r) => r.json())
  1513→      .then((d) => {
  1514→        setTo(d.employee?.officialEmail ?? "");
  1515→        setSubject(
  1516→          d.template?.emailSubject
  1517→            ? d.template.emailSubject.replace(
  1518→                /\{\{company\.name\}\}/g,
  1519→                "the company"
  1520→              )
  1521→            : `Your document - ${d.documentNumber}`
  1522→        );
  1523→        setBody(
  1524→          d.template?.emailBody ||
  1525→            `Dear ${d.employee?.fullName ?? ""},\n\nPlease find attached your document ${d.documentNumber}.\n\nRegards,\nHR Team`
  1526→        );
  1527→      })
  1528→      .catch(() => {
  1529→        setTo("");
  1530→        setSubject(`Your document - ${doc.documentNumber}`);
  1531→        setBody("");
  1532→      });
  1533→  }, [doc]);
  1534→
  1535→  async function handleSend() {
  1536→    if (!doc) return;
  1537→    if (!to || !subject || !body) {
  1538→      toast.error("To, subject and body are required.");
  1539→      return;
  1540→    }
  1541→    setSending(true);
  1542→    try {
  1543→      const res = await fetch(`/api/documents/${doc.id}/send-email`, {
  1544→        method: "POST",
  1545→        headers: { "Content-Type": "application/json" },
  1546→        body: JSON.stringify({ to, cc, bcc, subject, body }),
  1547→      });
  1548→      if (!res.ok) {
  1549→        const err = await res.json().catch(() => ({}));
  1550→        throw new Error(err.error || "Send failed");
  1551→      }
  1552→      toast.success(`Email sent to ${to}`);
  1553→      qc.invalidateQueries({ queryKey: ["documents"] });
  1554→      qc.invalidateQueries({ queryKey: ["email-logs"] });
  1555→      qc.invalidateQueries({ queryKey: ["dashboard"] });
  1556→      onClose();
  1557→    } catch (err: any) {
  1558→      toast.error(err.message);
  1559→    } finally {
  1560→      setSending(false);
  1561→    }
  1562→  }
  1563→
  1564→  return (
  1565→    <Dialog open={!!doc} onOpenChange={(o) => !o && onClose()}>
  1566→      <DialogContent className="max-w-[95vw] sm:max-w-2xl">
  1567→        <DialogHeader>
  1568→          <DialogTitle className="flex items-center gap-2">
  1569→            <Mail className="size-5 text-primary" />
  1570→            Send Document via Email
  1571→          </DialogTitle>
  1572→        </DialogHeader>
  1573→        <div className="space-y-3">
  1574→          <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
  1575→            <div className="md:col-span-2">
  1576→              <label className="text-xs font-medium text-muted-foreground mb-1.5 block">
  1577→                To *
  1578→              </label>
  1579→              <Input
  1580→                type="email"
  1581→                value={to}
  1582→                onChange={(e) => setTo(e.target.value)}
  1583→              />
  1584→            </div>
  1585→            <div>
  1586→              <label className="text-xs font-medium text-muted-foreground mb-1.5 block">
  1587→                CC

  1592→              <label className="text-xs font-medium text-muted-foreground mb-1.5 block">
  1593→                BCC

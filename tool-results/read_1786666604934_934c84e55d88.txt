   135→function OrganizationTab() {
   136→  const qc = useQueryClient();
   137→  const [saving, setSaving] = useState(false);
   138→  const [form, setForm] = useState<any>({
   139→    name: "",
   140→    legalName: "",
   141→    address: "",
   142→    city: "",
   143→    state: "",
   144→    country: "",
   145→    zipCode: "",
   146→    email: "",
   147→    phone: "",
   148→    website: "",
   149→    logo: "",
   150→    taxId: "",
   151→  });
   152→
   153→  const { data, isLoading } = useQuery({
   154→    queryKey: ["company"],
   155→    queryFn: () => fetch("/api/company").then((r) => r.json()),
   156→  });
   157→
   158→  useEffect(() => {
   159→    if (data) {
   160→      setForm({
   161→        name: data.name ?? "",
   162→        legalName: data.legalName ?? "",
   163→        address: data.address ?? "",
   164→        city: data.city ?? "",
   165→        state: data.state ?? "",
   166→        country: data.country ?? "",
   167→        zipCode: data.zipCode ?? "",
   168→        email: data.email ?? "",
   169→        phone: data.phone ?? "",
   170→        website: data.website ?? "",
   171→        logo: data.logo ?? "",
   172→        taxId: data.taxId ?? "",
   173→      });
   174→    }
   175→  }, [data]);
   176→
   177→  async function save() {
   178→    if (!form.name) {
   179→      toast.error("Company name is required");
   180→      return;
   181→    }
   182→    setSaving(true);
   183→    try {
   184→      const res = await fetch("/api/company", {
   185→        method: "PATCH",
   186→        headers: { "Content-Type": "application/json" },
   187→        body: JSON.stringify(form),
   188→      });
   189→      if (!res.ok) throw new Error();
   190→      toast.success("Company profile saved");
   191→      qc.invalidateQueries({ queryKey: ["company"] });
   192→    } catch {
   193→      toast.error("Failed to save company");
   194→    } finally {
   195→      setSaving(false);
   196→    }
   197→  }
   198→
   199→  if (isLoading) {
   200→    return (
   201→      <Card className="border-border/60 shadow-soft">
   202→        <CardContent className="p-6">
   203→          <div className="h-96 rounded-lg bg-muted/40 animate-pulse" />
   204→        </CardContent>
   205→      </Card>
   206→    );
   207→  }
   208→
   209→  const fieldGroups = [
   210→    {
   211→      title: "Legal Details",
   212→      icon: Building2,
   213→      fields: [
   214→        { key: "name", label: "Company Name *", full: true },
   215→        { key: "legalName", label: "Legal Name" },
   216→        { key: "taxId", label: "Tax ID / TIN" },
   217→      ],
   218→    },
   219→    {
   220→      title: "Contact Information",
   221→      icon: Mail,
   222→      fields: [
   223→        { key: "email", label: "Email" },
   224→        { key: "phone", label: "Phone" },
   225→        { key: "website", label: "Website" },
   226→        { key: "logo", label: "Logo URL", full: true },
   227→      ],
   228→    },
   229→    {
   230→      title: "Location",
   231→      icon: MapPin,
   232→      fields: [
   233→        { key: "address", label: "Address", full: true, textarea: true },
   234→        { key: "city", label: "City" },
   235→        { key: "state", label: "State / Province" },
   236→        { key: "country", label: "Country" },
   237→        { key: "zipCode", label: "ZIP / Postal Code" },
   238→      ],
   239→    },
   240→  ];
   241→
   242→  return (
   243→    <div className="space-y-4">
   244→      <Card className="border-border/60 shadow-soft">
   245→        <CardContent className="p-6 space-y-5">
   246→          {/* Header with logo preview */}
   247→          <div className="flex items-start justify-between pb-4 border-b border-border/60 gap-4">
   248→            <div>
   249→              <div className="font-semibold text-base">Organization Profile</div>
   250→              <p className="text-xs text-muted-foreground mt-0.5">
   251→                These details appear on documents, emails, and reports.
   252→              </p>
   253→            </div>
   254→            <div className="flex items-center gap-3">
   255→              <div className="size-14 rounded-xl bg-primary/10 border border-border/60 flex items-center justify-center overflow-hidden flex-shrink-0">
   256→                {form.logo ? (
   257→                  <img
   258→                    src={form.logo}
   259→                    alt="Logo"
   260→                    className="size-full object-contain p-1"
   261→                    onError={(e) => {
   262→                      (e.target as HTMLImageElement).style.display = "none";
   263→                    }}
   264→                  />
   265→                ) : (
   266→                  <Building2 className="size-6 text-primary" />
   267→                )}
   268→              </div>
   269→            </div>
   270→          </div>
   271→
   272→          {/* Grouped fields */}
   273→          {fieldGroups.map((group) => {
   274→            const GroupIcon = group.icon;
   275→            return (
   276→              <div key={group.title} className="space-y-3">
   277→                <div className="flex items-center gap-2 text-sm font-semibold text-foreground">
   278→                  <div className="size-7 rounded-lg bg-primary/10 text-primary flex items-center justify-center">
   279→                    <GroupIcon className="size-4" />
   280→                  </div>
   281→                  {group.title}
   282→                </div>
   283→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 pl-9">
   284→                  {group.fields.map((f) => (
   285→                    <div
   286→                      key={f.key}
   287→                      className={cn("space-y-1.5", f.full && "sm:col-span-2")}
   288→                    >
   289→                      <Label className="text-xs text-muted-foreground">
   290→                        {f.label}
   291→                      </Label>
   292→                      {f.textarea ? (
   293→                        <Textarea
   294→                          rows={2}
   295→                          value={form[f.key]}
   296→                          onChange={(e) =>
   297→                            setForm({ ...form, [f.key]: e.target.value })
   298→                          }
   299→                        />
   300→                      ) : (
   301→                        <Input
   302→                          value={form[f.key]}
   303→                          onChange={(e) =>
   304→                            setForm({ ...form, [f.key]: e.target.value })
   305→                          }
   306→                        />
   307→                      )}
   308→                    </div>
   309→                  ))}
   310→                </div>
   311→              </div>
   312→            );
   313→          })}
   314→        </CardContent>
   315→      </Card>
   316→
   317→      {/* Sticky save bar */}
   318→      <div className="sticky bottom-4 z-10 flex justify-end">
   319→        <div className="inline-flex items-center gap-3 rounded-xl border border-border/60 bg-card/95 backdrop-blur-md shadow-card-hover px-4 py-2.5">
   320→          <span className="text-xs text-muted-foreground hidden sm:inline">
   321→            Changes apply to all generated documents
   322→          </span>
   323→          <Button onClick={save} disabled={saving} size="sm">
   324→            {saving ? (
   325→              "Saving…"
   326→            ) : (
   327→              <>
   328→                <Save className="size-4 mr-1.5" /> Update Profile
   329→              </>
   330→            )}
   331→          </Button>
   332→        </div>
   333→      </div>
   334→    </div>
   335→  );
   336→}

  1298→  type?: "text" | "number" | "color" | "switch" | "select";

  1303→function SimpleFormDialog({
  1304→  title,
  1305→  open,
  1306→  onOpenChange,
  1307→  fields,
  1308→  initial,
  1309→  onSubmit,
  1310→}: {
  1311→  title: string;
  1312→  open: boolean;
  1313→  onOpenChange: (v: boolean) => void;
  1314→  fields: Field[];
  1315→  initial: any;
  1316→  onSubmit: (form: any) => Promise<void>;
  1317→}) {
  1318→  const [form, setForm] = useState<any>(initial);
  1319→  const [saving, setSaving] = useState(false);
  1320→
  1321→  useEffect(() => {
  1322→    setForm(initial);
  1323→  }, [initial]);
  1324→
  1325→  async function submit() {
  1326→    for (const f of fields) {
  1327→      if (f.required && !form[f.key]) {
  1328→        toast.error(`${f.label} is required`);
  1329→        return;
  1330→      }
  1331→    }
  1332→    setSaving(true);
  1333→    try {
  1334→      await onSubmit(form);
  1335→      toast.success("Saved");
  1336→      onOpenChange(false);
  1337→    } catch {
  1338→      toast.error("Failed to save");
  1339→    } finally {
  1340→      setSaving(false);
  1341→    }
  1342→  }
  1343→
  1344→  return (
  1345→    <Dialog open={open} onOpenChange={onOpenChange}>
  1346→      <DialogContent className="max-w-[95vw] sm:max-w-lg">
  1347→        <DialogHeader>
  1348→          <DialogTitle>{title}</DialogTitle>
  1349→        </DialogHeader>
  1350→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
  1351→          {fields.map((f) => (
  1352→            <div
  1353→              key={f.key}
  1354→              className={cn("space-y-1.5", f.textarea && "sm:col-span-2")}
  1355→            >
  1356→              <Label>{f.label}</Label>
  1357→              {f.textarea ? (
  1358→                <Textarea
  1359→                  rows={2}
  1360→                  value={form[f.key] ?? ""}
  1361→                  onChange={(e) =>
  1362→                    setForm({ ...form, [f.key]: e.target.value })
  1363→                  }
  1364→                />
  1365→              ) : f.type === "select" ? (
  1366→                <Select
  1367→                  value={form[f.key] || (f.allowNone ? "NONE" : "")}
  1368→                  onValueChange={(v) =>
  1369→                    setForm({
  1370→                      ...form,
  1371→                      [f.key]: f.allowNone && v === "NONE" ? "" : v,
  1372→                    })
  1373→                  }
  1374→                >
  1375→                  <SelectTrigger>
  1376→                    <SelectValue placeholder="Select…" />
  1377→                  </SelectTrigger>
  1378→                  <SelectContent>
  1379→                    {f.allowNone && <SelectItem value="NONE">None</SelectItem>}
  1380→                    {f.options?.map((o) => (
  1381→                      <SelectItem key={o.value} value={o.value}>
  1382→                        {o.label}
  1383→                      </SelectItem>
  1384→                    ))}
  1385→                  </SelectContent>
  1386→                </Select>
  1387→              ) : f.type === "switch" ? (
  1388→                <div className="flex items-center gap-2 pt-1">
  1389→                  <Switch
  1390→                    checked={!!form[f.key]}
  1391→                    onCheckedChange={(v) => setForm({ ...form, [f.key]: v })}
  1392→                  />
  1393→                  <span className="text-sm text-muted-foreground">
  1394→                    {form[f.key] ? "Enabled" : "Disabled"}
  1395→                  </span>
  1396→                </div>
  1397→              ) : f.type === "color" ? (
  1398→                <div className="flex items-center gap-2">
  1399→                  <input
  1400→                    type="color"
  1401→                    value={form[f.key] ?? "#10b981"}
  1402→                    onChange={(e) =>
  1403→                      setForm({ ...form, [f.key]: e.target.value })
  1404→                    }
  1405→                    className="size-9 rounded-md border border-border cursor-pointer"
  1406→                  />
  1407→                  <Input
  1408→                    value={form[f.key] ?? ""}
  1409→                    onChange={(e) =>
  1410→                      setForm({ ...form, [f.key]: e.target.value })
  1411→                    }
  1412→                    className="font-mono text-xs"
  1413→                  />
  1414→                </div>
  1415→              ) : (
  1416→                <Input
  1417→                  type={f.type === "number" ? "number" : "text"}
  1418→                  value={form[f.key] ?? ""}
  1419→                  onChange={(e) =>
  1420→                    setForm({
  1421→                      ...form,
  1422→                      [f.key]:
  1423→                        f.type === "number"
  1424→                          ? Number(e.target.value)
  1425→                          : e.target.value,
  1426→                    })
  1427→                  }
  1428→                />
  1429→              )}
  1430→            </div>
  1431→          ))}
  1432→        </div>
  1433→        <DialogFooter>
  1434→          <Button variant="outline" onClick={() => onOpenChange(false)}>
  1435→            Cancel
  1436→          </Button>
  1437→          <Button onClick={submit} disabled={saving}>
  1438→            {saving ? "Saving…" : "Save"}
  1439→          </Button>
  1440→        </DialogFooter>
  1441→      </DialogContent>
  1442→    </Dialog>
  1443→  );
  1444→}

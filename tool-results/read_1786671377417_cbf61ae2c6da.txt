   166→   166→export function AssetsModule() {
   167→   167→  const [search, setSearch] = useState("");
   168→   168→  const [type, setType] = useState("");
   169→   169→  const [status, setStatus] = useState("");
   170→   170→  const [view, setView] = useState<"table" | "grid">("table");
   171→   171→
   172→   172→  const [formOpen, setFormOpen] = useState(false);
   173→   173→  const [editAsset, setEditAsset] = useState<Asset | null>(null);
   174→   174→  const [assignAsset, setAssignAsset] = useState<Asset | null>(null);
   175→   175→  const [returnAsset, setReturnAsset] = useState<Asset | null>(null);
   176→   176→
   177→   177→  const qc = useQueryClient();
   178→   178→  const { data, isLoading, isError } = useQuery({
   179→   179→    queryKey: ["assets", search, type, status],
   180→   180→    queryFn: async () => {
   181→   181→      const params = new URLSearchParams();
   182→   182→      if (search) params.set("search", search);
   183→   183→      if (type) params.set("type", type);
   184→   184→      if (status) params.set("status", status);
   185→   185→      const r = await fetch(`/api/assets?${params.toString()}`);
   186→   186→      if (!r.ok) throw new Error("Failed to load assets");
   187→   187→      return r.json();
   188→   188→    },
   189→   189→  });
   190→   190→
   191→   191→  const assets: Asset[] = data?.items ?? [];
   192→   192→  const total = assets.length;
   193→   193→  const assigned = assets.filter((a) => a.status === "ASSIGNED").length;
   194→   194→  const available = assets.filter((a) => a.status === "AVAILABLE").length;
   195→   195→  const damaged = assets.filter((a) => a.condition === "DAMAGED").length;
   196→   196→
   197→   197→  function openCreate() {
   198→   198→    setEditAsset(null);
   199→   199→    setFormOpen(true);
   200→   200→  }
   201→   201→  function openEdit(a: Asset) {
   202→   202→    setEditAsset(a);
   203→   203→    setFormOpen(true);
   204→   204→  }
   205→   205→  async function deleteAsset(a: Asset) {
   206→   206→    if (!confirm(`Permanently delete asset "${a.name}"?`)) return;
   207→   207→    try {
   208→   208→      const r = await fetch(`/api/assets/${a.id}`, { method: "DELETE" });
   209→   209→      if (!r.ok) throw new Error("Failed to delete asset");
   210→   210→      toast.success(`Asset "${a.name}" deleted.`);
   211→   211→      qc.invalidateQueries({ queryKey: ["assets"] });
   212→   212→    } catch (err: any) {
   213→   213→      toast.error(err?.message || "Failed to delete asset.");
   214→   214→    }
   215→   215→  }
   216→   216→  async function retireAsset(a: Asset) {
   217→   217→    if (!confirm(`Retire asset "${a.name}"? It will be marked RETIRED.`))
   218→   218→      return;
   219→   219→    try {
   220→   220→      const r = await fetch(`/api/assets/${a.id}`, {
   221→   221→        method: "PATCH",
   222→   222→        headers: { "Content-Type": "application/json" },
   223→   223→        body: JSON.stringify({ status: "RETIRED" }),
   224→   224→      });
   225→   225→      if (!r.ok) throw new Error("Failed to retire asset");
   226→   226→      toast.success(`Asset "${a.name}" retired.`);
   227→   227→      qc.invalidateQueries({ queryKey: ["assets"] });
   228→   228→    } catch (err: any) {
   229→   229→      toast.error(err?.message || "Failed to retire asset.");
   230→   230→    }
   231→   231→  }
   232→   232→
   233→   233→  return (
   234→   234→    <div className="space-y-6">
   235→   235→      <PageHeader

   238→   238→        icon={<Package className="size-5" />}
   239→   239→        actions={
   240→   240→          <>
   241→   241→            <ExportButton
   242→   242→              module="assets"
   243→   243→              filters={{ search, status, type }}
   244→   244→            />
   245→   245→            <div className="flex rounded-md border border-border overflow-hidden">
   246→   246→              <Button
   247→   247→                variant={view === "table" ? "default" : "ghost"}
   248→   248→                size="sm"
   249→   249→                className="rounded-none gap-1.5"
   250→   250→                onClick={() => setView("table")}
   251→   251→                aria-label="Table view"
   252→   252→              >
   253→   253→                <List className="size-4" />
   254→   254→                <span className="hidden sm:inline">Table</span>
   255→   255→              </Button>
   256→   256→              <Button
   257→   257→                variant={view === "grid" ? "default" : "ghost"}
   258→   258→                size="sm"
   259→   259→                className="rounded-none gap-1.5"
   260→   260→                onClick={() => setView("grid")}
   261→   261→                aria-label="Grid view"
   262→   262→              >
   263→   263→                <LayoutGrid className="size-4" />
   264→   264→                <span className="hidden sm:inline">Grid</span>
   265→   265→              </Button>
   266→   266→            </div>
   267→   267→            <Button size="sm" onClick={openCreate} className="gap-1.5">
   268→   268→              <Plus className="size-4" />
   269→   269→              <span className="hidden sm:inline">Add Asset</span>
   270→   270→              <span className="sm:hidden">New</span>
   271→   271→            </Button>
   272→   272→          </>
   273→   273→        }
   274→   274→      />
   275→   275→
   276→   276→      {/* KPI cards */}
   277→   277→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3 md:gap-4">
   278→   278→        <KpiCard
   279→   279→          label="Total Assets"
   280→   280→          value={total}
   281→   281→          icon={Package}
   282→   282→          iconClass="bg-primary/10 text-primary"
   283→   283→          footer={<span className="text-muted-foreground">All equipment</span>}
   284→   284→        />
   285→   285→        <KpiCard
   286→   286→          label="Assigned"
   287→   287→          value={assigned}
   288→   288→          icon={PackageCheck}
   289→   289→          iconClass="bg-sky-500/15 text-sky-600"
   290→   290→          footer={<span className="text-muted-foreground">In use</span>}
   291→   291→        />
   292→   292→        <KpiCard
   293→   293→          label="Available"
   294→   294→          value={available}
   295→   295→          icon={PackageOpen}
   296→   296→          iconClass="bg-emerald-500/15 text-emerald-600"
   297→   297→          footer={<span className="text-muted-foreground">Free to assign</span>}
   298→   298→        />
   299→   299→        <KpiCard
   300→   300→          label="Damaged"
   301→   301→          value={damaged}
   302→   302→          icon={AlertTriangle}
   303→   303→          iconClass="bg-rose-500/15 text-rose-600"
   304→   304→          footer={<span className="text-muted-foreground">Needs repair</span>}
   305→   305→        />
   306→   306→      </div>
   307→   307→
   308→   308→      {/* Filter bar */}
   309→   309→      <div className="flex flex-col md:flex-row gap-3">
   310→   310→        <div className="relative flex-1">
   311→   311→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   312→   312→          <Input
   313→   313→            placeholder="Search by name, serial, or assignee…"
   314→   314→            value={search}

   316→   316→            className="pl-9"
   317→   317→          />
   318→   318→        </div>
   319→   319→        <Select

   322→   322→        >
   323→   323→          <SelectTrigger className="md:w-44">
   324→   324→            <SelectValue placeholder="All types" />
   325→   325→          </SelectTrigger>
   326→   326→          <SelectContent>
   327→   327→            <SelectItem value="ALL">All types</SelectItem>
   328→   328→            {ASSET_TYPES.map((t) => (
   329→   329→              <SelectItem key={t.value} value={t.value}>
   330→   330→                {t.label}
   331→   331→              </SelectItem>

   333→   333→          </SelectContent>
   334→   334→        </Select>
   335→   335→        <Select
   336→   336→          value={status || "ALL"}
   337→   337→          onValueChange={(v) => setStatus(v === "ALL" ? "" : v)}
   338→   338→        >
   339→   339→          <SelectTrigger className="md:w-44">
   340→   340→            <SelectValue placeholder="All statuses" />
   341→   341→          </SelectTrigger>
   342→   342→          <SelectContent>
   343→   343→            <SelectItem value="ALL">All statuses</SelectItem>
   344→   344→            {STATUSES.map((s) => (
   345→   345→              <SelectItem key={s} value={s}>
   346→   346→                {s.charAt(0) + s.slice(1).toLowerCase()}
   347→   347→              </SelectItem>
   348→   348→            ))}
   349→   349→          </SelectContent>
   350→   350→        </Select>
   351→   351→      </div>
   352→   352→
   353→   353→      {/* Result count */}
   354→   354→      <div className="flex items-center justify-between text-sm">
   355→   355→        <div className="text-muted-foreground">
   356→   356→          {isLoading ? (
   357→   357→            <Skeleton className="h-4 w-32" />
   358→   358→          ) : (
   359→   359→            <span>
   360→   360→              Showing <span className="font-semibold text-foreground">{assets.length}</span>{" "}
   361→   361→              asset{assets.length === 1 ? "" : "s"}
   362→   362→              {(type || status || search) && " (filtered)"}
   363→   363→            </span>
   364→   364→          )}
   365→   365→        </div>
   366→   366→      </div>
   367→   367→
   368→   368→      {/* Content */}
   369→   369→      {isLoading ? (
   370→   370→        <AssetsSkeleton view={view} />
   371→   371→      ) : isError ? (
   372→   372→        <EmptyState
   373→   373→          icon={AlertTriangle}
   374→   374→          title="Failed to load assets"
   375→   375→          description="Please try again. If the problem persists, check the dev server."
   376→   376→          actionLabel="Retry"
   377→   377→          onAction={() => qc.invalidateQueries({ queryKey: ["assets"] })}
   378→   378→        />
   379→   379→      ) : assets.length === 0 ? (
   380→   380→        <EmptyState
   381→   381→          icon={Package}
   382→   382→          title={search || type || status ? "No matching assets" : "No assets yet"}
   383→   383→          description={
   384→   384→            search || type || status
   385→   385→              ? "Try adjusting your filters."
   386→   386→              : "Start by adding your first company asset."
   387→   387→          }
   388→   388→          actionLabel="Add Asset"
   389→   389→          onAction={openCreate}
   390→   390→        />
   391→   391→      ) : view === "table" ? (
   392→   392→        <AssetsTable
   393→   393→          assets={assets}
   394→   394→          onEdit={openEdit}
   395→   395→          onDelete={deleteAsset}
   396→   396→          onAssign={(a) => setAssignAsset(a)}
   397→   397→          onReturn={(a) => setReturnAsset(a)}
   398→   398→          onRetire={retireAsset}
   399→   399→        />
   400→   400→      ) : (
   401→   401→        <AssetsGrid
   402→   402→          assets={assets}
   403→   403→          onEdit={openEdit}
   404→   404→          onDelete={deleteAsset}
   405→   405→          onAssign={(a) => setAssignAsset(a)}
   406→   406→          onReturn={(a) => setReturnAsset(a)}
   407→   407→          onRetire={retireAsset}
   408→   408→        />
   409→   409→      )}
   410→   410→
   411→   411→      {/* Create / Edit dialog */}
   412→   412→      <AssetFormDialog
   413→   413→        open={formOpen}
   414→   414→        onOpenChange={(o) => {
   415→   415→          setFormOpen(o);
   416→   416→          if (!o) setEditAsset(null);
   417→   417→        }}
   418→   418→        asset={editAsset}
   419→   419→        onSaved={() => qc.invalidateQueries({ queryKey: ["assets"] })}
   420→   420→      />
   421→   421→
   422→   422→      {/* Assign dialog */}
   423→   423→      <AssignDialog
   424→   424→        asset={assignAsset}
   425→   425→        onClose={() => setAssignAsset(null)}
   426→   426→        onSaved={() => {
   427→   427→          qc.invalidateQueries({ queryKey: ["assets"] });
   428→   428→          setAssignAsset(null);
   429→   429→        }}
   430→   430→      />
   431→   431→
   432→   432→      {/* Return dialog */}
   433→   433→      <ReturnDialog
   434→   434→        asset={returnAsset}
   435→   435→        onClose={() => setReturnAsset(null)}
   436→   436→        onSaved={() => {
   437→   437→          qc.invalidateQueries({ queryKey: ["assets"] });
   438→   438→          setReturnAsset(null);
   439→   439→        }}
   440→   440→      />
   441→   441→    </div>
   442→   442→  );
   443→   443→}

   858→   858→function AssetFormBody({
   859→   859→  asset,
   860→   860→  savingState,
   861→   861→  onSaved,
   862→   862→  onCancel,
   863→   863→}: {
   864→   864→  asset: Asset | null;
   865→   865→  savingState: [boolean, (b: boolean) => void];
   866→   866→  onSaved: () => void;
   867→   867→  onCancel: () => void;
   868→   868→}) {
   869→   869→  const [name, setName] = useState(asset?.name ?? "");
   870→   870→  const [type, setType] = useState<AssetType>(asset?.type ?? "LAPTOP");
   871→   871→  const [serialNumber, setSerialNumber] = useState(asset?.serialNumber ?? "");
   872→   872→  const [condition, setCondition] = useState<AssetCondition>(
   873→   873→    asset?.condition ?? "GOOD"
   874→   874→  );
   875→   875→  const [notes, setNotes] = useState(asset?.notes ?? "");
   876→   876→  const [saving, setSaving] = savingState;
   877→   877→
   878→   878→  async function submit() {
   879→   879→    if (!name.trim()) {
   880→   880→      toast.error("Asset name is required.");
   881→   881→      return;
   882→   882→    }
   883→   883→    setSaving(true);
   884→   884→    try {
   885→   885→      const body = { name, type, serialNumber, condition, notes };
   886→   886→      const r = asset
   887→   887→        ? await fetch(`/api/assets/${asset.id}`, {
   888→   888→            method: "PATCH",
   889→   889→            headers: { "Content-Type": "application/json" },
   890→   890→            body: JSON.stringify(body),
   891→   891→          })
   892→   892→        : await fetch("/api/assets", {
   893→   893→            method: "POST",
   894→   894→            headers: { "Content-Type": "application/json" },
   895→   895→            body: JSON.stringify(body),
   896→   896→          });
   897→   897→      if (!r.ok) {
   898→   898→        const e = await r.json().catch(() => ({}));
   899→   899→        throw new Error(e?.error || "Failed to save asset");
   900→   900→      }
   901→   901→      onSaved();
   902→   902→    } catch (err: any) {
   903→   903→      toast.error(err?.message || "Failed to save asset.");
   904→   904→    } finally {
   905→   905→      setSaving(false);
   906→   906→    }
   907→   907→  }

   910→   910→    <div className="space-y-4">

   921→   921→        <div className="space-y-1.5">

   924→   924→            <SelectTrigger id="asset-type">
   925→   925→              <SelectValue />
   926→   926→            </SelectTrigger>
   927→   927→            <SelectContent>
   928→   928→              {ASSET_TYPES.map((t) => (
   929→   929→                <SelectItem key={t.value} value={t.value}>
   930→   930→                  <span className="flex items-center gap-2">
   931→   931→                    <t.icon className="size-3.5" />

   934→   934→                </SelectItem>

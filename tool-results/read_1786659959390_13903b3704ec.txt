   201→
   202→function AllDocumentsTab({
   203→  onPreview,
   204→  onSendEmail,
   205→}: {
   206→  onPreview: (doc: any) => void;
   207→  onSendEmail: (doc: any) => void;
   208→}) {
   209→  const { data, isLoading } = useQuery({
   210→    queryKey: ["documents", "all-tab"],
   211→    queryFn: async () => {
   212→      const r = await fetch("/api/documents?pageSize=10");
   213→      return r.json();
   214→    },
   215→  });
   216→
   217→  const { data: emailStats } = useQuery({
   218→    queryKey: ["email-logs", "stats"],
   219→    queryFn: async () => {
   220→      const [sent, failed] = await Promise.all([
   221→        fetch("/api/email-logs?status=SENT&pageSize=1").then((r) => r.json()),
   222→        fetch("/api/email-logs?status=FAILED&pageSize=1").then((r) => r.json()),
   223→      ]);
   224→      return { sentToday: sent.total ?? 0, failed: failed.total ?? 0 };
   225→    },
   226→  });
   227→
   228→  const docs = data?.items ?? [];
   229→  const total = data?.total ?? 0;
   230→  const todayStr = new Date().toDateString();
   231→  const generatedToday = docs.filter(
   232→    (d: any) => new Date(d.createdAt).toDateString() === todayStr
   233→  ).length;
   234→  const sentToday = emailStats?.sentToday ?? 0;
   235→  const failedEmails = emailStats?.failed ?? 0;
   236→  const pendingApproval = docs.filter((d: any) =>
   237→    ["PENDING_APPROVAL", "GENERATED"].includes(d.status)
   238→  ).length;
   239→
   240→  return (
   241→    <div className="space-y-4">
   242→      <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-5 gap-3">
   243→        <KpiCard label="Total" value={total} icon={FileStack} iconClass="bg-primary/10 text-primary" />
   244→        <KpiCard label="Generated Today" value={generatedToday} icon={FilePlus} iconClass="bg-violet-500/10 text-violet-600" />
   245→        <KpiCard label="Sent Today" value={sentToday} icon={MailCheck} iconClass="bg-teal-500/10 text-teal-600" />
   246→        <KpiCard label="Pending Approval" value={pendingApproval} icon={Clock} iconClass="bg-amber-500/10 text-amber-600" />
   247→        <KpiCard label="Failed Emails" value={failedEmails} icon={MailX} iconClass="bg-rose-500/10 text-rose-600" />
   248→      </div>
   249→
   250→      <Card className="border-border/60 shadow-soft overflow-hidden">
   251→        <div className="flex items-center justify-between px-5 py-3 border-b border-border bg-muted/30">
   252→          <div className="font-medium text-sm">Recent Documents</div>
   253→        </div>
   254→        <DocumentsTable
   255→          docs={docs}
   256→          loading={isLoading}
   257→          onPreview={onPreview}
   258→          onSendEmail={onSendEmail}
   259→        />
   260→      </Card>
   261→    </div>
   262→  );
   263→}

   269→function TemplatesTab({
   270→  onEdit,
   271→  onCreate,
   272→}: {
   273→  onEdit: (id: string) => void;
   274→  onCreate: () => void;
   275→}) {
   276→  const qc = useQueryClient();
   277→  const [search, setSearch] = useState("");
   278→  const [type, setType] = useState("");
   279→  const [previewTpl, setPreviewTpl] = useState<any | null>(null);
   280→
   281→  const { data, isLoading } = useQuery({
   282→    queryKey: ["document-templates", "templates-tab", search, type],
   283→    queryFn: async () => {
   284→      const params = new URLSearchParams();
   285→      if (search) params.set("search", search);
   286→      if (type) params.set("type", type);
   287→      const r = await fetch(`/api/document-templates?${params.toString()}`);
   288→      return r.json();
   289→    },
   290→  });
   291→  const templates = data?.items ?? [];
   292→
   293→  async function archiveTemplate(id: string) {
   294→    if (!confirm("Archive this template? It will be hidden from the active list.")) return;
   295→    try {
   296→      const res = await fetch(`/api/document-templates/${id}`, {
   297→        method: "DELETE",
   298→      });
   299→      if (!res.ok) throw new Error("Failed to archive");
   300→      toast.success("Template archived.");
   301→      qc.invalidateQueries({ queryKey: ["document-templates"] });
   302→    } catch (err: any) {
   303→      toast.error(err.message);
   304→    }
   305→  }
   306→
   307→  async function duplicateTemplate(t: any) {
   308→    try {
   309→      const res = await fetch("/api/document-templates", {
   310→        method: "POST",
   311→        headers: { "Content-Type": "application/json" },
   312→        body: JSON.stringify({
   313→          ...t,
   314→          name: `${t.name} (Copy)`,
   315→          code: `${t.code}-COPY`,
   316→          status: "DRAFT",
   317→        }),
   318→      });
   319→      if (!res.ok) {
   320→        const err = await res.json().catch(() => ({}));
   321→        throw new Error(err.error || "Failed");
   322→      }
   323→      toast.success("Template duplicated.");
   324→      qc.invalidateQueries({ queryKey: ["document-templates"] });
   325→    } catch (err: any) {
   326→      toast.error(err.message);
   327→    }
   328→  }
   329→
   330→  return (
   331→    <div className="space-y-4">
   332→      <div className="flex flex-col md:flex-row gap-3">
   333→        <div className="relative flex-1">
   334→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   335→          <Input
   336→            placeholder="Search templates by name, code, description…"
   337→            value={search}
   338→            onChange={(e) => setSearch(e.target.value)}
   339→            className="pl-9"
   340→          />
   341→        </div>
   342→        <Select
   343→          value={type || "ALL"}
   344→          onValueChange={(v) => setType(v === "ALL" ? "" : v)}
   345→        >
   346→          <SelectTrigger className="md:w-56">
   347→            <SelectValue placeholder="All types" />
   348→          </SelectTrigger>
   349→          <SelectContent>
   350→            <SelectItem value="ALL">All types</SelectItem>
   351→            {DOC_TYPES.map((t) => (
   352→              <SelectItem key={t} value={t}>
   353→                {t.replace(/_/g, " ")}
   354→              </SelectItem>
   355→            ))}
   356→          </SelectContent>
   357→        </Select>
   358→        <Button onClick={onCreate} size="sm" className="md:w-auto">
   359→          <Plus className="size-4 mr-1.5" /> Create Template
   360→        </Button>
   361→      </div>
   362→
   363→      {isLoading && (
   364→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   365→          {Array.from({ length: 6 }).map((_, i) => (
   366→            <div key={i} className="h-44 rounded-xl bg-muted/40 animate-pulse" />
   367→          ))}
   368→        </div>
   369→      )}
   370→
   371→      {!isLoading && templates.length === 0 && (
   372→        <EmptyState
   373→          icon={FileText}
   374→          title="No templates found"
   375→          description="Create your first template to start generating HR documents."
   376→          actionLabel="Create Template"
   377→          onAction={onCreate}
   378→        />
   379→      )}
   380→
   381→      {!isLoading && templates.length > 0 && (
   382→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   383→          {templates.map((t: any) => (
   384→            <Card
   385→              key={t.id}
   386→              className="border-border/60 shadow-soft hover:shadow-card-hover hover:-translate-y-0.5 transition-all overflow-hidden flex flex-col"
   387→            >
   388→              <CardContent className="p-5 flex-1 flex flex-col">
   389→                <div className="flex items-start justify-between gap-2 mb-3">
   390→                  <div className="min-w-0 flex-1">
   391→                    <div className="font-semibold truncate">{t.name}</div>
   392→                    <div className="text-xs text-muted-foreground truncate">
   393→                      {t.description ?? "No description"}
   394→                    </div>
   395→                  </div>
   396→                  <Badge variant="outline" className="font-mono text-[10px]">
   397→                    {t.code}
   398→                  </Badge>
   399→                </div>
   400→
   401→                <div className="grid grid-cols-2 gap-2 text-xs mb-3">
   402→                  <div>
   403→                    <div className="text-muted-foreground">Type</div>
   404→                    <div className="font-medium">
   405→                      {t.type?.replace(/_/g, " ")}
   406→                    </div>
   407→                  </div>
   408→                  <div>
   409→                    <div className="text-muted-foreground">Category</div>
   410→                    <div className="font-medium">{t.category}</div>
   411→                  </div>

   422→                <div className="rounded-md bg-muted/40 p-2 text-[11px] text-muted-foreground mb-3 line-clamp-2 flex-1">

   493→function GeneratedTab({
   494→  onPreview,
   495→  onSendEmail,
   496→}: {
   497→  onPreview: (doc: any) => void;
   498→  onSendEmail: (doc: any) => void;
   499→}) {
   500→  const qc = useQueryClient();
   501→  const [search, setSearch] = useState("");
   502→  const [type, setType] = useState("");
   503→  const [status, setStatus] = useState("");
   504→  const [page, setPage] = useState(1);
   505→
   506→  const { data, isLoading } = useQuery({
   507→    queryKey: ["documents", "generated-tab", search, type, status, page],
   508→    queryFn: async () => {
   509→      const params = new URLSearchParams();
   510→      if (search) params.set("search", search);
   511→      if (type) params.set("type", type);
   512→      if (status) params.set("status", status);
   513→      params.set("page", String(page));
   514→      params.set("pageSize", "20");
   515→      const r = await fetch(`/api/documents?${params.toString()}`);
   516→      return r.json();
   517→    },
   518→    placeholderData: (prev) => prev,
   519→  });
   520→  const docs = data?.items ?? [];
   521→  const total = data?.total ?? 0;
   522→  const totalPages = data?.totalPages ?? 1;
   523→
   524→  async function archiveDoc(id: string) {
   525→    if (!confirm("Archive this document?")) return;
   526→    try {
   527→      await fetch(`/api/documents/${id}`, { method: "DELETE" });
   528→      toast.success("Document archived.");
   529→      qc.invalidateQueries({ queryKey: ["documents"] });
   530→    } catch {
   531→      toast.error("Archive failed.");
   532→    }
   533→  }
   534→
   535→  function download(id: string, format: "docx" | "pdf", docNumber: string) {
   536→    const url = `/api/documents/${id}/download?format=${format}`;
   537→    const a = document.createElement("a");
   538→    a.href = url;
   539→    a.download = `${docNumber}.${format}`;
   540→    document.body.appendChild(a);
   541→    a.click();
   542→    document.body.removeChild(a);
   543→    toast.success(`Downloading ${format.toUpperCase()}…`);
   544→  }
   545→
   546→  return (
   547→    <div className="space-y-4">
   548→      <div className="flex flex-col md:flex-row gap-3">
   549→        <div className="relative flex-1">
   550→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   551→          <Input
   552→            placeholder="Search by document number or title…"
   553→            value={search}
   554→            onChange={(e) => {
   555→              setSearch(e.target.value);
   556→              setPage(1);
   557→            }}
   558→            className="pl-9"
   559→          />
   560→        </div>
   561→        <Select
   562→          value={type || "ALL"}
   563→          onValueChange={(v) => {
   564→            setType(v === "ALL" ? "" : v);
   565→            setPage(1);
   566→          }}
   567→        >
   568→          <SelectTrigger className="md:w-48">
   569→            <SelectValue placeholder="All types" />
   570→          </SelectTrigger>
   571→          <SelectContent>
   572→            <SelectItem value="ALL">All types</SelectItem>
   573→            {DOC_TYPES.map((t) => (
   574→              <SelectItem key={t} value={t}>
   575→                {t.replace(/_/g, " ")}
   576→              </SelectItem>
   577→            ))}
   578→          </SelectContent>
   579→        </Select>
   580→        <Select
   581→          value={status || "ALL"}
   582→          onValueChange={(v) => {
   583→            setStatus(v === "ALL" ? "" : v);
   584→            setPage(1);
   585→          }}
   586→        >
   587→          <SelectTrigger className="md:w-48">
   588→            <SelectValue placeholder="All statuses" />
   589→          </SelectTrigger>
   590→          <SelectContent>
   591→            <SelectItem value="ALL">All statuses</SelectItem>
   592→            {DOC_STATUSES.map((s) => (
   593→              <SelectItem key={s} value={s}>
   594→                {s.replace(/_/g, " ")}
   595→              </SelectItem>
   596→            ))}
   597→          </SelectContent>
   598→        </Select>
   599→      </div>
   600→
   601→      <div className="text-sm text-muted-foreground">
   602→        Showing <span className="font-medium text-foreground">{docs.length}</span>{" "}
   603→        of <span className="font-medium text-foreground">{total}</span>{" "}
   604→        documents
   605→      </div>

  1025→function DocumentPreviewDialog({
  1026→  doc,
  1027→  onClose,
  1028→}: {
  1029→  doc: any | null;
  1030→  onClose: () => void;
  1031→}) {
  1032→  const [preview, setPreview] = useState<any | null>(null);
  1033→  const [loading, setLoading] = useState(false);
  1034→
  1035→  useEffect(() => {
  1036→    if (!doc) {
  1037→      // Defer the reset to avoid synchronous setState in effect body.
  1038→      const t = setTimeout(() => {
  1039→        setPreview(null);
  1040→        setLoading(false);
  1041→      }, 0);
  1042→      return () => clearTimeout(t);
  1043→    }
  1044→    let cancelled = false;
  1045→    // Use queueMicrotask so the rule about synchronous setState in effects
  1046→    // doesn't fire — we genuinely want to flip loading=true before the fetch.
  1047→    queueMicrotask(() => {
  1048→      if (!cancelled) setLoading(true);
  1049→    });
  1050→    fetch(`/api/documents/${doc.id}/preview`)
  1051→      .then((r) => r.json())
  1052→      .then((d) => {
  1053→        if (!cancelled) setPreview(d);
  1054→      })
  1055→      .catch(() => toast.error("Failed to load preview"))
  1056→      .finally(() => {
  1057→        if (!cancelled) setLoading(false);
  1058→      });
  1059→    return () => {
  1060→      cancelled = true;
  1061→    };
  1062→  }, [doc]);
  1063→
  1064→  return (
  1065→    <Dialog open={!!doc} onOpenChange={(o) => !o && onClose()}>
  1066→      <DialogContent className="max-w-4xl max-h-[92vh] p-0 gap-0">

  1164→function DirectSendEmailDialog({
  1165→  doc,

  1177→  const [body, setBody] = useState("");

  1202→        setBody("");

  1205→
  1206→  async function handleSend() {

  1209→      toast.error("To, subject and body are required.");
  1210→      return;
  1211→    }

   131→function OrganizationTab() {
   132→  const qc = useQueryClient();
   133→  const [saving, setSaving] = useState(false);
   134→  const [form, setForm] = useState<any>({
   135→    name: "",
   136→    legalName: "",
   137→    address: "",
   138→    city: "",
   139→    state: "",
   140→    country: "",
   141→    zipCode: "",
   142→    email: "",
   143→    phone: "",
   144→    website: "",
   145→    logo: "",
   146→    taxId: "",
   147→  });
   148→
   149→  const { data, isLoading } = useQuery({
   150→    queryKey: ["company"],
   151→    queryFn: () => fetch("/api/company").then((r) => r.json()),
   152→  });
   153→
   154→  useEffect(() => {
   155→    if (data) {
   156→      setForm({
   157→        name: data.name ?? "",
   158→        legalName: data.legalName ?? "",
   159→        address: data.address ?? "",
   160→        city: data.city ?? "",
   161→        state: data.state ?? "",
   162→        country: data.country ?? "",
   163→        zipCode: data.zipCode ?? "",
   164→        email: data.email ?? "",
   165→        phone: data.phone ?? "",
   166→        website: data.website ?? "",
   167→        logo: data.logo ?? "",
   168→        taxId: data.taxId ?? "",
   169→      });
   170→    }
   171→  }, [data]);
   172→
   173→  async function save() {
   174→    if (!form.name) {
   175→      toast.error("Company name is required");
   176→      return;
   177→    }
   178→    setSaving(true);
   179→    try {
   180→      const res = await fetch("/api/company", {
   181→        method: "PATCH",
   182→        headers: { "Content-Type": "application/json" },
   183→        body: JSON.stringify(form),
   184→      });
   185→      if (!res.ok) throw new Error();
   186→      toast.success("Company profile saved");
   187→      qc.invalidateQueries({ queryKey: ["company"] });
   188→    } catch {
   189→      toast.error("Failed to save company");
   190→    } finally {
   191→      setSaving(false);
   192→    }
   193→  }
   194→
   195→  if (isLoading) {
   196→    return (
   197→      <Card className="border-border/60 shadow-soft">
   198→        <CardContent className="p-6">
   199→          <div className="h-96 rounded-lg bg-muted/40 animate-pulse" />
   200→        </CardContent>
   201→      </Card>
   202→    );
   203→  }
   204→
   205→  const fieldGroups = [
   206→    {
   207→      title: "Legal Details",
   208→      icon: Building2,
   209→      fields: [
   210→        { key: "name", label: "Company Name *", full: true },
   211→        { key: "legalName", label: "Legal Name" },
   212→        { key: "taxId", label: "Tax ID / TIN" },
   213→      ],
   214→    },
   215→    {
   216→      title: "Contact Information",
   217→      icon: Mail,
   218→      fields: [
   219→        { key: "email", label: "Email" },
   220→        { key: "phone", label: "Phone" },
   221→        { key: "website", label: "Website" },
   222→        { key: "logo", label: "Logo URL", full: true },
   223→      ],
   224→    },
   225→    {
   226→      title: "Location",
   227→      icon: MapPin,
   228→      fields: [
   229→        { key: "address", label: "Address", full: true, textarea: true },
   230→        { key: "city", label: "City" },
   231→        { key: "state", label: "State / Province" },
   232→        { key: "country", label: "Country" },
   233→        { key: "zipCode", label: "ZIP / Postal Code" },
   234→      ],
   235→    },
   236→  ];
   237→
   238→  return (
   239→    <div className="space-y-4">
   240→      <Card className="border-border/60 shadow-soft">
   241→        <CardContent className="p-6 space-y-5">
   242→          {/* Header with logo preview */}
   243→          <div className="flex items-start justify-between pb-4 border-b border-border/60 gap-4">
   244→            <div>
   245→              <div className="font-semibold text-base">Organization Profile</div>
   246→              <p className="text-xs text-muted-foreground mt-0.5">
   247→                These details appear on documents, emails, and reports.
   248→              </p>
   249→            </div>
   250→            <div className="flex items-center gap-3">
   251→              <div className="size-14 rounded-xl bg-primary/10 border border-border/60 flex items-center justify-center overflow-hidden flex-shrink-0">
   252→                {form.logo ? (
   253→                  <img
   254→                    src={form.logo}
   255→                    alt="Logo"
   256→                    className="size-full object-contain p-1"
   257→                    onError={(e) => {
   258→                      (e.target as HTMLImageElement).style.display = "none";
   259→                    }}
   260→                  />
   261→                ) : (
   262→                  <Building2 className="size-6 text-primary" />
   263→                )}
   264→              </div>
   265→            </div>
   266→          </div>
   267→
   268→          {/* Grouped fields */}
   269→          {fieldGroups.map((group) => {
   270→            const GroupIcon = group.icon;
   271→            return (
   272→              <div key={group.title} className="space-y-3">
   273→                <div className="flex items-center gap-2 text-sm font-semibold text-foreground">
   274→                  <div className="size-7 rounded-lg bg-primary/10 text-primary flex items-center justify-center">
   275→                    <GroupIcon className="size-4" />
   276→                  </div>
   277→                  {group.title}
   278→                </div>
   279→                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 pl-9">
   280→                  {group.fields.map((f) => (
   281→                    <div
   282→                      key={f.key}
   283→                      className={cn("space-y-1.5", f.full && "sm:col-span-2")}
   284→                    >
   285→                      <Label className="text-xs text-muted-foreground">
   286→                        {f.label}
   287→                      </Label>
   288→                      {f.textarea ? (
   289→                        <Textarea
   290→                          rows={2}
   291→                          value={form[f.key]}
   292→                          onChange={(e) =>
   293→                            setForm({ ...form, [f.key]: e.target.value })
   294→                          }
   295→                        />
   296→                      ) : (
   297→                        <Input
   298→                          value={form[f.key]}
   299→                          onChange={(e) =>
   300→                            setForm({ ...form, [f.key]: e.target.value })
   301→                          }
   302→                        />
   303→                      )}
   304→                    </div>
   305→                  ))}
   306→                </div>
   307→              </div>
   308→            );
   309→          })}
   310→        </CardContent>
   311→      </Card>
   312→
   313→      {/* Sticky save bar */}
   314→      <div className="sticky bottom-4 z-10 flex justify-end">
   315→        <div className="inline-flex items-center gap-3 rounded-xl border border-border/60 bg-card/95 backdrop-blur-md shadow-card-hover px-4 py-2.5">
   316→          <span className="text-xs text-muted-foreground hidden sm:inline">
   317→            Changes apply to all generated documents
   318→          </span>
   319→          <Button onClick={save} disabled={saving} size="sm">
   320→            {saving ? (
   321→              "Saving…"
   322→            ) : (
   323→              <>
   324→                <Save className="size-4 mr-1.5" /> Update Profile
   325→              </>
   326→            )}
   327→          </Button>
   328→        </div>
   329→      </div>
   330→    </div>
   331→  );
   332→}

  1294→function SimpleFormDialog({
  1295→  title,
  1296→  open,
  1297→  onOpenChange,
  1298→  fields,
  1299→  initial,
  1300→  onSubmit,
  1301→}: {
  1302→  title: string;
  1303→  open: boolean;
  1304→  onOpenChange: (v: boolean) => void;
  1305→  fields: Field[];
  1306→  initial: any;
  1307→  onSubmit: (form: any) => Promise<void>;
  1308→}) {
  1309→  const [form, setForm] = useState<any>(initial);
  1310→  const [saving, setSaving] = useState(false);
  1311→
  1312→  useEffect(() => {
  1313→    setForm(initial);
  1314→  }, [initial]);
  1315→
  1316→  async function submit() {
  1317→    for (const f of fields) {
  1318→      if (f.required && !form[f.key]) {
  1319→        toast.error(`${f.label} is required`);
  1320→        return;
  1321→      }
  1322→    }
  1323→    setSaving(true);
  1324→    try {
  1325→      await onSubmit(form);
  1326→      toast.success("Saved");
  1327→      onOpenChange(false);
  1328→    } catch {
  1329→      toast.error("Failed to save");
  1330→    } finally {
  1331→      setSaving(false);
  1332→    }
  1333→  }
  1334→
  1335→  return (
  1336→    <Dialog open={open} onOpenChange={onOpenChange}>
  1337→      <DialogContent className="max-w-[95vw] sm:max-w-lg">
  1338→        <DialogHeader>
  1339→          <DialogTitle>{title}</DialogTitle>
  1340→        </DialogHeader>
  1341→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
  1342→          {fields.map((f) => (
  1343→            <div
  1344→              key={f.key}
  1345→              className={cn("space-y-1.5", f.textarea && "sm:col-span-2")}
  1346→            >
  1347→              <Label>{f.label}</Label>
  1348→              {f.textarea ? (
  1349→                <Textarea
  1350→                  rows={2}
  1351→                  value={form[f.key] ?? ""}
  1352→                  onChange={(e) =>
  1353→                    setForm({ ...form, [f.key]: e.target.value })
  1354→                  }
  1355→                />
  1356→              ) : f.type === "select" ? (
  1357→                <Select
  1358→                  value={form[f.key] || (f.allowNone ? "NONE" : "")}
  1359→                  onValueChange={(v) =>
  1360→                    setForm({
  1361→                      ...form,
  1362→                      [f.key]: f.allowNone && v === "NONE" ? "" : v,
  1363→                    })
  1364→                  }
  1365→                >
  1366→                  <SelectTrigger>
  1367→                    <SelectValue placeholder="Select…" />
  1368→                  </SelectTrigger>
  1369→                  <SelectContent>
  1370→                    {f.allowNone && <SelectItem value="NONE">None</SelectItem>}
  1371→                    {f.options?.map((o) => (
  1372→                      <SelectItem key={o.value} value={o.value}>
  1373→                        {o.label}
  1374→                      </SelectItem>
  1375→                    ))}
  1376→                  </SelectContent>
  1377→                </Select>
  1378→              ) : f.type === "switch" ? (
  1379→                <div className="flex items-center gap-2 pt-1">
  1380→                  <Switch
  1381→                    checked={!!form[f.key]}
  1382→                    onCheckedChange={(v) => setForm({ ...form, [f.key]: v })}
  1383→                  />
  1384→                  <span className="text-sm text-muted-foreground">
  1385→                    {form[f.key] ? "Enabled" : "Disabled"}
  1386→                  </span>
  1387→                </div>
  1388→              ) : f.type === "color" ? (
  1389→                <div className="flex items-center gap-2">
  1390→                  <input
  1391→                    type="color"
  1392→                    value={form[f.key] ?? "#10b981"}
  1393→                    onChange={(e) =>
  1394→                      setForm({ ...form, [f.key]: e.target.value })
  1395→                    }
  1396→                    className="size-9 rounded-md border border-border cursor-pointer"
  1397→                  />
  1398→                  <Input
  1399→                    value={form[f.key] ?? ""}
  1400→                    onChange={(e) =>
  1401→                      setForm({ ...form, [f.key]: e.target.value })
  1402→                    }
  1403→                    className="font-mono text-xs"
  1404→                  />
  1405→                </div>
  1406→              ) : (
  1407→                <Input
  1408→                  type={f.type === "number" ? "number" : "text"}
  1409→                  value={form[f.key] ?? ""}
  1410→                  onChange={(e) =>
  1411→                    setForm({
  1412→                      ...form,
  1413→                      [f.key]:
  1414→                        f.type === "number"
  1415→                          ? Number(e.target.value)
  1416→                          : e.target.value,
  1417→                    })
  1418→                  }
  1419→                />
  1420→              )}
  1421→            </div>
  1422→          ))}
  1423→        </div>
  1424→        <DialogFooter>
  1425→          <Button variant="outline" onClick={() => onOpenChange(false)}>
  1426→            Cancel
  1427→          </Button>
  1428→          <Button onClick={submit} disabled={saving}>
  1429→            {saving ? "Saving…" : "Save"}
  1430→          </Button>
  1431→        </DialogFooter>
  1432→      </DialogContent>
  1433→    </Dialog>
  1434→  );
  1435→}

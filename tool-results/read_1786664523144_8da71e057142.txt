   240→   240→function AllDocumentsTab({
   241→   241→  onPreview,
   242→   242→  onSendEmail,
   243→   243→}: {
   244→   244→  onPreview: (doc: any) => void;
   245→   245→  onSendEmail: (doc: any) => void;
   246→   246→}) {
   247→   247→  const { data, isLoading } = useQuery({
   248→   248→    queryKey: ["documents", "all-tab"],
   249→   249→    queryFn: async () => {
   250→   250→      const r = await fetch("/api/documents?pageSize=10");
   251→   251→      return r.json();
   252→   252→    },
   253→   253→  });
   254→   254→
   255→   255→  const { data: emailStats } = useQuery({
   256→   256→    queryKey: ["email-logs", "stats"],
   257→   257→    queryFn: async () => {
   258→   258→      const [sent, failed] = await Promise.all([
   259→   259→        fetch("/api/email-logs?status=SENT&pageSize=1").then((r) => r.json()),
   260→   260→        fetch("/api/email-logs?status=FAILED&pageSize=1").then((r) => r.json()),
   261→   261→      ]);
   262→   262→      return { sentToday: sent.total ?? 0, failed: failed.total ?? 0 };
   263→   263→    },
   264→   264→  });
   265→   265→
   266→   266→  const docs = data?.items ?? [];
   267→   267→  const total = data?.total ?? 0;
   268→   268→  const todayStr = new Date().toDateString();
   269→   269→  const generatedToday = docs.filter(
   270→   270→    (d: any) => new Date(d.createdAt).toDateString() === todayStr
   271→   271→  ).length;
   272→   272→  const sentToday = emailStats?.sentToday ?? 0;
   273→   273→  const failedEmails = emailStats?.failed ?? 0;
   274→   274→  const pendingApproval = docs.filter((d: any) =>
   275→   275→    ["PENDING_APPROVAL", "GENERATED"].includes(d.status)
   276→   276→  ).length;
   277→   277→
   278→   278→  return (
   279→   279→    <div className="space-y-4">
   280→   280→      <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-5 gap-3 sm:gap-3">
   281→   281→        <KpiCard label="Total" value={total} icon={FileStack} iconClass="bg-primary/10 text-primary" />
   282→   282→        <KpiCard label="Generated Today" value={generatedToday} icon={FilePlus} iconClass="bg-violet-500/10 text-violet-600" />
   283→   283→        <KpiCard label="Sent Today" value={sentToday} icon={MailCheck} iconClass="bg-teal-500/10 text-teal-600" />
   284→   284→        <KpiCard label="Pending Approval" value={pendingApproval} icon={Clock} iconClass="bg-amber-500/10 text-amber-600" />
   285→   285→        <KpiCard label="Failed Emails" value={failedEmails} icon={MailX} iconClass="bg-rose-500/10 text-rose-600" />
   286→   286→      </div>
   287→   287→
   288→   288→      <Card className="border-border/60 shadow-soft overflow-hidden">
   289→   289→        <div className="flex items-center justify-between px-5 py-3 border-b border-border bg-muted/30">
   290→   290→          <div className="font-medium text-sm">Recent Documents</div>
   291→   291→        </div>
   292→   292→        <DocumentsTable
   293→   293→          docs={docs}
   294→   294→          loading={isLoading}
   295→   295→          onPreview={onPreview}
   296→   296→          onSendEmail={onSendEmail}
   297→   297→        />
   298→   298→      </Card>
   299→   299→    </div>
   300→   300→  );
   301→   301→}

   325→   325→      const r = await fetch(`/api/document-templates?${params.toString()}`);

   328→   328→  });
   329→   329→  const templates = data?.items ?? [];
   330→   330→
   331→   331→  async function archiveTemplate(id: string) {
   332→   332→    if (!confirm("Archive this template? It will be hidden from the active list.")) return;
   333→   333→    try {
   334→   334→      const res = await fetch(`/api/document-templates/${id}`, {
   335→   335→        method: "DELETE",
   336→   336→      });
   337→   337→      if (!res.ok) throw new Error("Failed to archive");
   338→   338→      toast.success("Template archived.");
   339→   339→      qc.invalidateQueries({ queryKey: ["document-templates"] });
   340→   340→    } catch (err: any) {
   341→   341→      toast.error(err.message);
   342→   342→    }

   345→   345→  async function duplicateTemplate(t: any) {

   347→   347→      const res = await fetch("/api/document-templates", {

   349→   349→        headers: { "Content-Type": "application/json" },
   350→   350→        body: JSON.stringify({
   351→   351→          ...t,
   352→   352→          name: `${t.name} (Copy)`,
   353→   353→          code: `${t.code}-COPY`,
   354→   354→          status: "DRAFT",
   355→   355→        }),
   356→   356→      });
   357→   357→      if (!res.ok) {
   358→   358→        const err = await res.json().catch(() => ({}));
   359→   359→        throw new Error(err.error || "Failed");
   360→   360→      }
   361→   361→      toast.success("Template duplicated.");
   362→   362→      qc.invalidateQueries({ queryKey: ["document-templates"] });

   365→   365→    }

  1296→  1296→      if (!cancelled) setLoading(true);

  1298→  1298→    fetch(`/api/documents/${doc.id}/preview`)

  1303→  1303→      .catch(() => toast.error("Failed to load preview"))
  1304→  1304→      .finally(() => {
  1305→  1305→        if (!cancelled) setLoading(false);
  1306→  1306→      });
  1307→  1307→    return () => {
  1308→  1308→      cancelled = true;
  1309→  1309→    };

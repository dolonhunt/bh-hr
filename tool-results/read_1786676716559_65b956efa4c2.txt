   327→export function AssetsModule() {
   328→  const [search, setSearch] = useState("");
   329→  const [type, setType] = useState("");
   330→  const [status, setStatus] = useState("");
   331→  const [view, setView] = useState<"table" | "grid" | "depreciation">("table");
   332→
   333→  const [formOpen, setFormOpen] = useState(false);
   334→  const [editAsset, setEditAsset] = useState<Asset | null>(null);
   335→  const [assignAsset, setAssignAsset] = useState<Asset | null>(null);
   336→  const [returnAsset, setReturnAsset] = useState<Asset | null>(null);
   337→  const [depreciationAsset, setDepreciationAsset] = useState<DepreciationRow | null>(null);
   338→  const [maintenanceAsset, setMaintenanceAsset] = useState<Asset | null>(null);
   339→
   340→  const qc = useQueryClient();
   341→  const { data, isLoading, isError } = useQuery({
   342→    queryKey: ["assets", search, type, status],
   343→    queryFn: async () => {
   344→      const params = new URLSearchParams();
   345→      if (search) params.set("search", search);
   346→      if (type) params.set("type", type);
   347→      if (status) params.set("status", status);
   348→      const r = await fetch(`/api/assets?${params.toString()}`);
   349→      if (!r.ok) throw new Error("Failed to load assets");
   350→      return r.json();
   351→    },
   352→  });
   353→
   354→  const assets: Asset[] = data?.items ?? [];
   355→  const total = assets.length;
   356→  const assigned = assets.filter((a) => a.status === "ASSIGNED").length;
   357→  const available = assets.filter((a) => a.status === "AVAILABLE").length;
   358→  const damaged = assets.filter((a) => a.condition === "DAMAGED").length;
   359→
   360→  // Global maintenance summary for the Maintenance KPI card (table + grid views only).
   361→  const maintenanceSummaryQ = useQuery({
   362→    queryKey: ["assets", "maintenance", "summary"],
   363→    queryFn: async () => {
   364→      const r = await fetch("/api/assets/maintenance");
   365→      if (!r.ok) throw new Error("Failed to load maintenance summary");
   366→      return r.json();
   367→    },
   368→    enabled: view !== "depreciation",
   369→  });
   370→  const maintenanceSummary: GlobalMaintenanceSummary | undefined =
   371→    maintenanceSummaryQ.data?.summary;
   372→
   373→  function openCreate() {
   374→    setEditAsset(null);
   375→    setFormOpen(true);
   376→  }
   377→  function openEdit(a: Asset) {
   378→    setEditAsset(a);
   379→    setFormOpen(true);
   380→  }
   381→  async function deleteAsset(a: Asset) {
   382→    if (!confirm(`Permanently delete asset "${a.name}"?`)) return;
   383→    try {
   384→      const r = await fetch(`/api/assets/${a.id}`, { method: "DELETE" });
   385→      if (!r.ok) throw new Error("Failed to delete asset");
   386→      toast.success(`Asset "${a.name}" deleted.`);
   387→      qc.invalidateQueries({ queryKey: ["assets"] });
   388→    } catch (err: any) {
   389→      toast.error(err?.message || "Failed to delete asset.");
   390→    }
   391→  }
   392→  async function retireAsset(a: Asset) {
   393→    if (!confirm(`Retire asset "${a.name}"? It will be marked RETIRED.`))
   394→      return;
   395→    try {
   396→      const r = await fetch(`/api/assets/${a.id}`, {
   397→        method: "PATCH",
   398→        headers: { "Content-Type": "application/json" },
   399→        body: JSON.stringify({ status: "RETIRED" }),
   400→      });
   401→      if (!r.ok) throw new Error("Failed to retire asset");
   402→      toast.success(`Asset "${a.name}" retired.`);
   403→      qc.invalidateQueries({ queryKey: ["assets"] });
   404→    } catch (err: any) {
   405→      toast.error(err?.message || "Failed to retire asset.");
   406→    }
   407→  }
   408→
   409→  return (
   410→    <div className="space-y-6">
   411→      <PageHeader
   412→        title="Asset Management"
   413→        description="Track company assets and equipment assignments"
   414→        icon={<Package className="size-5" />}
   415→        actions={
   416→          <>
   417→            <ExportButton
   418→              module="assets"
   419→              filters={{ search, status, type }}
   420→            />
   421→            <div className="flex rounded-md border border-border overflow-hidden">
   422→              <Button
   423→                variant={view === "table" ? "default" : "ghost"}
   424→                size="sm"
   425→                className="rounded-none gap-1.5"
   426→                onClick={() => setView("table")}
   427→                aria-label="Table view"
   428→              >
   429→                <List className="size-4" />
   430→                <span className="hidden sm:inline">Table</span>
   431→              </Button>
   432→              <Button
   433→                variant={view === "grid" ? "default" : "ghost"}
   434→                size="sm"
   435→                className="rounded-none gap-1.5"
   436→                onClick={() => setView("grid")}
   437→                aria-label="Grid view"
   438→              >
   439→                <LayoutGrid className="size-4" />
   440→                <span className="hidden sm:inline">Grid</span>
   441→              </Button>
   442→              <Button
   443→                variant={view === "depreciation" ? "default" : "ghost"}
   444→                size="sm"
   445→                className="rounded-none gap-1.5"
   446→                onClick={() => setView("depreciation")}
   447→                aria-label="Depreciation view"
   448→              >
   449→                <TrendingDown className="size-4" />
   450→                <span className="hidden sm:inline">Depreciation</span>
   451→              </Button>
   452→            </div>
   453→            <Button size="sm" onClick={openCreate} className="gap-1.5">
   454→              <Plus className="size-4" />
   455→              <span className="hidden sm:inline">Add Asset</span>
   456→              <span className="sm:hidden">New</span>
   457→            </Button>
   458→          </>
   459→        }
   460→      />
   461→
   462→      {/* KPI cards */}
   463→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3 md:gap-4">
   464→        <KpiCard
   465→          label="Total Assets"
   466→          value={total}
   467→          icon={Package}
   468→          iconClass="bg-primary/10 text-primary"
   469→          footer={<span className="text-muted-foreground">All equipment</span>}
   470→        />
   471→        <KpiCard
   472→          label="Assigned"
   473→          value={assigned}
   474→          icon={PackageCheck}
   475→          iconClass="bg-sky-500/15 text-sky-600"
   476→          footer={<span className="text-muted-foreground">In use</span>}
   477→        />
   478→        <KpiCard
   479→          label="Available"
   480→          value={available}
   481→          icon={PackageOpen}
   482→          iconClass="bg-emerald-500/15 text-emerald-600"
   483→          footer={<span className="text-muted-foreground">Free to assign</span>}
   484→        />
   485→        <KpiCard
   486→          label="Damaged"
   487→          value={damaged}
   488→          icon={AlertTriangle}
   489→          iconClass="bg-rose-500/15 text-rose-600"
   490→          footer={<span className="text-muted-foreground">Needs repair</span>}
   491→        />
   492→      </div>
   493→
   494→      {/* Maintenance summary card (hidden in depreciation view) */}
   495→      {view !== "depreciation" && (
   496→        <MaintenanceSummaryCard
   497→          summary={maintenanceSummary}
   498→          isLoading={maintenanceSummaryQ.isLoading}
   499→          isError={maintenanceSummaryQ.isError}
   500→          onRetry={() =>
   501→            qc.invalidateQueries({ queryKey: ["assets", "maintenance", "summary"] })
   502→          }
   503→        />
   504→      )}
   505→
   506→      {/* Filter bar */}
   507→      <div className="flex flex-col md:flex-row gap-3">
   508→        <div className="relative flex-1">
   509→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   510→          <Input
   511→            placeholder="Search by name, serial, or assignee…"
   512→            value={search}
   513→            onChange={(e) => setSearch(e.target.value)}
   514→            className="pl-9"
   515→          />
   516→        </div>
   517→        <Select
   518→          value={type || "ALL"}
   519→          onValueChange={(v) => setType(v === "ALL" ? "" : v)}
   520→        >
   521→          <SelectTrigger className="md:w-44">
   522→            <SelectValue placeholder="All types" />
   523→          </SelectTrigger>
   524→          <SelectContent>
   525→            <SelectItem value="ALL">All types</SelectItem>
   526→            {ASSET_TYPES.map((t) => (
   527→              <SelectItem key={t.value} value={t.value}>
   528→                {t.label}
   529→              </SelectItem>
   530→            ))}
   531→          </SelectContent>
   532→        </Select>
   533→        <Select
   534→          value={status || "ALL"}
   535→          onValueChange={(v) => setStatus(v === "ALL" ? "" : v)}
   536→        >
   537→          <SelectTrigger className="md:w-44">
   538→            <SelectValue placeholder="All statuses" />
   539→          </SelectTrigger>
   540→          <SelectContent>
   541→            <SelectItem value="ALL">All statuses</SelectItem>
   542→            {STATUSES.map((s) => (
   543→              <SelectItem key={s} value={s}>
   544→                {s.charAt(0) + s.slice(1).toLowerCase()}
   545→              </SelectItem>
   546→            ))}
   547→          </SelectContent>
   548→        </Select>
   549→      </div>
   550→
   551→      {/* Result count */}
   552→      <div className="flex items-center justify-between text-sm">
   553→        <div className="text-muted-foreground">
   554→          {isLoading ? (
   555→            <Skeleton className="h-4 w-32" />
   556→          ) : (
   557→            <span>
   558→              Showing <span className="font-semibold text-foreground">{assets.length}</span>{" "}
   559→              asset{assets.length === 1 ? "" : "s"}
   560→              {(type || status || search) && " (filtered)"}
   561→            </span>
   562→          )}
   563→        </div>
   564→      </div>
   565→
   566→      {/* Content */}
   567→      {view === "depreciation" ? (
   568→        <DepreciationView
   569→          search={search}
   570→          type={type}
   571→          status={status}
   572→          onInspect={(row) => setDepreciationAsset(row)}
   573→        />
   574→      ) : isLoading ? (
   575→        <AssetsSkeleton view={view} />
   576→      ) : isError ? (
   577→        <EmptyState
   578→          icon={AlertTriangle}
   579→          title="Failed to load assets"

  1107→function AssetFormBody({
  1108→  asset,
  1109→  savingState,
  1110→  onSaved,
  1111→  onCancel,
  1112→}: {
  1113→  asset: Asset | null;
  1114→  savingState: [boolean, (b: boolean) => void];
  1115→  onSaved: () => void;
  1116→  onCancel: () => void;
  1117→}) {
  1118→  const [name, setName] = useState(asset?.name ?? "");
  1119→  const [type, setType] = useState<AssetType>(asset?.type ?? "LAPTOP");
  1120→  const [serialNumber, setSerialNumber] = useState(asset?.serialNumber ?? "");
  1121→  const [condition, setCondition] = useState<AssetCondition>(
  1122→    asset?.condition ?? "GOOD"
  1123→  );
  1124→  const [purchaseValue, setPurchaseValue] = useState<string>(
  1125→    asset?.purchaseValue !== undefined && asset.purchaseValue !== null
  1126→      ? String(asset.purchaseValue)
  1127→      : "1000"
  1128→  );
  1129→  const [notes, setNotes] = useState(asset?.notes ?? "");
  1130→  const [saving, setSaving] = savingState;
  1131→
  1132→  async function submit() {
  1133→    if (!name.trim()) {
  1134→      toast.error("Asset name is required.");
  1135→      return;
  1136→    }
  1137→    const pvNum = Number(purchaseValue);
  1138→    const pv =
  1139→      purchaseValue.trim() !== "" && !isNaN(pvNum) && pvNum >= 0
  1140→        ? pvNum
  1141→        : 1000;
  1142→    setSaving(true);
  1143→    try {
  1144→      const body = { name, type, serialNumber, condition, purchaseValue: pv, notes };
  1145→      const r = asset
  1146→        ? await fetch(`/api/assets/${asset.id}`, {
  1147→            method: "PATCH",
  1148→            headers: { "Content-Type": "application/json" },
  1149→            body: JSON.stringify(body),
  1150→          })
  1151→        : await fetch("/api/assets", {
  1152→            method: "POST",
  1153→            headers: { "Content-Type": "application/json" },
  1154→            body: JSON.stringify(body),
  1155→          });
  1156→      if (!r.ok) {
  1157→        const e = await r.json().catch(() => ({}));
  1158→        throw new Error(e?.error || "Failed to save asset");
  1159→      }
  1160→      onSaved();
  1161→    } catch (err: any) {
  1162→      toast.error(err?.message || "Failed to save asset.");
  1163→    } finally {
  1164→      setSaving(false);
  1165→    }
  1166→  }
  1167→
  1168→  return (
  1169→    <div className="space-y-4">
  1170→      <div className="space-y-1.5">
  1171→        <Label htmlFor="asset-name">Asset name *</Label>
  1172→        <Input
  1173→          id="asset-name"
